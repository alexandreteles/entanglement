8impl<'a> ProgramGraph<'a> {
9    pub(crate) fn scopes(&self, file_index: usize) -> impl Iterator<Item = ProgramScope> + '_ {
10        self.scopes_by_file[file_index]
11            .iter()
12            .map(|index| self.scopes[*index])
13            .filter(move |scope| self.file(file_index).allows(scope.id.context_id))
14    }
15
16    pub(crate) fn scope_at(
17        &self,
18        file_index: usize,
19        context_id: usize,
20        position: usize,
21    ) -> Option<ProgramScopeId> {
22        self.scopes(file_index)
23            .map(|scope| scope.id)
24            .filter(|scope| {
25                scope.context_id == context_id
26                    && scope.start_byte <= position
27                    && position < scope.end_byte
28            })
29            .min_by_key(|scope| scope.end_byte - scope.start_byte)
30    }
31
32    fn scope(&self, id: ProgramScopeId) -> Option<ProgramScope> {
33        self.scope_by_id
34            .get(&id)
35            .filter(|_| self.file(id.file_index).allows(id.context_id))
36            .map(|index| self.scopes[*index])
37    }
38
39    fn within_callable_scope(
40        &self,
41        file_index: usize,
42        context_id: usize,
43        position: usize,
44        start_byte: usize,
45        end_byte: usize,
46    ) -> bool {
47        let nearest = self
48            .scope_at(file_index, context_id, position)
49            .and_then(|id| self.scope(id));
50        std::iter::successors(nearest, |scope| self.scope(scope.parent?)).any(|scope| {
51            let current = scope.id;
52            current.start_byte < position
53                && current.start_byte <= start_byte
54                && end_byte <= current.end_byte
55                && self.callable_scopes[file_index].contains(&current)
56        })
57    }
58
59    pub(crate) fn is_local_definition(&self, file_index: usize, definition: &Definition) -> bool {
60        self.within_callable_scope(
61            file_index,
62            definition.context_id,
63            definition.start_byte,
64            definition.scope_start,
65            definition.scope_end,
66        )
67    }
68
69    pub(crate) fn is_local_scope(
70        &self,
71        file_index: usize,
72        context_id: usize,
73        scope_start: usize,
74        scope_end: usize,
75        position: usize,
76    ) -> bool {
77        self.within_callable_scope(file_index, context_id, position, scope_start, scope_end)
78    }
79}
80impl<'a> ProgramGraph<'a> {
81    pub(crate) fn bindings_named(
82        &self,
83        file_index: usize,
84        name: &str,
85        context_id: usize,
86        position: usize,
87    ) -> impl Iterator<Item = &ProgramBinding<'a>> {
88        self.bindings_by_file_name[file_index]
89            .get(name)
90            .into_iter()
91            .flatten()
92            .map(|index| &self.bindings[*index])
93            .filter(move |binding| {
94                self.file(file_index).allows(binding.scope.context_id)
95                    && binding.scope.context_id == context_id
96                    && binding.local.start_byte <= position
97                    && binding.scope.start_byte <= position
98                    && position < binding.scope.end_byte
99            })
100    }
101
102    /// Return the unique nearest binding for a name at a source position.
103    /// A narrower lexical scope wins; declarations in that same scope use
104    /// the latest declaration before the position. Equal-ranked alternatives
105    /// are ambiguous and do not provide evidence.
106    pub(super) fn binding_at(
107        &self,
108        file_index: usize,
109        name: &str,
110        context_id: usize,
111        position: usize,
112    ) -> Option<&ProgramBinding<'a>> {
113        let mut nearest = self
114            .bindings_named(file_index, name, context_id, position)
115            .collect::<Vec<_>>();
116        let narrowest = nearest.iter().map(|binding| binding.scope.width()).min()?;
117        nearest.retain(|binding| binding.scope.width() == narrowest);
118        let scope = nearest.first()?.scope;
119        if nearest.iter().any(|binding| {
120            binding.scope.start_byte != scope.start_byte || binding.scope.end_byte != scope.end_byte
121        }) {
122            return None;
123        }
124        let latest = nearest
125            .iter()
126            .map(|binding| binding.local.start_byte)
127            .max()?;
128        let mut winners = nearest
129            .into_iter()
130            .filter(|binding| binding.local.start_byte == latest);
131        let winner = winners.next()?;
132        (winners.next().is_none() && !winner.conflicting_evidence).then_some(winner)
133    }
134
135    pub(crate) fn has_local_binding(
136        &self,
137        file_index: usize,
138        name: &str,
139        context_id: usize,
140        position: usize,
141    ) -> bool {
142        self.bindings_named(file_index, name, context_id, position)
143            .next()
144            .is_some()
145    }
146}
147
148pub(super) fn binding_identity<'a>(
149    binding: &ProgramBinding<'a>,
150) -> (ProgramScopeId, &'a str, usize, usize) {
151    (
152        binding.scope,
153        binding.local.name.as_str(),
154        binding.local.start_byte,
155        binding.local.end_byte,
156    )
157}
158impl<'a> ProgramGraph<'a> {
159    /// Resolve a binding's declared type, following only bounded immutable
160    /// aliases that are visible at each alias declaration in the same
161    /// semantic context.
162    pub(crate) fn binding_type(
163        &self,
164        file_index: usize,
165        name: &str,
166        context_id: usize,
167        position: usize,
168    ) -> Option<String> {
169        match self.follow_alias(file_index, name, context_id, position, true)? {
170            BindingAlias::Type(type_name) => Some(type_name),
171            BindingAlias::Path { .. } => None,
172        }
173    }
174
175    /// Follow a simple immutable alias chain and return its terminal source
176    /// path. The terminal name is left for the owning resolver to check
177    /// against its imports, package, and visibility rules.
178    pub(crate) fn binding_alias(
179        &self,
180        file_index: usize,
181        name: &str,
182        context_id: usize,
183        position: usize,
184    ) -> Option<String> {
185        match self.follow_alias(file_index, name, context_id, position, false)? {
186            BindingAlias::Path { path, .. } => Some(path),
187            BindingAlias::Type(_) => None,
188        }
189    }
190
191    /// Return the terminal path only when an immutable alias chain begins at
192    /// an explicit callable-reference value such as `::handler`.
193    pub(crate) fn binding_callable_alias(
194        &self,
195        file_index: usize,
196        name: &str,
197        context_id: usize,
198        position: usize,
199    ) -> Option<String> {
200        match self.follow_alias(file_index, name, context_id, position, false)? {
201            BindingAlias::Path {
202                path,
203                callable_reference: true,
204            } => Some(path),
205            _ => None,
206        }
207    }
208
209    fn follow_alias(
210        &self,
211        file_index: usize,
212        name: &str,
213        context_id: usize,
214        position: usize,
215        resolve_type: bool,
216    ) -> Option<BindingAlias> {
217        let mut current = self.binding_at(file_index, name, context_id, position)?;
218        let mut seen = HashSet::new();
219        let mut callable_reference = false;
220        loop {
221            if let Some(type_name) = current.type_name.as_ref().filter(|_| resolve_type) {
222                return Some(BindingAlias::Type((*type_name).to_owned()));
223            }
224            if !current.immutable || !seen.insert(binding_identity(current)) {
225                return None;
226            }
227            callable_reference |= current.callable_reference;
228            let target = current.alias_target.as_deref()?;
229            let next = (!target.is_empty() && !target.contains('.'))
230                .then(|| self.binding_at(file_index, target, context_id, current.local.end_byte))
231                .flatten()
232                .filter(|binding| {
233                    resolve_type || (binding.immutable && binding.alias_target.is_some())
234                });
235            let Some(next) = next else {
236                return (!self.has_local_binding(
237                    file_index,
238                    target.split('.').next()?,
239                    context_id,
240                    current.local.end_byte,
241                ))
242                .then(|| BindingAlias::Path {
243                    path: target.to_owned(),
244                    callable_reference,
245                });
246            };
247            current = next;
248        }
249    }
250}
251
252enum BindingAlias {
253    Type(String),
254    Path {
255        path: String,
256        callable_reference: bool,
257    },
