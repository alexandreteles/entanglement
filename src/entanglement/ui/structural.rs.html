1//! Write compact structural summaries. JSON retains complete source evidence.
2
3use std::collections::BTreeMap;
4use std::fmt::Write;
5
6use crate::structure::StructuralReport;
7
8/// Append projection, coverage, cycle, and change summaries.
9pub(super) fn write(output: &mut String, report: &StructuralReport) {
10    let state = &report.state;
11    let _ = writeln!(output, "\nStructure: {}", state.snapshot.scope);
12    for graph in &state.graphs {
13        write_graph(output, graph);
14    }
15    write_coverage(output, &state.snapshot.coverage);
16    let _ = writeln!(
17        output,
18        "  Effective declarations: {}",
19        state.effective.len()
20    );
21    for declaration in &state.effective {
22        write_effective(output, declaration);
23    }
24    if let Some(delta) = &report.delta {
25        write_events(output, delta);
26    }
27}

29fn write_graph(output: &mut String, graph: &crate::structure::model::GraphAnalysis) {
30    let pc = graph
31        .pc
32        .value
33        .map_or_else(|| "null".to_owned(), |value| format!("{value:.4}"));
34    let _ = writeln!(
35        output,
36        "  {:?}/{:?}: units {}  edges {}  SCC {}  PC {} ({}/{})",
37        graph.granularity,
38        graph.projection,
39        graph.units.len(),
40        graph.edges.len(),
41        graph.cycles.len(),
42        pc,
43        graph.pc.numerator,
44        graph.pc.denominator
45    );
46    for cycle in &graph.cycles {
47        let _ = writeln!(output, "    cycle: {}", cycle.members.join(", "));
48        let _ = writeln!(output, "      path: {}", cycle.witness.units.join(" -> "));
49    }
50}

52fn write_coverage(output: &mut String, records: &[crate::structure::model::Coverage]) {
53    let mut coverage_counts = BTreeMap::new();
54    for coverage in records {
55        *coverage_counts
56            .entry((&coverage.language, coverage.status, coverage.effective))
57            .or_insert(0usize) += 1;
58    }
59    for ((language, status, effective), count) in coverage_counts {
60        let _ = writeln!(output, "  {language}: {status:?} {count}  OO {effective:?}");
61    }
62}

64fn write_events(output: &mut String, delta: &crate::structure::model::StructuralDelta) {
65    let _ = writeln!(output, "  Structural events: {}", delta.events.len());
66    for event in &delta.events {
67        let _ = writeln!(
68            output,
69            "    {} {:?}/{:?}: {}",
70            event.kind,
71            event.granularity,
72            event.projection,
73            event.units.join(", ")
74        );
75        if let Some(path) = event.after_path.as_ref().or(event.before_path.as_ref()) {
76            let _ = writeln!(output, "      path: {}", path.units.join(" -> "));
77            if let Some(evidence) = path.evidence.first() {
78                let _ = writeln!(
79                    output,
80                    "      source: {}:{}..{}  evidence {}",
81                    evidence.path,
82                    evidence.start,
83                    evidence.end,
84                    path.evidence.len()
85                );
86            }
87        }
88    }
89}

91fn write_effective(
92    output: &mut String,
93    declaration: &crate::structure::effective::EffectiveDeclaration,
94) {
95    let _ = writeln!(output, "    {} {:?}", declaration.unit, declaration.status);
96    for member in &declaration.members {
97        let status = if member.required {
98            "req"
99        } else if member.body.is_some() {
100            "body"
101        } else {
102            "unknown"
103        };
104        let _ = writeln!(
105            output,
106            "      {} {} <- {} [{}]",
107            member.name, status, member.origin, member.visibility
108        );
109    }
110    if !declaration.obligations.is_empty() {
111        let _ = writeln!(
112            output,
113            "      obligations: {}",
114            declaration.obligations.join(", ")
115        );
116    }
117    if !declaration.conflicts.is_empty() {
118        let _ = writeln!(
119            output,
120            "      conflicts: {}",
121            declaration.conflicts.join(", ")
122        );
123    }
124}
