19/// Analyze each supported target once and keep its logical paths as aliases.
20pub(crate) fn analyze_paths<I, T: Send>(
21    paths: I,
22    selection: Selection,
23    project: impl Fn(ParsedFile) -> T + Sync,
24) -> Result<Vec<T>>
25where
26    I: IntoIterator<Item = Result<PathBuf>>,
27    I::IntoIter: Send,
28{
29    let mut groups = BTreeMap::<PathBuf, BTreeSet<PathBuf>>::new();
30    for path in paths {
31        let path = path?;
32        groups
33            .entry(input::path_identity(&path)?)
34            .or_default()
35            .insert(path);
36    }
37
38    let parsed: Vec<_> = groups
39        .into_iter()
40        .par_bridge()
41        .map_init(
42            || Worker::new(selection).map_err(|error| error.to_string()),
43            |worker, (target, aliases)| match worker {
44                Ok(worker) => worker
45                    .analyze_target(target, aliases.into_iter().collect())
46                    .map(|file| file.map(|file| (file.facts.path.clone(), project(file)))),
47                Err(error) => Err(error.clone().into()),
48            },
49        )
50        .collect::<Result<Vec<_>>>()?;
51    let mut files: Vec<_> = parsed.into_iter().flatten().collect();
52    files.sort_by(|left, right| left.0.cmp(&right.0));
53    Ok(files.into_iter().map(|(_, file)| file).collect())
54}
