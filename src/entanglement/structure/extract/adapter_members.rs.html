3pub(super) fn member_kind(node: Node<'_>, source: &[u8]) -> Option<String> {
4    if !matches!(
5        node.kind(),
6        "method_definition" | "method_signature" | "abstract_method_signature"
7    ) {
8        return None;
9    }
10    let name = node.child_by_field_name("name")?;
11    if name.kind() == "computed_property_name" {
12        return None;
13    }
14    let mut cursor = node.walk();
15    let modifiers = node
16        .children(&mut cursor)
17        .filter(|child| !child.is_named())
18        .map(|child| child.kind())
19        .collect::<Vec<_>>();
20    let is_static = modifiers.contains(&"static");
21    if !is_static && super::capture::text(name, source) == "constructor" {
22        return Some("constructor".to_owned());
23    }
24    let accessor = modifiers
25        .into_iter()
26        .find(|kind| matches!(*kind, "get" | "set"));
27    Some(
28        match (is_static, accessor) {
29            (false, None) => "instance",
30            (true, None) => "static",
31            (false, Some("get")) => "get",
32            (false, _) => "set",
33            (true, Some("get")) => "static_get",
34            (true, _) => "static_set",
35        }
36        .to_owned(),
37    )
38}

40pub(super) fn constraints(
41    node: Node<'_>,
42    source: &[u8],
43) -> Vec<crate::structure::types::TypeExpression> {
44    if node.kind() != "type_spec" {
45        return Vec::new();
46    }
47    let Some(underlying) = node
48        .child_by_field_name("type")
49        .filter(|underlying| underlying.kind() == "interface_type")
50    else {
51        return Vec::new();
52    };
53    let mut cursor = underlying.walk();
54    underlying
55        .named_children(&mut cursor)
56        .filter(|child| child.kind() == "type_elem")
57        .map(|child| crate::structure::types::TypeExpression::from_node(child, source))
58        .collect()
59}

61pub(super) fn contract_fields(node: Node<'_>) -> Vec<Node<'_>> {
62    match node.kind() {
63        "struct_item" => node
64            .child_by_field_name("body")
65            .filter(|body| body.kind() == "ordered_field_declaration_list")
66            .map_or_else(Vec::new, |body| positional_fields(body, false)),
67        "enum_item" => {
68            let Some(body) = node.child_by_field_name("body") else {
69                return Vec::new();
70            };
71            let mut cursor = body.walk();
72            body.named_children(&mut cursor)
73                .filter_map(|variant| variant.child_by_field_name("body"))
74                .flat_map(enum_fields)
75                .collect()
76        }
77        _ => Vec::new(),
78    }
79}

81fn enum_fields(body: Node<'_>) -> Vec<Node<'_>> {
82    if body.kind() == "ordered_field_declaration_list" {
83        return positional_fields(body, true);
84    }
85    let mut cursor = body.walk();
86    body.named_children(&mut cursor)
87        .filter_map(|field| field.child_by_field_name("type"))
88        .collect()
89}

91fn positional_fields(body: Node<'_>, inherited_visibility: bool) -> Vec<Node<'_>> {
92    let mut cursor = body.walk();
93    let field_ids = body
94        .children_by_field_name("type", &mut cursor)
95        .map(|field| field.id())
96        .collect::<std::collections::BTreeSet<_>>();
97    let mut public = inherited_visibility;
98    let mut fields = Vec::new();
99    for child in body.named_children(&mut cursor) {
100        if child.kind() == "visibility_modifier" {
101            public = true;
102        }
103        if field_ids.contains(&child.id()) {
104            if public {
105                fields.push(child);
106            }
107            public = inherited_visibility;
108        }
109    }
110    fields
111}
