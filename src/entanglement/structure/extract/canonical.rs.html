3pub(super) fn canonical_types(
4    facts: &[FileFacts],
5    source_snapshot: &crate::snapshot::Snapshot,
6    symbols: &SymbolTargets,
7    relationships: &mut Vec<Relationship>,
8) {
9    for relation in crate::resolver::structural_type_relationships(facts, source_snapshot) {
10        let Some(fact) = facts
11            .iter()
12            .find(|fact| fact.analysis.path == relation.owner.file)
13        else {
14            continue;
15        };
16        let Some(source) = symbols
17            .get(&resolved_key(&relation.owner))
18            .and_then(|(_, target)| target.clone())
19        else {
20            continue;
21        };
22        let record = fact
23            .contexts
24            .iter()
25            .flat_map(|store| &store.semantic.structure)
26            .find(|record| {
27                record.is_type
28                    && record.start <= relation.owner.start_byte
29                    && relation.owner.start_byte < record.end
30            });
31        let range = record
32            .and_then(|record| {
33                record.relations.iter().find(|range| {
34                    range.kind == relation.kind
35                        && range.start <= relation.start
36                        && relation.start < range.end
37                })
38            })
39            .map(|range| (range.start, range.end))
40            .or_else(|| record.map(|record| (record.start, record.end)))
41            .unwrap_or((relation.start, relation.start));
42        let (resolution, target_module, target_type) = match relation.target {
43            Resolution::Exact(symbol) => match symbols.get(&resolved_key(&symbol)) {
44                Some((module, target)) => {
45                    (ResolutionStatus::Resolved, module.clone(), target.clone())
46                }
47                None => (ResolutionStatus::Unresolved, relation.name.clone(), None),
48            },
49            Resolution::External => (ResolutionStatus::External, relation.name.clone(), None),
50            Resolution::Ambiguous(_) => (ResolutionStatus::Ambiguous, relation.name.clone(), None),
51            Resolution::Unresolved => (ResolutionStatus::Unresolved, relation.name.clone(), None),
52        };
53        let contract = record.is_some_and(|record| record.public);
54        let evidence = vec![Evidence {
55            path: fact.analysis.path.clone(),
56            start: range.0,
57            end: range.1,
58            owner: source.clone(),
59            detail: relation.name.clone(),
60        }];
61        relationships.push(Relationship {
62            ordered: false,
63            source: module_id(&fact.analysis.path),
64            target: target_module,
65            kind: relation.kind,
66            contract,
67            resolution,
68            evidence: evidence.clone(),
69        });
70        relationships.push(Relationship {
71            ordered: false,
72            source,
73            target: target_type.unwrap_or(relation.name),
74            kind: relation.kind,
75            contract,
76            resolution,
77            evidence,
78        });
79    }
80}

82pub(super) fn canonical_modules(facts: &[FileFacts], relationships: &mut Vec<Relationship>) {
83    for (file, start, end, source, resolution) in crate::resolver::structural_modules(facts) {
84        let fact = &facts[file];
85        let module = module_id(&fact.analysis.path);
86        relationships.retain(|edge| {
87            !(edge.source == module
88                && matches!(
89                    edge.kind,
90                    RelationshipKind::Import | RelationshipKind::Reexport
91                )
92                && edge
93                    .evidence
94                    .iter()
95                    .any(|evidence| evidence.start == start && evidence.end == end))
96        });
97        let (target, status) = match resolution {
98            Resolution::Exact(symbol) => (module_id(&symbol.file), ResolutionStatus::Resolved),
99            Resolution::External => (source.clone(), ResolutionStatus::External),
100            Resolution::Ambiguous(_) => (source.clone(), ResolutionStatus::Ambiguous),
101            Resolution::Unresolved => (source.clone(), ResolutionStatus::Unresolved),
102        };
103        let reexport = fact
104            .contexts
105            .iter()
106            .flat_map(|store| &store.exports)
107            .any(|export| export.start_byte == start && export.end_byte == end);
108        relationships.push(Relationship {
109            ordered: false,
110            source: module.clone(),
111            target,
112            kind: if reexport {
113                RelationshipKind::Reexport
114            } else {
115                RelationshipKind::Import
116            },
117            contract: reexport,
118            resolution: status,
119            evidence: vec![Evidence {
120                path: fact.analysis.path.clone(),
121                start,
122                end,
123                owner: module,
124                detail: source,
125            }],
126        });
127    }
