3pub(super) fn public(
4    language: &str,
5    node: Node<'_>,
6    source: &[u8],
7    name: &str,
8    existing: Option<&crate::model::Definition>,
9) -> bool {
10    if language.contains("python") {
11        return !name.starts_with('_');
12    }
13    if language.contains("go") {
14        return name.chars().next().is_some_and(char::is_uppercase);
15    }
16    let prefix = node
17        .child_by_field_name("body")
18        .map_or(node.end_byte(), |body| body.start_byte());
19    let header = String::from_utf8_lossy(&source[node.start_byte()..prefix]);
20    if header
21        .split_whitespace()
22        .take(4)
23        .any(|word| word == "private")
24        || name.starts_with('#')
25    {
26        return false;
27    }
28    if language.contains("rust") {
29        let owner =
30            std::iter::successors(node.parent(), |ancestor| ancestor.parent()).find(|ancestor| {
31                matches!(
32                    ancestor.kind(),
33                    "trait_item" | "function_item" | "impl_item" | "source_file"
34                )
35            });
36        return owner.is_some_and(|owner| owner.kind() == "trait_item")
37            || existing.is_some_and(|definition| definition.is_public)
38            || header.starts_with("pub");
39    }
40    if language.contains("php") || language.contains("kotlin") {
41        return existing
42            .and_then(|definition| definition.declaration.as_ref())
43            .is_none_or(|declaration| {
44                matches!(
45                    declaration.visibility,
46                    crate::graph::SemanticVisibility::Public
47                        | crate::graph::SemanticVisibility::Protected
48                        | crate::graph::SemanticVisibility::Internal
49                )
50            });
51    }
52    true
53}

55pub(super) fn required(
56    node: Node<'_>,
57    source: &[u8],
58    kind: &str,
59    owners: &[String],
60    has_body: bool,
61    existing: Option<&crate::model::Definition>,
62) -> bool {
63    let declaration = existing.and_then(|definition| definition.declaration.as_deref());
64    if declaration.is_some_and(|declaration| declaration.modifiers.is_abstract) {
65        return true;
66    }
67    let written = super::capture::text(node, source);
68    if written
69        .split_whitespace()
70        .take(4)
71        .any(|word| word == "abstract")
72    {
73        return true;
74    }
75    if optional(node) {
76        return false;
77    }
78    match kind {
79        "property" if node.kind() == "property_signature" => true,
80        "member" if node.child_by_field_name("receiver").is_some() => false,
81        "member" => !has_body,
82        "associated" => {
83            node.child_by_field_name("type").is_none()
84                && node.child_by_field_name("value").is_none()
85        }
86        "property" if node.kind() == "const_item" => {
87            !owners.is_empty() && node.child_by_field_name("value").is_none()
88        }
89        "property" => {
90            declaration.is_some_and(|declaration| declaration.is_interface_member)
91                && !has_body
92                && !written.contains('=')
93                && !written.contains("get(")
94        }
95        _ => false,
96    }
97}
