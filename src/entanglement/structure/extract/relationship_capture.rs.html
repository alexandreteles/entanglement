4pub(super) fn capture_relationships(
5    facts: &[FileFacts],
6    fact: &FileFacts,
7    store: &SemanticStore,
8    symbols: &SymbolTargets,
9) -> Vec<Relationship> {
10    store
11        .references
12        .iter()
13        .flat_map(|reference| reference_relationships(facts, fact, store, symbols, reference))
14        .chain(
15            store
16                .imports
17                .iter()
18                .map(|import| import_relationship(fact, store, import)),
19        )
20        .collect()
21}

23fn import_relationship(
24    fact: &FileFacts,
25    store: &SemanticStore,
26    import: &crate::model::Import,
27) -> Relationship {
28    let source_module = module_id(&fact.analysis.path);
29    let name = import
30        .alias
31        .as_deref()
32        .or(import.imported_name.as_deref())
33        .or_else(|| import.path.last().map(String::as_str));
34    let used = name
35        .and_then(|name| {
36            store
37                .references
38                .iter()
39                .find(|reference| reference.path.first().is_some_and(|part| part == name))
40        })
41        .and_then(|reference| {
42            fact.analysis.resolution.iter().find(|item| {
43                item.start_byte == reference.start_byte
44                    && matches!(item.resolution, Resolution::Exact(_))
45            })
46        });
47    let target = used.and_then(|reference| match &reference.resolution {
48        Resolution::Exact(symbol) => Some(module_id(&symbol.file)),
49        _ => None,
50    });
51    Relationship {
52        ordered: false,
53        source: source_module.clone(),
54        resolution: if target.is_some() {
55            ResolutionStatus::Resolved
56        } else {
57            ResolutionStatus::Unresolved
58        },
59        target: target.unwrap_or_else(|| {
60            import
61                .source
62                .clone()
63                .unwrap_or_else(|| import.path.join("::"))
64        }),
65        kind: if import.is_public {
66            RelationshipKind::Reexport
67        } else {
68            RelationshipKind::Import
69        },
70        contract: import.is_public,
71        evidence: vec![Evidence {
72            path: fact.analysis.path.clone(),
73            start: import.start_byte,
74            end: import.end_byte,
75            owner: format!("{source_module}#{}", store.semantic.context_id),
76            detail: import.path.join("::"),
77        }],
78    }
79}
