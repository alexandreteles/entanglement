3/// Grammar decisions used by syntax capture. Shared analysis does not read grammar names.
4pub(super) fn declaration(kind: &str) -> Option<(&'static str, bool)> {
5    match kind {
6        "trait_item" | "interface_declaration" => Some(("interface", true)),
7        "struct_item"
8        | "enum_item"
9        | "union_item"
10        | "class_definition"
11        | "class_declaration"
12        | "abstract_class_declaration"
13        | "object_declaration"
14        | "trait_declaration" => Some(("type", true)),
15        "type_spec" => Some(("type", true)),
16        "type_item" | "associated_type" | "type_alias" | "type_alias_declaration" => {
17            Some(("alias", true))
18        }
19        "function_item"
20        | "function_definition"
21        | "function_declaration"
22        | "method_definition"
23        | "method_declaration"
24        | "method_signature"
25        | "function_signature_item"
26        | "method_elem"
27        | "abstract_method_signature" => Some(("member", false)),
28        "property_declaration"
29        | "property_signature"
30        | "field_definition"
31        | "public_field_definition"
32        | "field_declaration"
33        | "const_item"
34        | "const_declaration" => Some(("property", false)),
35        "assignment" | "property_element" | "property_promotion_parameter" | "const_element" => {
36            Some(("property", false))
37        }
38        _ => None,
39    }
40}

42pub(super) fn classification(node: Node<'_>, owners: &[String]) -> Option<(&'static str, bool)> {
43    if !owners.is_empty() && matches!(node.kind(), "associated_type" | "type_item") {
44        return Some(("associated", false));
45    }
46    if node.kind() == "type_spec"
47        && node
48            .child_by_field_name("type")
49            .is_some_and(|underlying| underlying.kind() == "interface_type")
50    {
51        return Some(("interface", true));
52    }
53    declaration(node.kind())
54}

56pub(super) fn name(node: Node<'_>) -> Option<Node<'_>> {
57    if node.kind() == "assignment" {
58        return node
59            .child_by_field_name("left")
60            .filter(|left| left.kind() == "identifier");
61    }
62    if node.kind() == "field_declaration" && node.child_by_field_name("name").is_none() {
63        return node.child_by_field_name("type").and_then(type_leaf);
64    }
65    node.child_by_field_name("name").or_else(|| {
66        let mut cursor = node.walk();
67        node.named_children(&mut cursor).find(|child| {
68            matches!(
69                child.kind(),
70                "simple_identifier"
71                    | "identifier"
72                    | "type_identifier"
73                    | "property_element"
74                    | "name"
75            )
76        })
77    })
78}

80pub(super) fn additional_names(node: Node<'_>, source: &[u8]) -> Vec<String> {
81    if node.kind() != "field_declaration" {
82        return Vec::new();
83    }
84    let mut cursor = node.walk();
85    node.children_by_field_name("name", &mut cursor)
86        .skip(1)
87        .map(|name| super::capture::text(name, source))
88        .collect()

91pub(super) fn relation(kind: &str) -> Option<super::super::model::RelationshipKind> {
92    use super::super::model::RelationshipKind;
93    match kind {
94        "super_interfaces" | "implements_clause" => Some(RelationshipKind::Implement),
95        "superclass"
96        | "base_class_clause"
97        | "extends_type_clause"
98        | "extends_clause"
99        | "delegation_specifiers"
100        | "trait_bounds"
101        | "class_heritage" => Some(RelationshipKind::Inherit),
102        "use_declaration" | "trait_use_clause" => Some(RelationshipKind::Compose),
103        _ => None,
104    }

128pub(super) fn limitations(language: &str) -> Vec<String> {
129    let codes: &[&str] = if language.contains("python") {
130        &["runtime_structure"]
131    } else if language.contains("go") {
132        &["structural_satisfaction", "generic_subst"]
133    } else if language.contains("rust") {
134        &[
135            "generic_subst",
136            "macro_output",
137            "ext_bounds",
138            "positional_members",
139        ]
140    } else if language.contains("php") {
141        &["dynamic_dispatch", "version_properties"]
142    } else if language.contains("kotlin") {
143        &["build_variants", "generic_subst"]
144    } else if language.contains("typescript") || language.contains("tsx") {
145        &["conditional_types", "mapped_types", "structural_compat"]
146    } else {
147        &["dynamic_bindings"]
148    };
149    codes.iter().map(|code| (*code).to_owned()).collect()
150}

158pub(super) fn public(
159    language: &str,
160    node: Node<'_>,
161    source: &[u8],
162    name: &str,
163    existing: Option<&crate::model::Definition>,
164) -> bool {
165    if language.contains("python") {
166        return !name.starts_with('_');
167    }
168    if language.contains("go") {
169        return name.chars().next().is_some_and(char::is_uppercase);
170    }
171    let prefix = node
172        .child_by_field_name("body")
173        .map_or(node.end_byte(), |body| body.start_byte());
174    let header = String::from_utf8_lossy(&source[node.start_byte()..prefix]);
175    if header
176        .split_whitespace()
177        .take(4)
178        .any(|word| word == "private")
179        || name.starts_with('#')
180    {
181        return false;
182    }
183    if language.contains("rust") {
184        let owner =
185            std::iter::successors(node.parent(), |ancestor| ancestor.parent()).find(|ancestor| {
186                matches!(
187                    ancestor.kind(),
188                    "trait_item" | "function_item" | "impl_item" | "source_file"
189                )
190            });
191        return owner.is_some_and(|owner| owner.kind() == "trait_item")
192            || existing.is_some_and(|definition| definition.is_public)
193            || header.starts_with("pub");
194    }
195    if language.contains("php") || language.contains("kotlin") {
196        return existing
197            .and_then(|definition| definition.declaration.as_ref())
198            .is_none_or(|declaration| {
199                matches!(
200                    declaration.visibility,
201                    crate::graph::SemanticVisibility::Public
202                        | crate::graph::SemanticVisibility::Protected
203                        | crate::graph::SemanticVisibility::Internal
204                )
205            });
206    }
207    true
208}

210pub(super) fn member_owner(node: Node<'_>, source: &[u8], owners: &[String]) -> Vec<String> {
211    let nearest = std::iter::successors(node.parent(), |ancestor| ancestor.parent())
212        .find(|ancestor| declaration(ancestor.kind()).is_some() || ancestor.kind() == "impl_item");
213    if nearest.is_some_and(|ancestor| {
214        declaration(ancestor.kind()).is_some_and(|(kind, _)| kind == "member")
215    }) {
216        return Vec::new();
217    }
218    node.child_by_field_name("receiver")
219        .and_then(type_leaf)
220        .map_or_else(
221            || owners.to_vec(),
222            |receiver_type| vec![super::capture::text(receiver_type, source)],
223        )
224}

226pub(super) fn receiver(node: Node<'_>, source: &[u8]) -> Option<String> {
227    if node.kind() == "field_declaration" && node.child_by_field_name("name").is_none() {
228        let mut cursor = node.walk();
229        let pointer = node
230            .children(&mut cursor)
231            .any(|child| child.kind() == "*" || child.kind() == "pointer_type");
232        return Some(
233            if pointer {
234                "embed_pointer"
235            } else {
236                "embed_value"
237            }
238            .to_owned(),
239        );
240    }
241    node.child_by_field_name("receiver").map(|receiver| {
242        if super::capture::text(receiver, source).contains('*') {
243            "pointer".to_owned()
244        } else {
245            "value".to_owned()
246        }
247    })
248}

250pub(super) fn impl_trait(node: Node<'_>, source: &[u8]) -> Option<String> {
251    std::iter::successors(node.parent(), |ancestor| ancestor.parent())
252        .find(|ancestor| ancestor.kind() == "impl_item")
253        .and_then(|implementation| implementation.child_by_field_name("trait"))
254        .map(|trait_node| super::capture::text(trait_node, source))
255}

257pub(super) fn required(
258    node: Node<'_>,
259    source: &[u8],
260    kind: &str,
261    owners: &[String],
262    has_body: bool,
263    existing: Option<&crate::model::Definition>,
264) -> bool {
265    let declaration = existing.and_then(|definition| definition.declaration.as_deref());
266    if declaration.is_some_and(|declaration| declaration.modifiers.is_abstract) {
267        return true;
268    }
269    let written = super::capture::text(node, source);
270    if written
271        .split_whitespace()
272        .take(4)
273        .any(|word| word == "abstract")
274    {
275        return true;
276    }
277    if optional(node) {
278        return false;
279    }
280    match kind {
281        "property" if node.kind() == "property_signature" => true,
282        "member" if node.child_by_field_name("receiver").is_some() => false,
283        "member" => !has_body,
284        "associated" => {
285            node.child_by_field_name("type").is_none()
286                && node.child_by_field_name("value").is_none()
287        }
288        "property" if node.kind() == "const_item" => {
289            !owners.is_empty() && node.child_by_field_name("value").is_none()
290        }
291        "property" => {
292            declaration.is_some_and(|declaration| declaration.is_interface_member)
293                && !has_body
294                && !written.contains('=')
295                && !written.contains("get(")
296        }
297        _ => false,
298    }
299}

316pub(super) fn declared_relations(node: Node<'_>) -> Vec<super::SyntaxRelation> {
317    use super::super::model::RelationshipKind;
318    let mut relations = Vec::new();
319    match node.kind() {
320        "class_definition" => {
321            if let Some(bases) = node.child_by_field_name("superclasses") {
322                relations.push(super::SyntaxRelation {
323                    start: bases.start_byte(),
324                    end: bases.end_byte(),
325                    kind: RelationshipKind::Inherit,
326                    ordered: true,
327                });
328            }
329        }
330        "type_spec" => {
331            if let Some(underlying) = node.child_by_field_name("type") {
332                embedded(underlying, &mut relations);
333            }
334        }
335        "type_alias_declaration" => {
336            if let Some(value) = node.child_by_field_name("value") {
337                alias_composition(value, &mut relations);
338            }
339        }
340        "class_declaration" | "trait_declaration" => {
341            if let Some(body) = super::capture::body_node(node) {
342                let mut cursor = body.walk();
343                relations.extend(
344                    body.named_children(&mut cursor)
345                        .filter(|child| {
346                            matches!(child.kind(), "use_declaration" | "trait_use_clause")
347                        })
348                        .map(|child| super::SyntaxRelation {
349                            start: child.start_byte(),
350                            end: child.end_byte(),
351                            kind: RelationshipKind::Compose,
352                            ordered: false,
353                        }),
354                );
355            }
356        }
357        _ => {}
358    }
359    relations
360}

362fn alias_composition(node: Node<'_>, relations: &mut Vec<super::SyntaxRelation>) {
363    use super::super::model::RelationshipKind;
364    match node.kind() {
365        "type_identifier" | "nested_type_identifier" => relations.push(super::SyntaxRelation {
366            start: node.start_byte(),
367            end: node.end_byte(),
368            kind: RelationshipKind::Compose,
369            ordered: false,
370        }),
371        "generic_type" => {
372            if let Some(name) = node.child_by_field_name("name") {
373                alias_composition(name, relations);
374            }
375        }
376        "intersection_type" | "parenthesized_type" => {
377            let mut cursor = node.walk();
378            for child in node.named_children(&mut cursor) {
379                alias_composition(child, relations);
380            }
381        }
382        _ => {}
383    }
384}

386fn embedded(node: Node<'_>, relations: &mut Vec<super::SyntaxRelation>) {
387    use super::super::model::RelationshipKind;
388    if node.kind() == "type_elem"
389        || (node.kind() == "field_declaration" && node.child_by_field_name("name").is_none())
390    {
391        relations.push(super::SyntaxRelation {
392            start: node.start_byte(),
393            end: node.end_byte(),
394            kind: RelationshipKind::Compose,
395            ordered: false,
396        });
397        relations.extend(argument_relations(node));
398        return;
399    }
400    let mut cursor = node.walk();
401    for child in node.named_children(&mut cursor) {
402        embedded(child, relations);
403    }
404}

429pub(super) fn member_kind(node: Node<'_>, source: &[u8]) -> Option<String> {
430    if !matches!(
431        node.kind(),
432        "method_definition" | "method_signature" | "abstract_method_signature"
433    ) {
434        return None;
435    }
436    let name = node.child_by_field_name("name")?;
437    if name.kind() == "computed_property_name" {
438        return None;
439    }
440    let mut cursor = node.walk();
441    let modifiers = node
442        .children(&mut cursor)
443        .filter(|child| !child.is_named())
444        .map(|child| child.kind())
445        .collect::<Vec<_>>();
446    let is_static = modifiers.contains(&"static");
447    if !is_static && super::capture::text(name, source) == "constructor" {
448        return Some("constructor".to_owned());
449    }
450    let accessor = modifiers
451        .into_iter()
452        .find(|kind| matches!(*kind, "get" | "set"));
453    Some(
454        match (is_static, accessor) {
455            (false, None) => "instance",
456            (true, None) => "static",
457            (false, Some("get")) => "get",
458            (false, _) => "set",
459            (true, Some("get")) => "static_get",
460            (true, _) => "static_set",
461        }
462        .to_owned(),
463    )
464}

466pub(super) fn constraints(
467    node: Node<'_>,
468    source: &[u8],
469) -> Vec<crate::structure::types::TypeExpression> {
470    if node.kind() != "type_spec" {
471        return Vec::new();
472    }
473    let Some(underlying) = node
474        .child_by_field_name("type")
475        .filter(|underlying| underlying.kind() == "interface_type")
476    else {
477        return Vec::new();
478    };
479    let mut cursor = underlying.walk();
480    underlying
481        .named_children(&mut cursor)
482        .filter(|child| child.kind() == "type_elem")
483        .map(|child| crate::structure::types::TypeExpression::from_node(child, source))
484        .collect()
485}

487pub(super) fn contract_fields(node: Node<'_>) -> Vec<Node<'_>> {
488    match node.kind() {
489        "struct_item" => node
490            .child_by_field_name("body")
491            .filter(|body| body.kind() == "ordered_field_declaration_list")
492            .map_or_else(Vec::new, |body| positional_fields(body, false)),
493        "enum_item" => {
494            let Some(body) = node.child_by_field_name("body") else {
495                return Vec::new();
496            };
497            let mut cursor = body.walk();
498            body.named_children(&mut cursor)
499                .filter_map(|variant| variant.child_by_field_name("body"))
500                .flat_map(enum_fields)
501                .collect()
502        }
503        _ => Vec::new(),
504    }
505}

507fn enum_fields(body: Node<'_>) -> Vec<Node<'_>> {
508    if body.kind() == "ordered_field_declaration_list" {
509        return positional_fields(body, true);
510    }
511    let mut cursor = body.walk();
512    body.named_children(&mut cursor)
513        .filter_map(|field| field.child_by_field_name("type"))
514        .collect()
515}

517fn positional_fields(body: Node<'_>, inherited_visibility: bool) -> Vec<Node<'_>> {
518    let mut cursor = body.walk();
519    let field_ids = body
520        .children_by_field_name("type", &mut cursor)
521        .map(|field| field.id())
522        .collect::<std::collections::BTreeSet<_>>();
523    let mut public = inherited_visibility;
524    let mut fields = Vec::new();
525    for child in body.named_children(&mut cursor) {
526        if child.kind() == "visibility_modifier" {
527            public = true;
528        }
529        if field_ids.contains(&child.id()) {
530            if public {
531                fields.push(child);
532            }
533            public = inherited_visibility;
534        }
535    }
536    fields
