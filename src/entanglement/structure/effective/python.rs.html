1use std::collections::{BTreeMap, BTreeSet, VecDeque};

3/// Return the Python C3 order, with the declaration first.
4///
5/// Base vectors must use source order. A missing entry is a leaf. The caller
6/// must report unresolved bases separately. Return a short error code for a
7/// base cycle, a repeated direct base, or an inconsistent precedence order.
8pub fn order(unit: &str, parents: &BTreeMap<String, Vec<String>>) -> Result<Vec<String>, String> {
9    let mut orders: BTreeMap<String, Vec<String>> = BTreeMap::new();
10    let mut active = BTreeSet::new();
11    let mut pending = vec![(unit.to_owned(), false)];
12    while let Some((current, expanded)) = pending.pop() {
13        if orders.contains_key(&current) {
14            continue;
15        }
16        let bases = parents.get(&current).cloned().unwrap_or_default();
17        if expanded {
18            let sequences = bases
19                .iter()
20                .map(|base| VecDeque::from(orders[base].clone()))
21                .chain(std::iter::once(bases.iter().cloned().collect()))
22                .collect();
23            let mut linearized = vec![current.clone()];
24            linearized.extend(merge(sequences)?);
25            active.remove(&current);
26            orders.insert(current, linearized);
27            continue;
28        }
29        if !active.insert(current.clone()) {
30            return Err("inheritance_cycle".into());
31        }
32        if bases.iter().collect::<BTreeSet<_>>().len() != bases.len() {
33            return Err("duplicate_base".into());
34        }
35        pending.push((current, true));
36        pending.extend(bases.into_iter().rev().map(|base| (base, false)));
37    }
38    Ok(orders.remove(unit).unwrap_or_default())
39}

41fn merge(mut sequences: Vec<VecDeque<String>>) -> Result<Vec<String>, String> {
42    let mut result = Vec::new();
43    while sequences.iter().any(|sequence| !sequence.is_empty()) {
44        let candidate = sequences
45            .iter()
46            .filter_map(VecDeque::front)
47            .find(|candidate| {
48                sequences
49                    .iter()
50                    .all(|sequence| !sequence.iter().skip(1).any(|entry| entry == *candidate))
51            })
52            .cloned()
53            .ok_or_else(|| "inconsistent_mro".to_owned())?;
54        for sequence in &mut sequences {
55            if sequence.front() == Some(&candidate) {
56                sequence.pop_front();
57            }
58        }
59        result.push(candidate);
60    }
61    Ok(result)
62}
