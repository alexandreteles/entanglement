3pub(super) fn parent_map(snapshot: &StructureSnapshot) -> BTreeMap<String, Vec<String>> {
4    let mut edges = snapshot
5        .relationships
6        .iter()
7        .filter(|edge| {
8            edge.resolution == ResolutionStatus::Resolved
9                && matches!(
10                    edge.kind,
11                    RelationshipKind::Inherit
12                        | RelationshipKind::Compose
13                        | RelationshipKind::Implement
14                )
15                && edge.source.starts_with("type:")
16                && edge.target.starts_with("type:")
17        })
18        .collect::<Vec<_>>();
19    edges.sort_by_key(|edge| {
20        edge.evidence
21            .iter()
22            .filter(|evidence| !matches!(evidence.detail.as_str(), "relation" | "embed_ptr"))
23            .map(|evidence| evidence.start)
24            .min()
25            .unwrap_or(0)
26    });
27    let mut parents = BTreeMap::<String, Vec<String>>::new();
28    for edge in edges {
29        let list = parents.entry(edge.source.clone()).or_default();
30        if !list.contains(&edge.target) {
31            list.push(edge.target.clone());
32        }
33    }
34    parents
35}

37pub(super) fn provenance(
38    unit: &str,
39    parents: &BTreeMap<String, Vec<String>>,
40) -> BTreeMap<String, Vec<String>> {
41    let mut paths = BTreeMap::from([(unit.to_owned(), vec![unit.to_owned()])]);
42    let mut pending = VecDeque::from([unit.to_owned()]);
43    while let Some(current) = pending.pop_front() {
44        for parent in parents.get(&current).into_iter().flatten() {
45            if paths.contains_key(parent) {
46                continue;
47            }
48            let mut path = paths[&current].clone();
49            path.push(parent.clone());
50            paths.insert(parent.clone(), path);
51            pending.push_back(parent.clone());
52        }
53    }
54    paths
55}
