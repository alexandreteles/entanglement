3pub(super) fn ambiguous_promotions(
4    members: &mut [EffectiveMember],
5    snapshot: &StructureSnapshot,
6) -> Vec<String> {
7    let mut adjacency = BTreeMap::<&str, BTreeSet<&str>>::new();
8    for edge in snapshot.relationships.iter().filter(|edge| {
9        edge.kind == RelationshipKind::Compose
10            && edge.resolution == super::super::super::model::ResolutionStatus::Resolved
11    }) {
12        adjacency
13            .entry(&edge.source)
14            .or_default()
15            .insert(&edge.target);
16    }
17    let mut obligations = Vec::new();
18    for member in members
19        .iter_mut()
20        .filter(|member| member.provenance.len() > 1)
21    {
22        let source = member.provenance.first().unwrap();
23        let target = member.provenance.last().unwrap();
24        if composition_paths(source, target, &adjacency) > 1 {
25            member.body = None;
26            obligations.push(format!("promotion_ambiguous:{}", member.name));
27        }
28    }
29    obligations
30}

32fn composition_paths(source: &str, target: &str, adjacency: &BTreeMap<&str, BTreeSet<&str>>) -> u8 {
33    let mut reached = BTreeMap::from([(source, (0, 1u8))]);
34    let mut pending = VecDeque::from([source]);
35    while let Some(current) = pending.pop_front() {
36        let (depth, count) = reached[current];
37        for next in adjacency.get(current).into_iter().flatten() {
38            match reached.get(next).copied() {
39                None => {
40                    reached.insert(next, (depth + 1, count));
41                    pending.push_back(next);
42                }
43                Some((previous_depth, previous_count)) if previous_depth == depth + 1 => {
44                    reached.insert(next, (previous_depth, (previous_count + count).min(2)));
45                }
46                _ => {}
47            }
48        }
49    }
50    reached.get(target).map_or(0, |(_, count)| *count)
51}

53pub(super) fn promote_pointer_receivers(
54    members: &mut [EffectiveMember],
55    snapshot: &StructureSnapshot,
56) {
57    for member in members
58        .iter_mut()
59        .filter(|member| member.receiver.as_deref() == Some("pointer"))
60    {
61        let embedded_pointer = member.provenance.windows(2).any(|pair| {
62            snapshot
63                .relationships
64                .iter()
65                .filter(|edge| {
66                    edge.source == pair[0]
67                        && edge.target == pair[1]
68                        && edge.kind == RelationshipKind::Compose
69                })
70                .flat_map(|edge| &edge.evidence)
71                .any(|evidence| evidence.detail == "embed_ptr")
72        });
73        if embedded_pointer {
74            member.receiver = Some("value".to_owned());
75        }
76    }
