3pub(super) fn script_choice(
4    choices: Vec<EffectiveMember>,
5    own: &[EffectiveMember],
6    name: &str,
7) -> Vec<EffectiveMember> {
8    let Some(kind) = choices
9        .first()
10        .and_then(|member| member.member_kind.as_deref())
11    else {
12        return choices;
13    };
14    if kind == "constructor" {
15        return own
16            .iter()
17            .filter(|member| {
18                member.name == name && member.member_kind.as_deref() == Some("constructor")
19            })
20            .cloned()
21            .collect();
22    }
23    let is_static = kind.starts_with("static");
24    let own_property: Vec<_> = own
25        .iter()
26        .filter(|member| {
27            member.name == name
28                && member
29                    .member_kind
30                    .as_deref()
31                    .is_some_and(|kind| kind.starts_with("static") == is_static)
32        })
33        .collect();
34    if !own_property.is_empty() {
35        return own_property
36            .into_iter()
37            .filter(|member| member.member_kind.as_deref() == Some(kind))
38            .cloned()
39            .collect();
40    }
41    let minimum = choices
42        .iter()
43        .map(|member| member.provenance.len())
44        .min()
45        .unwrap_or(0);
46    choices
47        .into_iter()
48        .filter(|member| member.provenance.len() == minimum)
49        .collect()
50}

52pub(super) fn ambiguous_members(members: &[EffectiveMember], code: &str) -> Vec<String> {
53    let mut origins = BTreeMap::<(&str, Option<&str>), BTreeSet<&str>>::new();
54    for member in members {
55        origins
56            .entry((&member.name, member.member_kind.as_deref()))
57            .or_default()
58            .insert(&member.origin);
59    }
60    origins
61        .into_iter()
62        .filter(|(_, origins)| origins.len() > 1)
63        .map(|((name, _), _)| format!("{code}:{name}"))
64        .collect()
65}

67impl Policy {
68    pub(in crate::structure::effective) fn conflicts(
69        &self,
70        members: &mut [EffectiveMember],
71    ) -> Vec<String> {
72        if matches!(self, Self::Rust | Self::TypeContracts | Self::GoInterface) {
73            return Vec::new();
74        }
75        if matches!(self, Self::ScriptClass) {
76            for member in members.iter_mut().filter(|member| {
77                member.member_kind.is_none()
78                    && member
79                        .evidence
80                        .iter()
81                        .any(|evidence| evidence.detail == "member")
82            }) {
83                member.body = None;
84            }
85        }
86        let mut bodies = BTreeMap::<(String, Option<String>), BTreeSet<String>>::new();
87        for member in members.iter() {
88            if let Some(body) = &member.body {
89                bodies
90                    .entry((member.name.clone(), member.member_kind.clone()))
91                    .or_default()
92                    .insert(body.clone());
93            }
94        }
95        let conflicts: BTreeSet<_> = bodies
96            .into_iter()
97            .filter(|(_, origins)| origins.len() > 1)
98            .map(|(name, _)| name)
99            .collect();
100        for member in members {
101            if conflicts.contains(&(member.name.clone(), member.member_kind.clone())) {
102                member.body = None;
103            }
104        }
105        if matches!(self, Self::ScriptClass) {
106            return Vec::new();
107        }
108        conflicts
109            .into_iter()
110            .map(|(name, _)| format!("member:{name}"))
111            .collect()
112    }
