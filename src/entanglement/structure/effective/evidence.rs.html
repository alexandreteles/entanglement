3impl Analysis<'_> {
4    pub(super) fn evidence(&self, unit: &str) -> Vec<Evidence> {
5        let mut evidence = declaration_evidence(unit, self.facts);
6        let mut visited = BTreeSet::from([unit.to_owned()]);
7        let mut pending = vec![unit.to_owned()];
8        while let Some(source) = pending.pop() {
9            for edge in self.snapshot.relationships.iter().filter(|edge| {
10                edge.source == source
11                    && edge.kind == RelationshipKind::Type
12                    && edge.resolution == ResolutionStatus::Resolved
13            }) {
14                if self
15                    .declarations
16                    .get(&edge.target)
17                    .is_some_and(|target| target.record.alias.is_some())
18                    && visited.insert(edge.target.clone())
19                {
20                    evidence.extend(declaration_evidence(&edge.target, self.facts));
21                    pending.push(edge.target.clone());
22                }
23            }
24        }
25        evidence.sort();
26        evidence.dedup();
27        evidence
28    }
29}

31pub(super) fn declaration_evidence(unit: &str, facts: &[FileFacts]) -> Vec<Evidence> {
32    facts
33        .iter()
34        .flat_map(|fact| {
35            fact.contexts.iter().flat_map(move |store| {
36                store
37                    .semantic
38                    .structure
39                    .iter()
40                    .filter(move |record| {
41                        record.is_type && extract::type_id(fact, store, record) == unit
42                    })
43                    .map(move |record| Evidence {
44                        path: fact.analysis.path.clone(),
45                        start: record.start,
46                        end: record.end,
47                        owner: unit.to_owned(),
48                        detail: record.kind.clone(),
49                    })
50            })
51        })
52        .collect()
53}
