3pub(super) fn eligible(
4    declaration: &EffectiveDeclaration,
5    records: &BTreeMap<String, TypeRecord<'_>>,
6    snapshot: &StructureSnapshot,
7) -> bool {
8    let Some(record) = records.get(&declaration.unit) else {
9        return false;
10    };
11    declaration.status != CoverageStatus::Unavailable
12        && declaration.conflicts.is_empty()
13        && !declaration.obligations.iter().any(|obligation| {
14            obligation.starts_with("unresolved:") || obligation.starts_with("promotion_ambiguous")
15        })
16        && !record.record.types.iter().any(|expression| {
17            expression.form.contains("type_parameter")
18                || expression.text.contains('~')
19                || expression.text.contains('|')
20        })
21        && constraints_known(record, records, snapshot)
22        && snapshot
23            .relationships
24            .iter()
25            .filter(|edge| {
26                edge.source == declaration.unit && edge.kind == RelationshipKind::Compose
27            })
28            .all(|edge| edge.resolution == ResolutionStatus::Resolved)
29}

31fn constraints_known(
32    record: &TypeRecord<'_>,
33    records: &BTreeMap<String, TypeRecord<'_>>,
34    snapshot: &StructureSnapshot,
35) -> bool {
36    let mut visited = BTreeSet::new();
37    let mut pending = vec![extract::type_id(record.fact, record.store, record.record)];
38    while let Some(unit) = pending.pop() {
39        if !visited.insert(unit.clone()) {
40            continue;
41        }
42        let Some(record) = records.get(&unit) else {
43            return false;
44        };
45        if record.store.semantic.structure_errors
46            || record.record.types.iter().any(|expression| {
47                expression.form.contains("type_parameter")
48                    || expression.text.contains('~')
49                    || expression.text.contains('|')
50            })
51        {
52            return false;
53        }
54        if !direct_constraints_known(record, records, snapshot) {
55            return false;
56        }
57        pending.extend(
58            snapshot
59                .relationships
60                .iter()
61                .filter(|edge| {
62                    edge.source == unit
63                        && edge.kind == RelationshipKind::Compose
64                        && edge.resolution == ResolutionStatus::Resolved
65                })
66                .map(|edge| edge.target.clone()),
67        );
68    }
69    true
70}

72fn direct_constraints_known(
73    record: &TypeRecord<'_>,
74    records: &BTreeMap<String, TypeRecord<'_>>,
75    snapshot: &StructureSnapshot,
76) -> bool {
77    record
78        .record
79        .types
80        .iter()
81        .filter(|expression| expression.form == "type_elem")
82        .all(|expression| {
83            snapshot
84                .relationships
85                .iter()
86                .filter(|edge| {
87                    edge.source == extract::type_id(record.fact, record.store, record.record)
88                        && edge.kind == RelationshipKind::Compose
89                        && edge.resolution == ResolutionStatus::Resolved
90                })
91                .filter_map(|edge| records.get(&edge.target))
92                .any(|target| {
93                    target.record.kind == "interface" && target.record.name == expression.text
94                })
95        })
96}
