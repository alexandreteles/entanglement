11/// A structural interface proof. It does not create a nominal dependency.
12#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
13pub struct Satisfaction {
14    pub contract: String,
15    pub receiver: String,
16    pub status: CoverageStatus,
17    pub members: Vec<String>,
18}

20struct TypeRecord<'a> {
21    file: usize,
22    fact: &'a FileFacts,
23    store: &'a SemanticStore,
24    record: &'a extract::SyntaxDeclaration,
25}

27/// Prove method-set satisfaction only with complete, resolved signatures.
28/// Generic substitutions and type-set constraints remain symbolic.
29pub(super) fn analyze(
30    facts: &[FileFacts],
31    snapshot: &StructureSnapshot,
32    declarations: &mut [EffectiveDeclaration],
33) {
34    let records = facts
35        .iter()
36        .enumerate()
37        .flat_map(|(file, fact)| {
38            fact.contexts
39                .iter()
40                .filter(|store| store.semantic.structure_language.as_deref() == Some("source.go"))
41                .flat_map(move |store| {
42                    store
43                        .semantic
44                        .structure
45                        .iter()
46                        .filter(|record| record.is_type)
47                        .map(move |record| {
48                            (
49                                extract::type_id(fact, store, record),
50                                TypeRecord {
51                                    file,
52                                    fact,
53                                    store,
54                                    record,
55                                },
56                            )
57                        })
58                })
59        })
60        .collect::<BTreeMap<_, _>>();
61    if records.is_empty() {
62        return;
63    }
64    let packages = crate::resolver::structural_go_contexts(facts);
65    let signatures = method_signatures(facts, &packages);
66    let contracts = declarations
67        .iter()
68        .filter(|declaration| {
69            records.get(&declaration.unit).is_some_and(|record| {
70                record.record.kind == "interface"
71                    && packages.contains_key(&(record.file, record.store.semantic.context_id))
72            }) && eligible(declaration, &records, snapshot)
73        })
74        .cloned()
75        .collect::<Vec<_>>();
76    for declaration in declarations
77        .iter_mut()
78        .filter(|declaration| eligible(declaration, &records, snapshot))
79    {
80        let record = &records[&declaration.unit];
81        if !packages.contains_key(&(record.file, record.store.semantic.context_id)) {
82            continue;
83        }
84        declaration.satisfies = contracts
85            .iter()
86            .filter(|contract| contract.unit != declaration.unit)
87            .filter_map(|contract| {
88                requirements(contract, &signatures).map(|requirements| (contract, requirements))
89            })
90            .flat_map(|(contract, requirements)| {
91                receiver_proofs(
92                    declaration,
93                    contract,
94                    &requirements,
95                    &signatures,
96                    record.record.kind == "interface",
97                )
98            })
99            .collect();
100    }
101}

103fn receiver_proofs(
104    declaration: &EffectiveDeclaration,
105    contract: &EffectiveDeclaration,
106    requirements: &BTreeMap<String, String>,
107    signatures: &BTreeMap<String, String>,
108    is_interface: bool,
109) -> Vec<Satisfaction> {
110    ["value", "pointer"]
111        .into_iter()
112        .filter(|receiver| !is_interface || *receiver == "value")
113        .filter(|receiver| {
114            satisfied(
115                declaration,
116                receiver,
117                requirements,
118                signatures,
119                is_interface,
120            )
121        })
122        .map(|receiver| Satisfaction {
123            contract: contract.unit.clone(),
124            receiver: receiver.to_owned(),
125            status: CoverageStatus::Complete,
126            members: requirements.keys().cloned().collect(),
127        })
128        .collect()
129}

131fn eligible(
132    declaration: &EffectiveDeclaration,
133    records: &BTreeMap<String, TypeRecord<'_>>,
134    snapshot: &StructureSnapshot,
135) -> bool {
136    let Some(record) = records.get(&declaration.unit) else {
137        return false;
138    };
139    declaration.status != CoverageStatus::Unavailable
140        && declaration.conflicts.is_empty()
141        && !declaration.obligations.iter().any(|obligation| {
142            obligation.starts_with("unresolved:") || obligation.starts_with("promotion_ambiguous")
143        })
144        && !record.record.types.iter().any(|expression| {
145            expression.form.contains("type_parameter")
146                || expression.text.contains('~')
147                || expression.text.contains('|')
148        })
149        && constraints_known(record, records, snapshot)
150        && snapshot
151            .relationships
152            .iter()
153            .filter(|edge| {
154                edge.source == declaration.unit && edge.kind == RelationshipKind::Compose
155            })
156            .all(|edge| edge.resolution == ResolutionStatus::Resolved)
157}

159fn constraints_known(
160    record: &TypeRecord<'_>,
161    records: &BTreeMap<String, TypeRecord<'_>>,
162    snapshot: &StructureSnapshot,
163) -> bool {
164    let mut visited = BTreeSet::new();
165    let mut pending = vec![extract::type_id(record.fact, record.store, record.record)];
166    while let Some(unit) = pending.pop() {
167        if !visited.insert(unit.clone()) {
168            continue;
169        }
170        let Some(record) = records.get(&unit) else {
171            return false;
172        };
173        if record.store.semantic.structure_errors
174            || record.record.types.iter().any(|expression| {
175                expression.form.contains("type_parameter")
176                    || expression.text.contains('~')
177                    || expression.text.contains('|')
178            })
179        {
180            return false;
181        }
182        if !direct_constraints_known(record, records, snapshot) {
183            return false;
184        }
185        pending.extend(
186            snapshot
187                .relationships
188                .iter()
189                .filter(|edge| {
190                    edge.source == unit
191                        && edge.kind == RelationshipKind::Compose
192                        && edge.resolution == ResolutionStatus::Resolved
193                })
194                .map(|edge| edge.target.clone()),
195        );
196    }
197    true
198}

200fn direct_constraints_known(
201    record: &TypeRecord<'_>,
202    records: &BTreeMap<String, TypeRecord<'_>>,
203    snapshot: &StructureSnapshot,
204) -> bool {
205    record
206        .record
207        .types
208        .iter()
209        .filter(|expression| expression.form == "type_elem")
210        .all(|expression| {
211            snapshot
212                .relationships
213                .iter()
214                .filter(|edge| {
215                    edge.source == extract::type_id(record.fact, record.store, record.record)
216                        && edge.kind == RelationshipKind::Compose
217                        && edge.resolution == ResolutionStatus::Resolved
218                })
219                .filter_map(|edge| records.get(&edge.target))
220                .any(|target| {
221                    target.record.kind == "interface" && target.record.name == expression.text
222                })
223        })
224}

226fn requirements(
227    contract: &EffectiveDeclaration,
228    signatures: &BTreeMap<String, String>,
229) -> Option<BTreeMap<String, String>> {
230    let mut requirements = BTreeMap::new();
231    for member in contract.members.iter().filter(|member| member.required) {
232        let signature = signatures.get(&member.origin)?;
233        if requirements
234            .insert(member.name.clone(), signature.clone())
235            .is_some_and(|previous| previous != *signature)
236        {
237            return None;
238        }
239    }
240    Some(requirements)
241}

243fn satisfied(
244    declaration: &EffectiveDeclaration,
245    receiver: &str,
246    requirements: &BTreeMap<String, String>,
247    signatures: &BTreeMap<String, String>,
248    is_interface: bool,
249) -> bool {
250    requirements.iter().all(|(name, signature)| {
251        let matches = declaration
252            .members
253            .iter()
254            .filter(|member| {
255                member.name == *name
256                    && (is_interface || !member.required)
257                    && (receiver == "pointer" || member.receiver.as_deref() != Some("pointer"))
258            })
259            .collect::<Vec<_>>();
260        let unique = matches.len() == 1 || is_interface;
261        unique
262            && !matches.is_empty()
263            && matches
264                .iter()
265                .all(|member| signatures.get(&member.origin) == Some(signature))
266    })
267}

269fn method_signatures(
270    facts: &[FileFacts],
271    packages: &BTreeMap<(usize, usize), String>,
272) -> BTreeMap<String, String> {
273    facts
274        .iter()
275        .enumerate()
276        .flat_map(|(file, fact)| {
277            fact.contexts.iter().flat_map(move |store| {
278                store.semantic.structure.iter().filter_map(move |record| {
279                    let package = packages.get(&(file, store.semantic.context_id))?;
280                    let signature = record.signature.as_ref()?;
281                    if signature.generic {
282                        return None;
283                    }
284                    let parameters = signature
285                        .parameters
286                        .iter()
287                        .map(|expression| {
288                            type_key(expression, fact, record, facts, &mut BTreeSet::new())
289                        })
290                        .collect::<Option<Vec<_>>>()?;
291                    let results = signature
292                        .results
293                        .iter()
294                        .map(|expression| {
295                            type_key(expression, fact, record, facts, &mut BTreeSet::new())
296                        })
297                        .collect::<Option<Vec<_>>>()?;
298                    Some((
299                        super::member_id(fact, store, record),
300                        format!(
301                            "{:?}",
302                            (
303                                parameters,
304                                results,
305                                signature.variadic,
306                                (!record.name.chars().next().is_some_and(char::is_uppercase))
307                                    .then_some(package)
308                            )
309                        ),
310                    ))
311                })
312            })
313        })
314        .collect()
315}

317fn type_key(
318    expression: &TypeExpression,
319    fact: &FileFacts,
320    record: &extract::SyntaxDeclaration,
321    facts: &[FileFacts],
322    seen: &mut BTreeSet<String>,
323) -> Option<String> {
324    let resolved = fact.analysis.resolution.iter().find(|reference| {
325        record.start <= reference.start_byte
326            && reference.end_byte <= record.body_start.unwrap_or(record.end)
327            && reference.path.join(".") == expression.text
328    });
329    if let Some(reference) =
330        resolved.filter(|reference| !matches!(reference.resolution, Resolution::Unresolved))
331    {
332        return match &reference.resolution {
333            Resolution::Exact(symbol) => named_type_key(symbol, facts, seen),
334            Resolution::External => {
335                builtin(&expression.text).map(|builtin| format!("builtin:{builtin}"))
336            }
337            _ => None,
338        };
339    }
340    if let Some(builtin) = builtin(&expression.text) {
341        let shadowed = facts
342            .iter()
343            .flat_map(|fact| &fact.contexts)
344            .flat_map(|store| &store.semantic.structure)
345            .any(|record| record.is_type && record.name == expression.text);
346        return (!shadowed).then(|| format!("builtin:{builtin}"));
347    }
348    if matches!(
349        expression.form.as_str(),
350        "type_identifier"
351            | "qualified_type"
352            | "struct_type"
353            | "interface_type"
354            | "generic_type"
355            | "type_instantiation_expression"
356    ) {
357        return None;
358    }
359    if expression.children.is_empty() {
360        return Some(format!("{}:{}", expression.form, expression.text));
361    }
362    let children = expression
363        .children
364        .iter()
365        .map(|child| type_key(child, fact, record, facts, &mut seen.clone()))
366        .collect::<Option<Vec<_>>>()?;
367    Some(format!("{}:{children:?}", expression.form))
368}

370fn named_type_key(
371    symbol: &crate::model::SymbolId,
372    facts: &[FileFacts],
373    seen: &mut BTreeSet<String>,
374) -> Option<String> {
375    let identity = format!("{}:{:?}:{}", symbol.file, symbol.module, symbol.name);
376    let target = facts
377        .iter()
378        .find(|fact| fact.analysis.path == symbol.file)?;
379    let alias = target
380        .contexts
381        .iter()
382        .flat_map(|store| &store.semantic.structure)
383        .find(|record| {
384            record.start <= symbol.start_byte
385                && symbol.start_byte < record.end
386                && record.name == symbol.name
387                && record.alias.is_some()
388        });
389    let Some(alias) = alias else {
390        return Some(identity);
391    };
392    if !seen.insert(identity) {
393        return None;
394    }
395    type_key(alias.alias.as_ref()?, target, alias, facts, seen)
396}

398fn builtin(name: &str) -> Option<&str> {
399    match name {
400        "byte" => Some("uint8"),
401        "rune" => Some("int32"),
402        "any" => Some("interface{}"),
403        "bool" | "string" | "int" | "int8" | "int16" | "int32" | "int64" | "uint" | "uint8"
404        | "uint16" | "uint32" | "uint64" | "uintptr" | "float32" | "float64" | "complex64"
405        | "complex128" | "error" => Some(name),
406        _ => None,
407    }
