1use super::*;

3impl Analysis<'_> {
4    pub(super) fn declaration(
5        &self,
6        unit: &str,
7        declaration: &Declaration<'_>,
8    ) -> EffectiveDeclaration {
9        if self.ambiguous.contains(unit) {
10            return EffectiveDeclaration {
11                unit: unit.to_owned(),
12                evidence: declaration_evidence(unit, self.facts),
13                status: CoverageStatus::Unavailable,
14                members: Vec::new(),
15                types: Vec::new(),
16                obligations: Vec::new(),
17                conflicts: vec!["identity_ambiguous".to_owned()],
18                satisfies: Vec::new(),
19            };
20        }
21        let policy = adapters::Policy::for_declaration(
22            declaration
23                .store
24                .semantic
25                .structure_language
26                .as_deref()
27                .unwrap_or(""),
28            &declaration.record.kind,
29        );
30        let mut obligations = self
31            .snapshot
32            .relationships
33            .iter()
34            .filter(|edge| {
35                edge.source == unit
36                    && matches!(
37                        edge.kind,
38                        RelationshipKind::Inherit
39                            | RelationshipKind::Compose
40                            | RelationshipKind::Implement
41                    )
42                    && edge.resolution != ResolutionStatus::Resolved
43            })
44            .map(|edge| format!("unresolved:{}", edge.target))
45            .collect::<Vec<_>>();
46        let paths = provenance(unit, &self.parents);
47        obligations.extend(
48            paths
49                .keys()
50                .filter(|origin| self.ambiguous.contains(*origin))
51                .map(|origin| format!("ambiguous_origin:{origin}")),
52        );
53        let mut conflicts = Vec::new();
54        let order = policy
55            .order(unit, &self.parents, &paths)
56            .unwrap_or_else(|error| {
57                conflicts.push(error);
58                vec![unit.to_owned()]
59            });
60        let selected = self
61            .selections
62            .iter()
63            .filter(|selection| {
64                symbol_unit(&selection.owner, &self.declarations).as_deref() == Some(unit)
65            })
66            .map(|selection| (selection.name.clone(), selection.origins.as_slice()))
67            .collect::<BTreeMap<_, _>>();
68        let candidates = candidates(&order, &paths, &self.direct);
69        let mut members = self.select_members(unit, candidates, &selected, &policy, &order);
70        self.add_aliases(unit, &selected, &mut members);
71        obligations.extend(self.apply_selection(unit, &mut members));
72        conflicts.extend(policy.conflicts(&mut members));
73        obligations.extend(policy.finish(&mut members, self.snapshot));
74        let mut types: Vec<TypeStructure> = declaration
75            .record
76            .types
77            .iter()
78            .map(|written| TypeStructure {
79                written: written.clone(),
80                expanded: expand_alias(
81                    written,
82                    unit,
83                    &self.declarations,
84                    self.snapshot,
85                    &mut BTreeSet::new(),
86                ),
87                origin: unit.to_owned(),
88            })
89            .chain(
90                declaration
91                    .record
92                    .alias
93                    .iter()
94                    .map(|written| TypeStructure {
95                        written: written.clone(),
96                        expanded: None,
97                        origin: unit.to_owned(),
98                    }),
99            )
100            .chain(
101                members
102                    .iter()
103                    .flat_map(|member| member_types(&member.origin, self.facts)),
104            )
105            .collect();
106        for structure in &mut types {
107            if structure.expanded.is_some() {
108                continue;
109            }
110            let source = self
111                .direct
112                .iter()
113                .find(|(_, members)| {
114                    members
115                        .iter()
116                        .any(|member| member.origin == structure.origin)
117                })
118                .map_or(unit, |(source, _)| source);
119            structure.expanded = expand_alias(
120                &structure.written,
121                source,
122                &self.declarations,
123                self.snapshot,
124                &mut BTreeSet::new(),
125            );
126        }
127        members
128            .sort_by(|left, right| (&left.name, &left.origin).cmp(&(&right.name, &right.origin)));
129        obligations.sort();
130        obligations.dedup();
131        conflicts.sort();
132        conflicts.dedup();
133        EffectiveDeclaration {
134            unit: unit.to_owned(),
135            evidence: self.evidence(unit),
136            status: if declaration.store.semantic.structure_errors {
137                CoverageStatus::Unavailable
138            } else {
139                CoverageStatus::Partial
140            },
141            members,
142            types,
143            obligations,
144            conflicts,
145            satisfies: Vec::new(),
146        }
147    }
148}
