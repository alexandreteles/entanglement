3fn ordered_targets(graph: &GraphAnalysis) -> BTreeMap<String, Vec<String>> {
4    let mut supports = BTreeMap::<String, BTreeSet<(usize, usize, String)>>::new();
5    for edge in graph.edges.iter().filter(|edge| edge.ordered) {
6        supports.entry(edge.source.clone()).or_default().extend(
7            edge.evidence
8                .iter()
9                .filter(|evidence| !matches!(evidence.detail.as_str(), "relation" | "embed_ptr"))
10                .map(|evidence| (evidence.start, evidence.end, edge.target.clone())),
11        );
12    }
13    supports
14        .into_iter()
15        .map(|(source, targets)| {
16            (
17                source,
18                targets.into_iter().map(|(_, _, target)| target).collect(),
19            )
20        })
21        .collect()
22}

24pub(super) fn compare_order(
25    previous: &GraphAnalysis,
26    current: &GraphAnalysis,
27    matched: &BTreeSet<String>,
28    events: &mut Vec<StructuralEvent>,
29) {
30    let previous_order = ordered_targets(previous);
31    let current_order = ordered_targets(current);
32    for (source, before) in previous_order
33        .iter()
34        .filter(|(source, _)| matched.contains(*source))
35    {
36        let Some(after) = current_order.get(source).filter(|after| *after != before) else {
37            continue;
38        };
39        let mut previous_targets = before.clone();
40        previous_targets.sort();
41        let mut current_targets = after.clone();
42        current_targets.sort();
43        if previous_targets != current_targets {
44            continue;
45        }
46        let changed: Vec<_> = previous
47            .edges
48            .iter()
49            .chain(&current.edges)
50            .filter(|edge| edge.ordered && edge.source == *source)
51            .cloned()
52            .collect();
53        let mut finding = event(
54            "relationship_order_changed",
55            current,
56            vec![source.clone()],
57            &changed,
58        );
59        finding.before = before.clone();
60        finding.after = after.clone();
61        finding.before_paths = before
62            .iter()
63            .filter_map(|target| graph::witness(previous, source, target))
64            .collect();
65        finding.after_paths = after
66            .iter()
67            .filter_map(|target| graph::witness(current, source, target))
68            .collect();
69        finding.before_path = finding.before_paths.first().cloned();
70        finding.after_path = finding.after_paths.first().cloned();
71        events.push(finding);
72    }
