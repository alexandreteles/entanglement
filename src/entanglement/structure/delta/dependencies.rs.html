3pub(super) fn compare_dependencies(
4    current: &GraphAnalysis,
5    coverage_lost: bool,
6    changed: &[Relationship],
7    events: &mut Vec<StructuralEvent>,
8) {
9    for edge in changed {
10        let added = current
11            .edges
12            .iter()
13            .any(|candidate| edge_key(candidate) == edge_key(edge));
14        if !added && coverage_lost {
15            continue;
16        }
17        let mut finding = event(
18            if added {
19                "dependency_added"
20            } else {
21                "dependency_removed"
22            },
23            current,
24            vec![edge.source.clone(), edge.target.clone()],
25            std::slice::from_ref(edge),
26        );
27        if added {
28            finding.after = vec![format!("{:?}", edge.kind)];
29        } else {
30            finding.before = vec![format!("{:?}", edge.kind)];
31        }
32        events.push(finding);
33    }
34}

36pub(super) fn compare_reachability(
37    previous: &GraphAnalysis,
38    current: &GraphAnalysis,
39    matched: &BTreeSet<String>,
40    coverage_lost: bool,
41    changed: &[Relationship],
42    events: &mut Vec<StructuralEvent>,
43) {
44    let previous_paths = reachable_pairs(previous);
45    let current_paths = reachable_pairs(current);
46    for source in matched {
47        for target in matched.iter().filter(|target| *target != source) {
48            let key = (source.clone(), target.clone());
49            let old_reached = previous_paths.contains(&key);
50            let new_reached = current_paths.contains(&key);
51            if old_reached == new_reached || (!new_reached && coverage_lost) {
52                continue;
53            }
54            let old = graph::witness(previous, source, target);
55            let new = graph::witness(current, source, target);
56            let mut finding = event(
57                if new.is_some() {
58                    "reach_added"
59                } else {
60                    "reach_removed"
61                },
62                current,
63                vec![source.clone(), target.clone()],
64                &path_changes(old.as_ref(), new.as_ref(), changed),
65            );
66            finding.before_path = old;
67            finding.after_path = new;
68            events.push(finding);
69        }
70    }
71}

77pub(super) fn changed_edges(before: &[Relationship], after: &[Relationship]) -> Vec<Relationship> {
78    before
79        .iter()
80        .filter(|edge| {
81            !after
82                .iter()
83                .any(|candidate| edge_key(edge) == edge_key(candidate))
84        })
85        .chain(after.iter().filter(|edge| {
86            !before
87                .iter()
88                .any(|candidate| edge_key(edge) == edge_key(candidate))
89        }))
90        .cloned()
91        .collect()
92}

94fn reachable_pairs(graph: &GraphAnalysis) -> BTreeSet<(String, String)> {
95    graph
96        .impacts
97        .iter()
98        .flat_map(|impact| {
99            impact.dependencies.iter().map(|target| {
100                (
101                    graph.units[impact.unit].clone(),
102                    graph.units[*target].clone(),
103                )
104            })
105        })
106        .collect()
107}

109fn path_changes(
110    before: Option<&Witness>,
111    after: Option<&Witness>,
112    changed: &[Relationship],
113) -> Vec<Relationship> {
114    changed
115        .iter()
116        .filter(|edge| {
117            before.into_iter().chain(after).any(|path| {
118                path.units
119                    .windows(2)
120                    .any(|pair| pair[0] == edge.source && pair[1] == edge.target)
121            })
122        })
123        .cloned()
124        .collect()
125}
