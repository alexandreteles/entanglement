61/// Build module and type facts from the canonical stores and resolved references.
62/// Embedded contexts retain their existing context IDs. Containment is not an edge.
63pub(crate) fn snapshot(
64    facts: &[FileFacts],
65    scope: &str,
66    source_snapshot: &crate::snapshot::Snapshot,
67) -> StructureSnapshot {
68    let symbols = symbol_targets(facts);
69    let relationships = facts
70        .iter()
71        .flat_map(|fact| {
72            fact.contexts
73                .iter()
74                .flat_map(|store| capture_relationships(facts, fact, store, &symbols))
75        })
76        .collect();
77    let mut result = StructureSnapshot {
78        scope: scope.to_owned(),
79        units: units(facts),
80        coverage: coverage(facts),
81        relationships,
82        ..Default::default()
83    };
84    canonical_modules(facts, &mut result.relationships);
85    canonical_types(facts, source_snapshot, &symbols, &mut result.relationships);
86    result
87        .relationships
88        .extend(super::resources::relationships(facts));
89    result.relationships = deduplicate(result.relationships);
90    result
91}

93pub(crate) fn containing_type<'a>(
94    _fact: &FileFacts,
95    store: &'a SemanticStore,
96    position: usize,
97) -> Option<&'a SyntaxDeclaration> {
98    let enclosing = store
99        .semantic
100        .structure
101        .iter()
102        .filter(|record| record.start <= position && position < record.end)
103        .min_by_key(|record| record.end - record.start)?;
104    if enclosing.is_type {
105        return Some(enclosing);
106    }
107    store
108        .semantic
109        .structure
110        .iter()
111        .filter(|record| {
112            record.is_type
113                && (record.start <= position && position < record.end
114                    || enclosing
115                        .owner
116                        .last()
117                        .is_some_and(|owner| owner == &record.name))
118        })
119        .min_by_key(|record| record.end - record.start)
120}

122pub(crate) fn type_id(
123    fact: &FileFacts,
124    store: &SemanticStore,
125    record: &SyntaxDeclaration,
126) -> String {
127    format!(
128        "type:{}#{}:{}",
129        fact.analysis.path,
130        store
131            .semantic
132            .structure_language
133            .as_deref()
134            .unwrap_or("unknown"),
135        qualified(record)
136    )
137}

139pub(crate) fn qualified(record: &SyntaxDeclaration) -> String {
140    record
141        .owner
142        .iter()
143        .chain(std::iter::once(&record.name))
144        .cloned()
145        .collect::<Vec<_>>()
146        .join("::")
147}

164fn deduplicate(relationships: Vec<Relationship>) -> Vec<Relationship> {
165    let mut edges = BTreeMap::<_, Relationship>::new();
166    for relationship in relationships {
167        let key = (
168            relationship.source.clone(),
169            relationship.target.clone(),
170            relationship.kind,
171            relationship.contract,
172            relationship.ordered,
173            relationship.resolution,
174        );
175        match edges.entry(key) {
176            std::collections::btree_map::Entry::Vacant(entry) => {
177                entry.insert(relationship);
178            }
179            std::collections::btree_map::Entry::Occupied(mut entry) => {
180                entry.get_mut().evidence.extend(relationship.evidence)
181            }
182        }
183    }
184    edges
185        .into_values()
186        .map(|mut edge| {
187            edge.evidence.sort();
188            edge.evidence.dedup();
189            edge
190        })
191        .collect()
192}
