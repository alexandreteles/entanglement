21/// Compare stable identities with complete paths through both snapshots.
22///
23/// A lost unit or reduced coverage prevents claims of structural improvement.
24/// Renames without a stable identity remain unmatched.
25pub fn compare(before: &StructureSnapshot, after: &StructureSnapshot) -> StructuralDelta {
26    let mut events = Vec::new();
27    for granularity in [UnitKind::Module, UnitKind::Type] {
28        for projection in [Projection::Contract, Projection::Full] {
29            let previous = graph::analyze_projection(before, granularity, projection);
30            let current = graph::analyze_projection(after, granularity, projection);
31            compare_projection(before, after, &previous, &current, &mut events);
32        }
33    }
34    events.sort_by(|left, right| {
35        (left.granularity, left.projection, &left.kind, &left.units).cmp(&(
36            right.granularity,
37            right.projection,
38            &right.kind,
39            &right.units,
40        ))
41    });
42    StructuralDelta { events }
43}

45fn event(
46    kind: &str,
47    graph: &GraphAnalysis,
48    units: Vec<String>,
49    changed: &[Relationship],
50) -> StructuralEvent {
51    StructuralEvent {
52        kind: kind.into(),
53        granularity: graph.granularity,
54        projection: graph.projection,
55        units,
56        before: Vec::new(),
57        after: Vec::new(),
58        changed: changed.to_vec(),
59        before_path: None,
60        after_path: None,
61        before_paths: Vec::new(),
62        after_paths: Vec::new(),
63    }
64}

66fn compare_projection(
67    before: &StructureSnapshot,
68    after: &StructureSnapshot,
69    previous: &GraphAnalysis,
70    current: &GraphAnalysis,
71    events: &mut Vec<StructuralEvent>,
72) {
73    let previous_units: BTreeSet<_> = previous.units.iter().cloned().collect();
74    let current_units: BTreeSet<_> = current.units.iter().cloned().collect();
75    let mut ambiguous: BTreeSet<_> = before
76        .units
77        .iter()
78        .chain(&after.units)
79        .filter(|unit| {
80            before
81                .units
82                .iter()
83                .filter(|candidate| candidate.id == unit.id)
84                .count()
85                > 1
86                || after
87                    .units
88                    .iter()
89                    .filter(|candidate| candidate.id == unit.id)
90                    .count()
91                    > 1
92        })
93        .map(|unit| unit.id.clone())
94        .collect();
95    ambiguous.retain(|unit| previous_units.contains(unit) || current_units.contains(unit));
96    ambiguous.extend(rename_candidates(
97        before,
98        after,
99        &previous_units,
100        &current_units,
101        previous.granularity,
102    ));
103    let matched: BTreeSet<_> = previous_units
104        .intersection(&current_units)
105        .filter(|unit| !ambiguous.contains(*unit))
106        .cloned()
107        .collect();
108    let coverage_lost = coverage_lost(before, after)
109        || before.units.iter().any(|unit| {
110            !current_units.contains(&unit.id)
111                && unit.kind == previous.granularity
112                && !after.removed_paths.contains(&unit.path)
113                && !after.coverage.iter().any(|coverage| {
114                    coverage.path == unit.path
115                        && matches!(
116                            coverage.status,
117                            CoverageStatus::Complete | CoverageStatus::Partial
118                        )
119                })
120        });
121    let changed = changed_edges(&previous.edges, &current.edges);
122    if coverage_lost {
123        let mut finding = event(
124            "coverage_lost",
125            current,
126            previous_units.difference(&current_units).cloned().collect(),
127            &changed,
128        );
129        finding.before = before
130            .coverage
131            .iter()
132            .map(|coverage| format!("{}:{:?}", coverage.path, coverage.status))
133            .collect();
134        finding.after = after
135            .coverage
136            .iter()
137            .map(|coverage| format!("{}:{:?}", coverage.path, coverage.status))
138            .collect();
139        events.push(finding);
140    }
141    compare_order(previous, current, &matched, events);
142    let improvement_unavailable = coverage_lost || !ambiguous.is_empty();
143    compare_dependencies(current, improvement_unavailable, &changed, events);
144    compare_reachability(
145        previous,
146        current,
147        &matched,
148        improvement_unavailable,
149        &changed,
150        events,
151    );
152    let uncertain: BTreeSet<_> = previous
153        .cycles
154        .iter()
155        .chain(&current.cycles)
156        .filter(|cycle| cycle.members.iter().any(|unit| ambiguous.contains(unit)))
157        .flat_map(|cycle| cycle.members.iter().cloned())
158        .chain(ambiguous.iter().cloned())
159        .collect();
160    if !uncertain.is_empty() {
161        let mut finding = event(
162            "match_uncertain",
163            current,
164            uncertain.iter().cloned().collect(),
165            &[],
166        );
167        finding.before = before
168            .units
169            .iter()
170            .filter(|unit| uncertain.contains(&unit.id))
171            .map(|unit| {
172                format!(
173                    "{}:hash={}:shape={}",
174                    unit.id,
175                    unit.source_hash.as_deref().unwrap_or("?"),
176                    unit.structural_fingerprint.as_deref().unwrap_or("?")
177                )
178            })
179            .collect();
180        finding.after = after
181            .units
182            .iter()
183            .filter(|unit| uncertain.contains(&unit.id))
184            .map(|unit| {
185                format!(
186                    "{}:hash={}:shape={}",
187                    unit.id,
188                    unit.source_hash.as_deref().unwrap_or("?"),
189                    unit.structural_fingerprint.as_deref().unwrap_or("?")
190                )
191            })
192            .collect();
193        events.push(finding);
194    }
195    compare_cycles(
196        previous,
197        current,
198        coverage_lost,
199        &changed,
200        &uncertain,
201        events,
202    );
203}
