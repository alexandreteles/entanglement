9impl QueryFacts {
10    pub(super) fn record_injections(
11        &mut self,
12        analyzer: &QueryAnalyzer,
13        pattern_index: usize,
14        captures: &[QueryCapture<'_>],
15        source: &[u8],
16    ) {
17        let Some(content_id) = analyzer.injection_content else {
18            return;
19        };
20        let mut properties = analyzer
21            .injection_patterns
22            .get(&pattern_index)
23            .cloned()
24            .unwrap_or_default();
25        infer_language_capture(analyzer, captures, &mut properties);
26        let Some(language) = injection_language(&properties, captures, source) else {
27            return;
28        };
29        let host_ranges = host_ranges(analyzer, captures);
30        for content in captures.iter().filter(|item| item.index == content_id) {
31            let node = content.node;
32            self.events.push(event(SyntaxRole::InjectionLanguage, node));
33            let content_range = node.range();
34            let guest_ranges = if properties.include_children {
35                ranges::subtract(content_range, &host_ranges)
36            } else {
37                let mut cursor = node.walk();
38                let children = node
39                    .named_children(&mut cursor)
40                    .map(|child| child.range())
41                    .collect::<Vec<_>>();
42                ranges::subtract(content_range, &children)
43            };
44            self.injections.push(InjectionRequest {
45                language: language.clone(),
46                range: node.start_byte()..node.end_byte(),
47                guest_ranges,
48                priority: properties.priority,
49                inherit_scope: properties.inherit_scope,
50                inherit_metrics: properties.inherit_metrics,
51                inherit_context: properties.inherit_context,
52                share_bindings: properties.share_bindings,
53                publish_exports: properties.publish_exports,
54                registered_only: properties.registered_only,
55                serialized_bindings: Vec::new(),
56            });
57        }
58    }
59}

61fn infer_language_capture(
62    analyzer: &QueryAnalyzer,
63    captures: &[QueryCapture<'_>],
64    properties: &mut super::InjectionProperties,
65) {
66    if properties.language.is_some() || properties.language_capture.is_some() {
67        return;
68    }
69    properties.language_capture = analyzer
70        .injection_language
71        .filter(|id| captures.iter().any(|capture| capture.index == *id));
72}
73
74fn injection_language(
75    properties: &super::InjectionProperties,
76    captures: &[QueryCapture<'_>],
77    source: &[u8],
78) -> Option<String> {
79    properties.language.clone().or_else(|| {
80        properties.language_capture.and_then(|id| {
81            captures
82                .iter()
83                .find(|capture| capture.index == id)
84                .map(|capture| {
85                    String::from_utf8_lossy(&source[capture.node.byte_range()]).into_owned()
86                })
87        })
88    })
89}
90
91fn host_ranges(analyzer: &QueryAnalyzer, captures: &[QueryCapture<'_>]) -> Vec<tree_sitter::Range> {
92    analyzer.injection_host.map_or_else(Vec::new, |host_id| {
93        captures
94            .iter()
95            .filter(|item| item.index == host_id)
96            .map(|item| item.node.range())
97            .collect()
98    })
99}

101pub(super) fn merge(injections: &mut Vec<InjectionRequest>) {
102    injections.sort_by_key(|item| (item.range.start, item.range.end, item.language.clone()));
103    let mut merged: Vec<InjectionRequest> = Vec::with_capacity(injections.len());
104    for item in injections.drain(..) {
105        match merged
106            .last_mut()
107            .filter(|current| current.range == item.range && current.language == item.language)
108        {
109            Some(current) => current.merge_duplicate(&item),
110            None => merged.push(item),
111        }
112    }
113    for item in &mut merged {
114        ranges::normalize(&mut item.guest_ranges);
115    }
116    *injections = merged;
117}
