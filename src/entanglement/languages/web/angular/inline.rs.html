7/// Analyze static Angular templates without changing source byte positions.
8/// Escaped strings and template substitutions require source maps and are skipped.
9pub(in crate::languages::web) fn inline_templates(
10    root: Node<'_>,
11    source: &[u8],
12    captured: &mut CapturedTree,
13) {
14    if !captured
15        .facts
16        .imports
17        .iter()
18        .any(|import| import.source.as_deref() == Some("@angular/core"))
19    {
20        return;
21    }
22    let mut pending = vec![root];
23    while let Some(node) = pending.pop() {
24        if node.kind() == "call_expression"
25            && component_call(node, source, captured)
26            && let Some(metadata) = node
27                .child_by_field_name("arguments")
28                .and_then(|arguments| children(arguments).next())
29                .filter(|metadata| metadata.kind() == "object")
30        {
31            captured
32                .injections
33                .extend(children(metadata).filter_map(|pair| template(pair, source)));
34        }
35        pending.extend(children(node));
36    }
37}

39fn component_call(node: Node<'_>, source: &[u8], captured: &CapturedTree) -> bool {
40    let Some(function) = node.child_by_field_name("function") else {
41        return false;
42    };
43    let (binding, namespace) = match function.kind() {
44        "identifier" => (text(function, source), false),
45        "member_expression" => {
46            if function
47                .child_by_field_name("property")
48                .map(|property| text(property, source))
49                != Some("Component")
50            {
51                return false;
52            }
53            let Some(object) = function
54                .child_by_field_name("object")
55                .filter(|node| node.kind() == "identifier")
56            else {
57                return false;
58            };
59            (text(object, source), true)
60        }
61        _ => return false,
62    };
63    captured.facts.imports.iter().any(|import| {
64        import.source.as_deref() == Some("@angular/core")
65            && import.alias.as_deref() == Some(binding)
66            && import.namespace == namespace
67            && (namespace || import.imported_name.as_deref() == Some("Component"))
68    }) && !captured.facts.locals.iter().any(|local| {
69        local.name == binding
70            && local.scope_start <= node.start_byte()
71            && node.end_byte() <= local.scope_end
72    }) && !captured.facts.definitions.iter().any(|definition| {
73        definition.name == binding
74            && definition.scope_start <= node.start_byte()
75            && node.end_byte() <= definition.scope_end
76    })
77}

79fn template(pair: Node<'_>, source: &[u8]) -> Option<InjectionRequest> {
80    if pair.kind() != "pair" {
81        return None;
82    }
83    let key = pair.child_by_field_name("key")?;
84    if text(key, source).trim_matches(['\'', '"']) != "template" {
85        return None;
86    }
87    let value = pair.child_by_field_name("value")?;
88    if !matches!(value.kind(), "string" | "template_string")
89        || children(value)
90            .any(|node| matches!(node.kind(), "escape_sequence" | "template_substitution"))
91        || text(value, source).contains('\\')
92    {
93        return None;
94    }
95    let range = value.start_byte() + 1..value.end_byte().checked_sub(1)?;
96    if range.is_empty() {
97        return None;
98    }
99    let mut start = value.start_position();
100    start.column += 1;
101    let mut end = value.end_position();
102    end.column = end.column.checked_sub(1)?;
103    Some(InjectionRequest {
104        language: "angular".into(),
105        range: range.clone(),
106        guest_ranges: vec![tree_sitter::Range {
107            start_byte: range.start,
108            end_byte: range.end,
109            start_point: start,
110            end_point: end,
111        }],
112        priority: 3,
113        inherit_scope: false,
114        inherit_metrics: false,
115        inherit_context: false,
116        share_bindings: false,
117        publish_exports: false,
118        registered_only: false,
119        serialized_bindings: Vec::new(),
120    })
121}
