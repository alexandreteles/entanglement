3pub(super) fn parse_names(node: Node<'_>, source: &[u8], declaration: bool) -> Vec<String> {
4    let body = String::from_utf8_lossy(&source[node.byte_range()]);
5    let typed = node
6        .named_child(0)
7        .is_some_and(|child| child.kind() == "ts");
8    let wrapped = if declaration {
9        format!("let {body};")
10    } else {
11        format!("function __entanglement({body}) {{}}")
12    };
13    let language: Language = if typed {
14        tree_sitter_typescript::LANGUAGE_TYPESCRIPT.into()
15    } else {
16        tree_sitter_javascript::LANGUAGE.into()
17    };
18    let mut parser = Parser::new();
19    if parser.set_language(&language).is_err() {
20        return Vec::new();
21    }
22    let Some(tree) = parser.parse(&wrapped, None) else {
23        return Vec::new();
24    };
25    let kind = if declaration {
26        "variable_declarator"
27    } else {
28        "formal_parameters"
29    };
30    let Some(pattern) = find_kind(tree.root_node(), kind).and_then(|node| {
31        declaration
32            .then(|| node.child_by_field_name("name"))
33            .flatten()
34            .or(Some(node))
35    }) else {
36        return Vec::new();
37    };
38    super::super::facts::binding_names(pattern, wrapped.as_bytes())
39        .into_iter()
40        .map(|(name, _)| name)
41        .collect()
42}

44fn find_kind<'tree>(node: Node<'tree>, kind: &str) -> Option<Node<'tree>> {
45    if node.kind() == kind {
46        return Some(node);
47    }
48    let mut cursor = node.walk();
49    node.named_children(&mut cursor)
50        .find_map(|child| find_kind(child, kind))
51}
