11impl Registry {
12    /// Select a registered language from grammar metadata for a file path.
13    pub(crate) fn select_file(&self, path: &Path) -> Result<Option<LanguageChoice>> {
14        let selected = self.loader.language_configuration_for_file_name(path)?;
15        let selected = match selected {
16            Some(selected) => Some(selected),
17            None if path.is_file() => self.select_first_line(path)?,
18            None => None,
19        };
20        let Some((language, configuration)) = selected else {
21            return Ok(None);
22        };
23        self.choice(
24            configuration.scope.clone(),
25            configuration.language_name.clone(),
26            language,
27        )
28    }
29
30    /// A first line that is not UTF-8 cannot match a shebang pattern.
31    fn select_first_line(
32        &self,
33        path: &Path,
34    ) -> Result<Option<(Language, &LanguageConfiguration<'static>)>> {
35        match self
36            .loader
37            .language_configuration_for_first_line_regex(path)
38        {
39            Err(LoaderError::IO(error)) if error.error.kind() == ErrorKind::InvalidData => Ok(None),
40            selected => Ok(selected?),
41        }
42    }
43
44    /// Select a registered language from loader injection metadata.
45    pub(crate) fn select_injection(&self, name: &str) -> Result<Option<LanguageChoice>> {
46        let Some((language, configuration)) = self
47            .loader
48            .language_configuration_for_injection_string(name)?
49        else {
50            return Ok(None);
51        };
52        self.choice(
53            configuration.scope.clone(),
54            configuration.language_name.clone(),
55            language,
56        )
57    }
58
59    /// Run the registered language query and return normalized syntax facts.
60    /// Collect Halstead terminal tokens only when `include_tokens` is true.
61    pub(crate) fn capture(
62        &self,
63        id: &str,
64        tree: &Tree,
65        source: &[u8],
66        include_tokens: bool,
67    ) -> Result<CapturedTree> {
68        self.handlers
69            .get(id)
70            .ok_or_else(|| -> crate::Error {
71                std::io::Error::other(format!("No analyzer is registered for language {id}")).into()
72            })?
73            .capture(tree, source, include_tokens)
74    }
75
76    /// Return all manifest names declared by registered language descriptors.
77    pub(crate) fn project_manifests(&self) -> Vec<&'static str> {
78        let mut manifests = self
79            .specs
80            .values()
81            .flat_map(|spec| spec.project_manifests.iter().copied())
82            .collect::<Vec<_>>();
83        manifests.sort_unstable();
84        manifests.dedup();
85        manifests
86    }
87
88    fn choice(
89        &self,
90        scope: Option<String>,
91        name: String,
92        language: tree_sitter::Language,
93    ) -> Result<Option<LanguageChoice>> {
94        let Some(id) = scope.filter(|id| self.handlers.contains_key(id)) else {
95            return Ok(None);
96        };
97        let spec = self.specs[id.as_str()];
98        Ok(Some(LanguageChoice {
99            id,
100            name,
101            language,
102            resolution_family: spec.resolution_family,
103            file_module_rules: spec.file_module_rules,
104        }))
105    }
