30#[derive(Clone)]
31struct InjectionProperties {
32    language: Option<String>,
33    language_capture: Option<u32>,
34    include_children: bool,
35    priority: i32,
36    inherit_scope: bool,
37    inherit_metrics: bool,
38    inherit_context: bool,
39    share_bindings: bool,
40    publish_exports: bool,
41    registered_only: bool,
42}

44impl Default for InjectionProperties {
45    fn default() -> Self {
46        Self {
47            language: None,
48            language_capture: None,
49            include_children: false,
50            priority: 0,
51            inherit_scope: false,
52            inherit_metrics: false,
53            inherit_context: false,
54            share_bindings: false,
55            publish_exports: true,
56            registered_only: false,
57        }
58    }
59}

61impl InjectionProperties {
62    fn from_query_pattern(query: &Query, pattern_index: usize) -> Option<Self> {
63        let mut properties = Self::default();
64        let mut found = false;
65        for property in query.property_settings(pattern_index) {
66            match property.key.as_ref() {
67                "injection.language" => {
68                    found = true;
69                    if let Some(capture_id) = property.capture_id {
70                        properties.language_capture = Some(capture_id as u32);
71                    } else {
72                        properties.language = property.value.as_deref().map(str::to_owned);
73                    }
74                }
75                "injection.include-children" => {
76                    found = true;
77                    properties.include_children = property.value.as_deref() != Some("false");
78                }
79                "injection.priority" => {
80                    found = true;
81                    properties.priority = property
82                        .value
83                        .as_deref()
84                        .and_then(|value| value.parse().ok())
85                        .unwrap_or_default();
86                }
87                "injection.inherit-scope" => {
88                    properties.inherit_scope = property.value.as_deref() == Some("true");
89                }
90                "injection.inherit-metrics" => {
91                    properties.inherit_metrics = property.value.as_deref() == Some("true");
92                }
93                "injection.inherit-context" => {
94                    properties.inherit_context = property.value.as_deref() == Some("true");
95                }
96                "injection.share-bindings" => {
97                    properties.share_bindings = property.value.as_deref() == Some("true");
98                }
99                "injection.publish-exports" => {
100                    properties.publish_exports = property.value.as_deref() != Some("false");
101                }
102                "injection.registered-only" => {
103                    found = true;
104                    properties.registered_only = property.value.as_deref() == Some("true");
105                }
106                _ => {}
107            }
108        }
109        found.then_some(properties)
110    }
111}

113/// Common facts emitted by language queries before syntax-specific adapters run.
114#[derive(Default)]
115pub(crate) struct QueryFacts {
116    pub events: Vec<SyntaxEvent>,
117    pub parents: HashMap<usize, Option<usize>>,
118    pub functions: Vec<FunctionScope>,
119    pub tokens: Vec<HalsteadToken>,
120    pub injections: Vec<InjectionRequest>,
121}
122
123impl QueryAnalyzer {
124    /// Compile one query and index standard capture names and injection settings.
125    pub(crate) fn new(language: &Language, source: &str) -> Result<Self> {
126        let query = Query::new(language, source)?;
127        let roles = query
128            .capture_names()
129            .iter()
130            .enumerate()
131            .filter_map(|(index, name)| roles::capture_role(name).map(|role| (index as u32, role)))
132            .collect();
133        let capture = |name| query.capture_index_for_name(name);
134        let injection_content = capture("injection.content");
135        let injection_language = capture("injection.language");
136        let injection_host = capture("injection.host");
137        let injection_patterns = (0..query.pattern_count())
138            .filter_map(|index| {
139                InjectionProperties::from_query_pattern(&query, index)
140                    .map(|properties| (index, properties))
141            })
142            .collect();
143        Ok(Self {
144            syntax_node: capture("syntax.node"),
145            function: capture("metric.function"),
146            function_name: capture("metric.function.name"),
147            injection_content,
148            injection_language,
149            injection_host,
150            injection_patterns,
151            roles,
152            query,
153        })
154    }
155
156    /// Find a capture ID for a language adapter without exposing the compiled query.
157    pub(crate) fn capture_id(&self, name: &str) -> Option<u32> {
158        self.query.capture_index_for_name(name)
159    }
160
161    /// Capture shared syntax facts and let one language adapter consume each match.
162    pub(crate) fn capture_with<'tree>(
163        &self,
164        tree: &'tree Tree,
165        source: &'tree [u8],
166        include_tokens: bool,
167        mut on_match: impl FnMut(usize, &[QueryCapture<'tree>], &[u8]),
168    ) -> QueryFacts {
169        let mut facts = QueryFacts::default();
170        let included = ranges::included(tree);
171        let mut cursor = QueryCursor::new();
172        let mut matches = cursor.matches(&self.query, tree.root_node(), source);
173        let has_errors = tree.root_node().has_error();
174        while let Some(query_match) = matches.next() {
175            let captures = query_match.captures();
176            let retained;
177            let captures = if has_errors {
178                retained = captures
179                    .iter()
180                    .copied()
181                    .filter(|capture| {
182                        Some(capture.index) == self.syntax_node
183                            || !tokens::recovered_jsx_prose(capture.node, source)
184                    })
185                    .collect::<Vec<_>>();
186                retained.as_slice()
187            } else {
188                captures
189            };
190            facts.record_syntax(self, captures, &included, source);
191            facts.record_injections(self, query_match.pattern_index, captures, source);
192            on_match(query_match.pattern_index, captures, source);
193        }
194        facts.finish(tree, source, include_tokens, &included);
195        facts
196    }
197
198    /// Capture common syntax facts when no language-specific adapter is needed.
199    pub(crate) fn capture(&self, tree: &Tree, source: &[u8], include_tokens: bool) -> QueryFacts {
200        self.capture_with(tree, source, include_tokens, |_, _, _| {})
201    }
202}
203
204impl QueryFacts {
205    fn finish(
206        &mut self,
207        tree: &Tree,
208        source: &[u8],
209        include_tokens: bool,
210        included: &[tree_sitter::Range],
211    ) {
212        self.deduplicate();
213        injections::merge(&mut self.injections);
214        if include_tokens {
215            self.tokens = tokens::capture(tree.root_node(), source, included);
216            let comments = self
217                .events
218                .iter()
219                .filter(|event| event.role == SyntaxRole::Comment)
220                .map(|event| event.start_byte..event.end_byte)
221                .collect::<Vec<_>>();
222            self.tokens.retain(|token| {
223                !comments
224                    .iter()
225                    .any(|range| range.start <= token.start_byte && token.end_byte <= range.end)
226            });
227        }
228    }
