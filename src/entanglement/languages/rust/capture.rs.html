12impl Analyzer {
13    pub(in crate::languages) fn new(language: &Language) -> Result<Self> {
14        let source = format!(
15            "{}\n{}\n{}",
16            tree_sitter_rust::TAGS_QUERY,
17            tree_sitter_rust::INJECTIONS_QUERY,
18            ANALYSIS_QUERY
19        );
20        let query = QueryAnalyzer::new(language, &source)?;
21        let id = |name| query.capture_id(name).expect("required capture");
22        let definitions = [
23            ("definition.function", DefinitionKind::Function),
24            ("definition.method", DefinitionKind::Method),
25            ("definition.class", DefinitionKind::Other),
26            ("definition.interface", DefinitionKind::Trait),
27            ("definition.module", DefinitionKind::Module),
28            ("definition.macro", DefinitionKind::Macro),
29            ("definition.constant", DefinitionKind::Constant),
30            ("definition.static", DefinitionKind::Static),
31        ]
32        .into_iter()
33        .map(|(name, kind)| (id(name), kind))
34        .collect::<Vec<_>>();
35        let local_scopes = [
36            ("local.declaration", LocalScopeKind::Block),
37            ("local.parameter", LocalScopeKind::Function),
38            ("local.loop", LocalScopeKind::Loop),
39            ("local.condition", LocalScopeKind::Condition),
40            ("local.match", LocalScopeKind::Match),
41            ("local.closure", LocalScopeKind::Closure),
42        ]
43        .into_iter()
44        .map(|(name, scope)| (id(name), scope))
45        .collect();
46        let reference_kinds = [
47            ("reference.value", ReferenceKind::Value),
48            ("reference.type", ReferenceKind::Type),
49        ]
50        .into_iter()
51        .map(|(name, kind)| (id(name), kind))
52        .collect();
53        let captures = Captures {
54            local_scopes,
55            reference_kinds,
56            import: id("import"),
57            reference_path: id("reference.path"),
58            local_pattern: id("local.pattern"),
59            name: id("name"),
60            definitions,
61            reference_call: id("reference.call"),
62            reference_implementation: id("reference.implementation"),
63        };
64        Ok(Self { query, captures })
65    }
66}

68impl<'a> CaptureFacts<'a> {
69    pub(super) fn record_semantics(
70        &mut self,
71        query: &Captures,
72        captures: &[tree_sitter::QueryCapture<'a>],
73        source: &[u8],
74    ) {
75        for capture in captures {
76            let node = capture.node;
77            if capture.index == query.reference_path {
78                self.paths.push(RawPath {
79                    node,
80                    kind: if node.kind() == "scoped_type_identifier" {
81                        ReferenceKind::Type
82                    } else {
83                        ReferenceKind::Qualified
84                    },
85                });
86            }
87            if let Some(kind) = query.reference_kinds.get(&capture.index) {
88                self.references.push(raw_reference(node, *kind, source));
89            }
90            if capture.index == query.import {
91                self.import_roots.push(node);
92            }
93            if let Some(scope) = query.local_scopes.get(&capture.index) {
94                self.locals
95                    .push(raw_local(captures, node, query, *scope, source));
96            }
97        }
98    }
99
100    pub(super) fn record_tags(
101        &mut self,
102        query: &Captures,
103        captures: &[tree_sitter::QueryCapture<'a>],
104        source: &[u8],
105    ) {
106        let call = captures
107            .iter()
108            .any(|item| item.index == query.reference_call);
109        let implementation = captures
110            .iter()
111            .any(|item| item.index == query.reference_implementation);
112        for name in captures.iter().filter(|item| item.index == query.name) {
113            let name_node = name.node;
114            self.definitions.extend(query.definitions.iter().filter_map(
115                |(definition_id, kind)| {
116                    captures
117                        .iter()
118                        .find(|item| {
119                            item.index == *definition_id
120                                && item.node.start_byte() <= name_node.start_byte()
121                                && name_node.end_byte() <= item.node.end_byte()
122                        })
123                        .map(|definition| raw_definition(definition.node, name_node, kind))
124                },
125            ));
126            if call {
127                let kind = if name_node.kind() == "field_identifier" {
128                    ReferenceKind::Method
129                } else {
130                    ReferenceKind::Call
131                };
132                self.references.push(raw_reference(name_node, kind, source));
133            }
134            if implementation {
135                self.references
136                    .push(raw_reference(name_node, ReferenceKind::Type, source));
137            }
138        }
139    }
140}

142fn raw_definition<'tree>(
143    node: tree_sitter::Node<'tree>,
144    name: tree_sitter::Node<'tree>,
145    kind: &DefinitionKind,
146) -> RawDefinition<'tree> {
147    RawDefinition {
148        node,
149        name,
150        kind: if is_method(node) {
151            DefinitionKind::Method
152        } else if *kind == DefinitionKind::Method {
153            DefinitionKind::Function
154        } else {
155            definition_kind(kind, node.kind())
156        },
157    }
158}

160fn raw_local<'tree>(
161    captures: &[tree_sitter::QueryCapture<'tree>],
162    node: tree_sitter::Node<'tree>,
163    query: &Captures,
164    scope: LocalScopeKind,
165    source: &[u8],
166) -> RawLocal<'tree> {
167    RawLocal {
168        owner: node,
169        names: captures
170            .iter()
171            .find(|capture| capture.index == query.local_pattern)
172            .map_or_else(
173                || vec![("self".to_owned(), node.byte_range())],
174                |capture| super::scopes::binding_names(capture.node, source),
175            ),
176        scope,
177    }
178}

180fn definition_kind(kind: &DefinitionKind, node_kind: &str) -> DefinitionKind {
181    if *kind != DefinitionKind::Other {
182        return kind.clone();
183    }
184    match node_kind {
185        "struct_item" => DefinitionKind::Struct,
186        "enum_item" => DefinitionKind::Enum,
187        "union_item" => DefinitionKind::Union,
188        "type_item" => DefinitionKind::TypeAlias,
189        _ => DefinitionKind::Other,
190    }
191}

193fn is_method(node: tree_sitter::Node<'_>) -> bool {
194    node.parent().is_some_and(|declarations| {
195        declarations.kind() == "declaration_list"
196            && declarations
197                .parent()
198                .is_some_and(|owner| matches!(owner.kind(), "impl_item" | "trait_item"))
199    })
200}

202pub(super) fn has_visibility(node: tree_sitter::Node<'_>) -> bool {
203    let mut cursor = node.walk();
204    node.children(&mut cursor)
205        .any(|child| child.kind() == "visibility_modifier")
