14impl<'a> CaptureFacts<'a> {
15    fn use_ranges(&self) -> Vec<Range<usize>> {
16        self.import_roots
17            .iter()
18            .map(|node| node.byte_range())
19            .collect()
20    }
21
22    fn binding_ranges(&self) -> Vec<Range<usize>> {
23        self.locals
24            .iter()
25            .flat_map(|local| local.names.iter().map(|(_, range)| range.clone()))
26            .collect()
27    }
28
29    pub(super) fn normalize_references(
30        &mut self,
31        modules: &[ModuleSpan],
32        root: &Range<usize>,
33    ) -> Vec<Reference> {
34        let use_ranges = self.use_ranges();
35        let binding_ranges = self.binding_ranges();
36        let definition_ranges = self
37            .definitions
38            .iter()
39            .map(|definition| definition.name.byte_range())
40            .collect::<Vec<_>>();
41        let call_ranges = self
42            .references
43            .iter()
44            .filter(|reference| {
45                matches!(reference.kind, ReferenceKind::Call | ReferenceKind::Method)
46            })
47            .map(|reference| reference.node.byte_range())
48            .collect::<Vec<_>>();
49        let mut references = std::mem::take(&mut self.references);
50        references.extend(self.qualified_paths(&use_ranges));
51        let qualified_ranges = references
52            .iter()
53            .filter(|reference| {
54                matches!(
55                    reference.kind,
56                    ReferenceKind::Qualified | ReferenceKind::Type
57                )
58            })
59            .map(|reference| reference.node.byte_range())
60            .collect::<Vec<_>>();
61        references.retain(|reference| {
62            !matches!(reference.kind, ReferenceKind::Value | ReferenceKind::Type)
63                || !definition_ranges
64                    .iter()
65                    .chain(&use_ranges)
66                    .chain(&binding_ranges)
67                    .chain(&call_ranges)
68                    .any(|range| contains(range, &reference.node.byte_range()))
69                    && !qualified_ranges.iter().any(|range| {
70                        range != &reference.node.byte_range()
71                            && contains(range, &reference.node.byte_range())
72                    })
73        });
74        references
75            .into_iter()
76            .map(|reference| {
77                let module = relative_module(reference.node.start_byte(), modules);
78                let (scope_start, scope_end) = lexical_scope(reference.node, false, root);
79                let call_owner =
80                    call_owner(reference.node, &reference.node.byte_range(), reference.kind);
81                Reference {
82                    path: reference.path,
83                    module,
84                    kind: reference.kind,
85                    start_byte: reference.node.start_byte(),
86                    end_byte: reference.node.end_byte(),
87                    scope_start,
88                    scope_end,
89                    call_owner,
90                    execution_owner: call_owner,
91                    access: None,
92                    context_id: 0,
93                }
94            })
95            .collect()
96    }
97
98    fn qualified_paths(&self, use_ranges: &[Range<usize>]) -> Vec<RawReference<'a>> {
99        let candidates = self
100            .paths
101            .iter()
102            .filter_map(|path| {
103                let segments = path_segments(path.node, self.source);
104                (!segments.is_empty()).then_some(RawReference {
105                    node: path.node,
106                    path: segments,
107                    kind: path.kind,
108                })
109            })
110            .filter(|reference| {
111                !use_ranges
112                    .iter()
113                    .any(|range| contains(range, &reference.node.byte_range()))
114            })
115            .collect::<Vec<_>>();
116        let ranges = candidates
117            .iter()
118            .map(|reference| reference.node.byte_range())
119            .collect::<Vec<_>>();
120        candidates
121            .into_iter()
122            .filter(|inner| {
123                !ranges.iter().any(|outer| {
124                    outer != &inner.node.byte_range() && contains(outer, &inner.node.byte_range())
125                })
126            })
127            .collect()
128    }
129}
