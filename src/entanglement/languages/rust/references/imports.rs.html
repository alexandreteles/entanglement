10impl CaptureFacts<'_> {
11    pub(in crate::languages::rust) fn normalize_imports(
12        &self,
13        modules: &[ModuleSpan],
14        root: &Range<usize>,
15    ) -> Vec<Import> {
16        let mut raw = Vec::new();
17        for node in &self.import_roots {
18            if let Some(argument) = node.child_by_field_name("argument") {
19                flatten_use_node(argument, &[], has_visibility(*node), self.source, &mut raw);
20            }
21        }
22        raw.into_iter()
23            .map(|import| {
24                let (scope_start, scope_end) = lexical_scope(import.node, false, root);
25                Import {
26                    path: import.path,
27                    alias: import.alias,
28                    source: None,
29                    imported_name: None,
30                    namespace: false,
31                    start_byte: import.node.start_byte(),
32                    end_byte: import.node.end_byte(),
33                    module: relative_module(import.node.start_byte(), modules),
34                    scope_start,
35                    scope_end,
36                    is_public: import.is_public,
37                    context_id: 0,
38                }
39            })
40            .collect()
41    }
42}

44fn flatten_use_node<'tree>(
45    node: Node<'tree>,
46    prefix: &[String],
47    is_public: bool,
48    source: &[u8],
49    imports: &mut Vec<RawImport<'tree>>,
50) {
51    match node.kind() {
52        "scoped_use_list" => {
53            let path = field_path(node, prefix, source);
54            if let Some(list) = node.child_by_field_name("list") {
55                flatten_use_node(list, &path, is_public, source, imports);
56            }
57        }
58        "use_list" => {
59            let mut cursor = node.walk();
60            node.children(&mut cursor)
61                .filter(|child| is_use_node(*child))
62                .for_each(|child| flatten_use_node(child, prefix, is_public, source, imports));
63        }
64        "use_as_clause" => {
65            let path = field_path(node, prefix, source);
66            let alias = node
67                .child_by_field_name("alias")
68                .map(|alias| node_text(alias, source));
69            push_import(node, path, alias, is_public, imports);
70        }
71        "use_wildcard" => {
72            let mut path = prefix.to_vec();
73            let mut cursor = node.walk();
74            path.extend(
75                node.children(&mut cursor)
76                    .filter(|child| is_path_node(*child))
77                    .flat_map(|child| path_segments(child, source)),
78            );
79            path.push("*".into());
80            push_import(node, path, None, is_public, imports);
81        }
82        "self" if !prefix.is_empty() => {
83            push_import(node, prefix.to_vec(), None, is_public, imports)
84        }
85        _ => {
86            let mut path = prefix.to_vec();
87            path.extend(path_segments(node, source));
88            push_import(node, path, None, is_public, imports);
89        }
90    }
91}

93fn field_path(node: Node<'_>, prefix: &[String], source: &[u8]) -> Vec<String> {
94    let mut path = prefix.to_vec();
95    if let Some(path_node) = node.child_by_field_name("path") {
96        path.extend(path_segments(path_node, source));
97    }
98    path
99}

101fn push_import<'tree>(
102    node: Node<'tree>,
103    path: Vec<String>,
104    alias: Option<String>,
105    is_public: bool,
106    imports: &mut Vec<RawImport<'tree>>,
107) {
108    if !path.is_empty() {
109        imports.push(RawImport {
110            node,
111            path,
112            alias,
113            is_public,
114        });
115    }
