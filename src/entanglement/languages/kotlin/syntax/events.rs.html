7/// Add Kotlin-specific branches that are not explicit in the parse tree.
8/// A `when` without `else` may be exhaustive by type information, but that
9/// cannot be established from syntax alone. Count a conservative implicit
10/// fall-through arm for source-only analysis.
11pub(in crate::languages::kotlin) fn append_kotlin_decision_events(
12    nodes: &[Node<'_>],
13    events: &mut Vec<SyntaxEvent>,
14    parents: &mut HashMap<usize, Option<usize>>,
15) {
16    for node in nodes
17        .iter()
18        .copied()
19        .filter(|node| node.kind() == "when_expression")
20    {
21        let entries = (0..node.child_count())
22            .filter_map(|index| node.child(index))
23            .filter(|child| child.kind() == "when_entry")
24            .collect::<Vec<_>>();
25        let has_else = entries.iter().any(|entry| {
26            (0..entry.child_count())
27                .filter_map(|index| entry.child(index))
28                .any(|child| child.kind() == "else")
29        });
30        if has_else {
31            continue;
32        }
33        if let Some(entry) = entries.first() {
34            parents.insert(entry.id(), Some(node.id()));
35            let position = node.end_position();
36            events.push(SyntaxEvent {
37                role: SyntaxRole::Case,
38                start_byte: node.end_byte(),
39                end_byte: node.end_byte(),
40                start_row: position.row,
41                end_row: position.row,
42                end_column: position.column,
43                node_id: entry.id(),
44                parent_id: Some(node.id()),
45                terminal: true,
46            });
47        } else {
48            let start = node.start_position();
49            let end = node.end_position();
50            events.push(SyntaxEvent {
51                role: SyntaxRole::Condition,
52                start_byte: node.start_byte(),
53                end_byte: node.end_byte(),
54                start_row: start.row,
55                end_row: end.row,
56                end_column: end.column,
57                node_id: node.id(),
58                parent_id: node.parent().map(|parent| parent.id()),
59                terminal: false,
60            });
61        }
62    }
63}

65/// Remove an `else` point when the branch is an `else if`; the nested `if`
66/// event already represents that branch in cognitive complexity.
67pub(in crate::languages::kotlin) fn remove_else_events_for_else_if(
68    nodes: &[Node<'_>],
69    events: &mut Vec<SyntaxEvent>,
70) {
71    let else_if_bodies = nodes
72        .iter()
73        .filter(|node| {
74            node.kind() == "control_structure_body"
75                && node
76                    .parent()
77                    .is_some_and(|parent| parent.kind() == "if_expression")
78                && (0..node.named_child_count()).any(|index| {
79                    node.named_child(index as u32)
80                        .is_some_and(|child| child.kind() == "if_expression")
81                })
82        })
83        .map(|node| node.id())
84        .collect::<Vec<_>>();
85    events.retain(|event| {
86        event.role != SyntaxRole::CognitiveElse || !else_if_bodies.contains(&event.node_id)
87    });
88}
