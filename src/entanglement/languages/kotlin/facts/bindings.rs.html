3pub(super) fn collect_locals(
4    source: &[u8],
5    nodes: &[Node<'_>],
6    declared_names: &HashSet<(usize, usize)>,
7    result: &mut Capture,
8) {
9    for node in nodes.iter().copied() {
10        match node.kind() {
11            "parameter" | "class_parameter" | "variable_declaration" => {
12                collect_value_binding(node, source, declared_names, result);
13            }
14            _ => {}
15        }
16    }
17}

19fn collect_value_binding(
20    node: Node<'_>,
21    source: &[u8],
22    declared_names: &HashSet<(usize, usize)>,
23    result: &mut Capture,
24) {
25    let Some(name_node) = named_children(node)
26        .into_iter()
27        .find(|child| child.kind() == "simple_identifier")
28    else {
29        return;
30    };
31    if declared_names.contains(&(name_node.start_byte(), name_node.end_byte())) {
32        return;
33    }
34    let Some(scope) = binding_scope(node) else {
35        return;
36    };
37    let type_name = direct_type_children(node)
38        .next_back()
39        .map(|ty| normalized_type(text(ty, source)));
40    let (alias_target, callable_reference) = local_alias(node, source);
41    result.locals.push(LocalBinding {
42        evidence: BindingEvidence::Both,
43        context_id: 0,
44        name: text(name_node, source).trim_matches('`').to_owned(),
45        start_byte: name_node.start_byte(),
46        end_byte: name_node.end_byte(),
47        scope_start: scope.start,
48        scope_end: scope.end,
49        type_name,
50        alias_target,
51        callable_reference,
52        immutable: is_immutable_binding(node, source),
53    });
54}

56fn local_alias(node: Node<'_>, source: &[u8]) -> (Option<String>, bool) {
57    if node.kind() != "variable_declaration" {
58        return (None, false);
59    }
60    let Some(property) = node
61        .parent()
62        .filter(|parent| parent.kind() == "property_declaration")
63    else {
64        return (None, false);
65    };
66    let Some(initializer) = property_initializer(property, node) else {
67        return (None, false);
68    };
69    initializer_alias(initializer, source)
70}

72pub(super) fn initializer_alias(initializer: Node<'_>, source: &[u8]) -> (Option<String>, bool) {
73    match initializer.kind() {
74        "callable_reference" => (
75            reference_target(initializer, source).map(|(path, _)| path.join(".")),
76            true,
77        ),
78        "simple_identifier" | "navigation_expression" => (
79            simple_path(initializer, source).map(|path| path.join(".")),
80            false,
81        ),
82        _ => (None, false),
83    }
84}

86fn is_immutable_binding(node: Node<'_>, source: &[u8]) -> bool {
87    if node.kind() != "variable_declaration" {
88        return true;
89    }
90    !node
91        .parent()
92        .filter(|parent| parent.kind() == "property_declaration")
93        .and_then(|parent| {
94            named_children(parent)
95                .into_iter()
96                .find(|child| child.kind() == "binding_pattern_kind")
97        })
98        .is_some_and(|kind| text(kind, source) == "var")
99}

101pub(super) fn collect_context_facts(
102    nodes: &[Node<'_>],
103    contexts: &[crate::languages::kotlin::syntax::ContextParameters],
104    result: &mut Capture,
105) {
106    for declaration in nodes.iter().copied().filter(|node| {
107        matches!(
108            node.kind(),
109            "function_declaration"
110                | "property_declaration"
111                | "class_declaration"
112                | "object_declaration"
113        )
114    }) {
115        let Some(context) =
116            crate::languages::kotlin::syntax::context_for_declaration(declaration, contexts)
117        else {
118            continue;
119        };
120        let Some(scope) = (if matches!(
121            declaration.kind(),
122            "class_declaration" | "object_declaration"
123        ) {
124            class_body_scope(declaration)
125        } else {
126            Some(executable_body(declaration))
127        }) else {
128            continue;
129        };
130        for parameter in &context.parameters {
131            let type_name = normalized_type(parameter.type_name.clone());
132            match (&parameter.name, &parameter.name_range) {
133                (None, _) => {
134                    let markers = result
135                        .definitions
136                        .iter()
137                        .filter_map(|definition| definition.declaration.as_deref())
138                        .find(|fact| {
139                            fact.kind == DeclarationKind::Class
140                                && fact.start_byte < declaration.start_byte()
141                                && fact.declared_type.as_deref().is_some_and(|name| {
142                                    name.rsplit('.').next() == type_name.rsplit('.').next()
143                                })
144                        })
145                        .map(|fact| fact.dsl_markers.clone())
146                        .unwrap_or_default();
147                    result.semantic.receivers.push(ReceiverScope {
148                        start_byte: scope.start,
149                        end_byte: scope.end,
150                        receiver_type: type_name.clone(),
151                        implicit: true,
152                        label: Some(type_name),
153                        dsl_markers: markers,
154                    });
155                }
156                (Some(name), Some(name_range)) => {
157                    result.locals.push(LocalBinding {
158                        evidence: BindingEvidence::Both,
159                        context_id: 0,
160                        name: name.clone(),
161                        start_byte: name_range.start,
162                        end_byte: name_range.end,
163                        scope_start: scope.start,
164                        scope_end: scope.end,
165                        type_name: Some(type_name),
166                        alias_target: None,
167                        callable_reference: false,
168                        immutable: true,
169                    });
170                }
171                _ => {}
172            }
173        }
174    }
175}

177pub(super) fn binding_scope(node: Node<'_>) -> Option<Range<usize>> {
178    if matches!(node.kind(), "parameter" | "class_parameter") {
179        let owner = nearest_declaration(node)?;
180        if owner.kind() == "class_declaration" {
181            return class_body_scope(owner).or(Some(range(owner)));
182        }
183        return Some(executable_body(owner));
184    }
185    if node.kind() == "context_parameter" {
186        return nearest_declaration(node).map(executable_body);
187    }
188    let owner = nearest_declaration(node)?;
189    if !is_executable_scope(owner.kind()) {
190        return None;
191    }
192    let scope = ancestors(node)
193        .take_while(|ancestor| ancestor.start_byte() >= owner.start_byte())
194        .find(|ancestor| {
195            matches!(
196                ancestor.kind(),
197                "statements" | "lambda_literal" | "function_body"
198            )
199        })
200        .map(range)
201        .unwrap_or_else(|| executable_body(owner));
202    // A local is visible only after its full initializer/declaration.
203    let start = if node.kind() == "variable_declaration" {
204        node.parent()
205            .filter(|parent| parent.kind() == "property_declaration")
206            .map_or(node.end_byte(), |parent| parent.end_byte())
207    } else {
208        node.end_byte()
209    };
210    Some(start..scope.end)
211}

213pub(super) fn property_initializer<'tree>(
214    property: Node<'tree>,
215    declaration: Node<'tree>,
216) -> Option<Node<'tree>> {
217    named_children(property)
218        .into_iter()
219        .filter(|child| child.start_byte() >= declaration.end_byte())
220        .find(|child| {
221            !matches!(
222                child.kind(),
223                "type_constraints" | "getter" | "setter" | "receiver_type"
224            )
225        })
226}
