3pub(super) fn nearest_declaration(node: Node<'_>) -> Option<Node<'_>> {
4    ancestors(node).find(|ancestor| {
5        matches!(
6            ancestor.kind(),
7            "function_declaration"
8                | "secondary_constructor"
9                | "anonymous_function"
10                | "lambda_literal"
11                | "class_declaration"
12                | "object_declaration"
13                | "companion_object"
14                | "getter"
15                | "setter"
16                | "anonymous_initializer"
17        )
18    })

21pub(super) fn collect_calls(
22    nodes: &[Node<'_>],
23    source: &[u8],
24    bindings: &[LocalBinding],
25) -> Vec<CallFacts> {
26    nodes
27        .iter()
28        .copied()
29        .filter(|node| matches!(node.kind(), "call_expression" | "constructor_invocation"))
30        .map(|call| {
31            let children = named_children(call);
32            let suffix = children
33                .iter()
34                .copied()
35                .find(|child| child.kind() == "call_suffix");
36            let argument_containers = children
37                .iter()
38                .copied()
39                .chain(suffix.into_iter().flat_map(named_children))
40                .filter(|child| child.kind() == "value_arguments")
41                .collect::<Vec<_>>();
42            let mut arguments = argument_containers
43                .into_iter()
44                .flat_map(named_children)
45                .filter(|argument| argument.kind() == "value_argument")
46                .filter_map(|argument| call_argument(argument, source, bindings, false))
47                .collect::<Vec<_>>();
48            if let Some(suffix) = suffix {
49                arguments.extend(
50                    named_children(suffix)
51                        .into_iter()
52                        .filter(|child| child.kind() == "annotated_lambda")
53                        .filter_map(|annotated| {
54                            all_nodes(annotated)
55                                .into_iter()
56                                .find(|child| child.kind() == "lambda_literal")
57                        })
58                        .filter_map(|lambda| call_argument(lambda, source, bindings, true)),
59                );
60            }
61            let target = children
62                .iter()
63                .copied()
64                .find(|child| !matches!(child.kind(), "call_suffix" | "value_arguments"))
65                .and_then(|node| reference_target(node, source))
66                .map(|(_, target)| SourceRange {
67                    start_byte: target.start,
68                    end_byte: target.end,
69                });
70            CallFacts {
71                start_byte: call.start_byte(),
72                end_byte: call.end_byte(),
73                target,
74                receiver_type: None,
75                arguments,
76                ..CallFacts::default()
77            }
78        })
79        .collect()
80}

82pub(super) fn call_argument(
83    node: Node<'_>,
84    source: &[u8],
85    bindings: &[LocalBinding],
86    is_trailing_lambda: bool,
87) -> Option<CallArgumentFact> {
88    let (lambda, name, expression, start_byte, end_byte) = if node.kind() == "lambda_literal" {
89        (Some(node), None, node, node.start_byte(), node.end_byte())
90    } else if node.kind() == "value_argument" {
91        let children = named_children(node);
92        let argument_name = children
93            .iter()
94            .copied()
95            .find(|child| is_argument_label(*child, node, source))
96            .map(|name| text(name, source).trim_matches('`').to_owned());
97        let expression = children.iter().copied().rfind(|child| {
98            child.kind() != "annotation" && !is_argument_label(*child, node, source)
99        });
100        let lambda = expression.and_then(|expression| match expression.kind() {
101            "lambda_literal" => Some(expression),
102            "annotated_lambda" => all_nodes(expression)
103                .into_iter()
104                .find(|child| child.kind() == "lambda_literal"),
105            _ => None,
106        });
107        let expression = expression?;
108        (
109            lambda,
110            argument_name,
111            expression,
112            lambda.map_or(expression.start_byte(), |lambda| lambda.start_byte()),
113            lambda.map_or(expression.end_byte(), |lambda| lambda.end_byte()),
114        )
115    } else {
116        return None;
117    };
118    let type_name = expression_type(expression, source).or_else(|| {
119        lambda
120            .is_none()
121            .then(|| argument_binding_type(node, source, bindings))
122            .flatten()
123    });
124    let lambda_parameters = lambda.and_then(|lambda| {
125        named_children(lambda)
126            .into_iter()
127            .find(|child| child.kind() == "lambda_parameters")
128    });
129    let parameters = lambda_parameters.map(named_children).unwrap_or_default();
130    let explicit_lambda_parameters = parameters
131        .iter()
132        .copied()
133        .filter(|parameter| parameter.kind() == "variable_declaration")
134        .map(|parameter| {
135            direct_type_children(parameter)
136                .next_back()
137                .map(|ty| normalized_type(text(ty, source)))
138        })
139        .collect();
140    let lambda_parameter_names = parameters
141        .into_iter()
142        .map(|parameter| {
143            named_children(parameter)
144                .into_iter()
145                .find(|child| child.kind() == "simple_identifier")
146                .map(|name| text(name, source).trim_matches('`').to_owned())
147                .unwrap_or_default()
148        })
149        .collect();
150    Some(CallArgumentFact {
151        start_byte,
152        end_byte,
153        name,
154        type_name,
155        is_lambda: lambda.is_some(),
156        is_trailing_lambda,
157        lambda_label: lambda.and_then(|lambda| explicit_lambda_label(lambda, source)),
158        lambda_has_explicit_parameters: lambda_parameters.is_some(),
159        lambda_parameter_names,
160        lambda_parameter_types: explicit_lambda_parameters,
161        lambda_receiver_type: None,
162    })
163}

165pub(super) fn explicit_lambda_label(lambda: Node<'_>, source: &[u8]) -> Option<String> {
166    let wrapper = lambda.parent()?;
167    matches!(wrapper.kind(), "prefix_expression" | "annotated_lambda")
168        .then(|| {
169            named_children(wrapper)
170                .into_iter()
171                .find(|child| child.kind() == "label")
172        })
173        .flatten()
174        .map(|label| text(label, source))
175        .map(|label| label.trim_end_matches('@').trim().to_owned())
176        .filter(|label| !label.is_empty())
177}

179pub(super) fn expression_type(node: Node<'_>, source: &[u8]) -> Option<String> {
180    match node.kind() {
181        "integer_literal" | "hex_literal" | "bin_literal" | "long_literal" | "unsigned_literal" => {
182            integer_literal_type(&text(node, source)).map(str::to_owned)
183        }
184        "real_literal" | "float_literal" => Some(real_literal_type(&text(node, source)).to_owned()),
185        "character_literal" => Some("Char".to_owned()),
186        "string_literal" | "multi_line_string_literal" => Some("String".to_owned()),
187        "boolean_literal" => Some("Boolean".to_owned()),
188        _ => None,
189    }
190}

194        "Float"
195    } else {
196        "Double"
197    }

200pub(super) fn integer_literal_type(literal: &str) -> Option<&'static str> {
201    let lowercase = literal.replace('_', "").to_ascii_lowercase();
202    let (digits, suffix) = ["ul", "lu", "u", "l"]
203        .into_iter()
204        .find_map(|suffix| {
205            lowercase
206                .strip_suffix(suffix)
207                .map(|digits| (digits, suffix))
208        })
209        .unwrap_or((&lowercase, ""));
210    let (radix, digits) = [("0x", 16), ("0b", 2)]
211        .into_iter()
212        .find_map(|(prefix, radix)| digits.strip_prefix(prefix).map(|digits| (radix, digits)))
213        .unwrap_or((10, digits));
214    let value = u64::from_str_radix(digits, radix).ok()?;
215    match suffix {
216        "ul" | "lu" => Some("ULong"),
217        "u" if value <= u32::MAX as u64 => Some("UInt"),
218        "u" => Some("ULong"),
219        "l" if value <= i64::MAX as u64 => Some("Long"),
220        "l" => None,
221        "" if value <= i32::MAX as u64 => Some("Int"),
222        "" if value <= i64::MAX as u64 => Some("Long"),
223        _ => None,
224    }
225}

227fn argument_binding_type(
228    node: Node<'_>,
229    source: &[u8],
230    bindings: &[LocalBinding],
231) -> Option<String> {
232    // The former enrichment pass chose the first containing value_argument in
233    // source order. Keep that choice for nested calls while avoiding a full-tree scan.
234    let argument = super::super::ancestors(node)
235        .filter(|node| node.kind() == "value_argument")
236        .min_by_key(|node| (node.start_byte(), node.end_byte()))?;
237    let expression = named_children(argument).into_iter().rfind(|child| {
238        child.kind() != "annotation" && !is_argument_label(*child, argument, source)
239    })?;
240    if expression.kind() != "simple_identifier" {
241        return None;
242    }
243    let name = text(expression, source).trim_matches('`').to_owned();
244    bindings
245        .iter()
246        .filter(|binding| binding.evidence != BindingEvidence::Lexical)
247        .filter(|binding| {
248            binding.name == name
249                && binding.start_byte < expression.start_byte()
250                && binding.scope_start <= expression.start_byte()
251                && expression.start_byte() < binding.scope_end
252        })
253        .max_by_key(|binding| binding.start_byte)
254        .and_then(|binding| binding.type_name.clone())
255}
