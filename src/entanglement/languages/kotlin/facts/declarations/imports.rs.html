3pub(crate) fn package_name(nodes: &[Node<'_>], source: &[u8]) -> Vec<String> {
4    let Some(header) = nodes
5        .iter()
6        .copied()
7        .find(|node| node.kind() == "package_header")
8    else {
9        return Vec::new();
10    };
11    named_children(header)
12        .into_iter()
13        .find(|node| node.kind() == "identifier")
14        .map(|node| split_name(&text(node, source)))
15        .unwrap_or_default()
16}

18pub(crate) fn split_name(value: &str) -> Vec<String> {
19    value
20        .split('.')
21        .filter(|part| !part.is_empty())
22        .map(|part| part.trim_matches('`').to_owned())
23        .collect()
24}

26pub(crate) fn collect_imports(
27    root: Node<'_>,
28    nodes: &[Node<'_>],
29    source: &[u8],
30    package: &[String],
31) -> Vec<Import> {
32    let mut imports = Vec::new();
33    for node in nodes
34        .iter()
35        .copied()
36        .filter(|node| node.kind() == "import_header")
37    {
38        let path_node = named_children(node)
39            .into_iter()
40            .find(|child| child.kind() == "identifier");
41        let Some(path_node) = path_node else {
42            continue;
43        };
44        let path = split_name(&text(path_node, source));
45        if path.is_empty() {
46            continue;
47        }
48        let wildcard = named_children(node)
49            .into_iter()
50            .any(|child| child.kind() == "wildcard_import");
51        let alias = named_children(node)
52            .into_iter()
53            .find(|child| child.kind() == "import_alias")
54            .and_then(|alias| named_children(alias).into_iter().next())
55            .map(|alias| text(alias, source).trim_matches('`').to_owned());
56        let imported_name = (!wildcard).then(|| path.last().cloned()).flatten();
57        imports.push(Import {
58            path,
59            alias,
60            source: None,
61            imported_name,
62            namespace: wildcard,
63            start_byte: node.start_byte(),
64            end_byte: node.end_byte(),
65            module: ModulePath(package.to_vec()),
66            scope_start: root.start_byte(),
67            scope_end: root.end_byte(),
68            is_public: false,
69            context_id: 0,
70        });
71    }
72    imports
73}
