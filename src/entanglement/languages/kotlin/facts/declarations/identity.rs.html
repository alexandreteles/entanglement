3pub(crate) fn declaration_kind(node: Node<'_>) -> DeclarationKind {
4    match node.kind() {
5        "function_declaration" | "anonymous_function" | "lambda_literal" => {
6            DeclarationKind::Function
7        }
8        "secondary_constructor" => DeclarationKind::Constructor,
9        "class_declaration" => DeclarationKind::Class,
10        "object_declaration" => DeclarationKind::Object,
11        "companion_object" => DeclarationKind::CompanionObject,
12        "property_declaration" => DeclarationKind::Property,
13        "type_alias" => DeclarationKind::TypeAlias,
14        "getter" | "setter" => DeclarationKind::Accessor,
15        _ => DeclarationKind::Other,
16    }
17}

19pub(crate) fn definition_kind(node: Node<'_>, owner: &[String]) -> DefinitionKind {
20    match node.kind() {
21        "class_declaration" => DefinitionKind::Struct,
22        "type_alias" => DefinitionKind::TypeAlias,
23        "function_declaration" | "anonymous_function" | "lambda_literal" => {
24            if owner.is_empty() {
25                DefinitionKind::Function
26            } else {
27                DefinitionKind::Method
28            }
29        }
30        "secondary_constructor" | "getter" | "setter" | "anonymous_initializer" => {
31            DefinitionKind::Method
32        }
33        _ => DefinitionKind::Other,
34    }
35}

37pub(crate) fn declaration_name_node(node: Node<'_>) -> Option<Node<'_>> {
38    let wanted = match node.kind() {
39        "class_declaration" | "object_declaration" | "type_alias" | "companion_object" => {
40            return node.child_by_field_name("name").or_else(|| {
41                named_children(node).into_iter().find(|child| {
42                    matches!(
43                        child.kind(),
44                        "type_identifier" | "simple_identifier" | "identifier"
45                    )
46                })
47            });
48        }
49        "function_declaration" | "property_declaration" => "simple_identifier",
50        _ => return None,
51    };
52    if node.kind() == "property_declaration" {
53        return all_nodes(node)
54            .into_iter()
55            .filter(|child| child.kind() == "variable_declaration")
56            .flat_map(named_children)
57            .find(|child| child.kind() == wanted);
58    }
59    named_children(node)
60        .into_iter()
61        .find(|child| child.kind() == wanted)
62}

64pub(crate) fn declaration_name(node: Node<'_>, source: &[u8]) -> Option<String> {
65    declaration_name_node(node)
66        .map(|name| text(name, source).trim_matches('`').to_owned())
67        .or_else(|| (node.kind() == "companion_object").then(|| "Companion".to_owned()))
68}

92pub(crate) fn owner_path_for(
93    node: Node<'_>,
94    declaration_range: &Range<usize>,
95    excluded_recovered: Option<&Range<usize>>,
96    source: &[u8],
97    recovered_types: &[crate::languages::kotlin::syntax::RecoveredType],
98) -> Vec<String> {
99    let mut owners = node
100        .parent()
101        .into_iter()
102        .flat_map(ancestors)
103        .filter(|parent| {
104            matches!(
105                parent.kind(),
106                "class_declaration" | "object_declaration" | "companion_object"
107            )
108        })
109        .filter_map(|parent| {
110            declaration_name(parent, source).map(|name| (parent.start_byte(), name))
111        })
112        .collect::<Vec<_>>();
113    owners.extend(
114        recovered_types
115            .iter()
116            .filter(|recovered| {
117                Some(&recovered.declaration_range) != excluded_recovered
118                    && recovered.body_range.start <= declaration_range.start
119                    && declaration_range.end <= recovered.body_range.end
120            })
121            .map(|recovered| {
122                let name = recovered.name_range.as_ref().map_or_else(
123                    || "Companion".to_owned(),
124                    |name_range| {
125                        source
126                            .get(name_range.clone())
127                            .map(String::from_utf8_lossy)
128                            .map(|name| name.trim_matches('`').to_owned())
129                            .unwrap_or_default()
130                    },
131                );
132                (recovered.declaration_range.start, name)
133            }),
134    );
135    owners.sort_by_key(|(start, _)| *start);
136    owners.dedup_by(|left, right| left.0 == right.0 && left.1 == right.1);
137    owners.into_iter().map(|(_, name)| name).collect()
138}

140pub(crate) fn nearest_type_declaration(node: Node<'_>) -> Option<Node<'_>> {
141    ancestors(node)

143        .find(|parent| {

146                "class_declaration" | "object_declaration" | "companion_object"

151pub(crate) fn is_interface_member_declaration(
152    node: Node<'_>,
153    recovered_types: &[crate::languages::kotlin::syntax::RecoveredType],
154) -> bool {
155    if nearest_type_declaration(node).is_some_and(|owner| {
156        (0..owner.child_count()).any(|index| {
157            owner
158                .child(index)
159                .is_some_and(|child| child.kind() == "interface")
160        })
161    }) {
162        return true;
163    }
164    let Some(recovered) =
165        crate::languages::kotlin::syntax::recovered_type_for(node, recovered_types).filter(
166            |recovered| {
167                recovered.kind == crate::languages::kotlin::syntax::RecoveredTypeKind::Interface
168            },
169        )
170    else {
171        return false;
172    };
173    ancestors(node)
174        .take_while(|ancestor| !is_executable_scope(ancestor.kind()))
175        .any(|ancestor| {
176            ancestor.kind() == "ERROR" && ancestor.byte_range() == recovered.error_range
177        })
178}

180pub(crate) fn declaration_scope(
181    node: Node<'_>,
182    root: Node<'_>,
183    recovered_types: &[crate::languages::kotlin::syntax::RecoveredType],
184) -> Range<usize> {
185    if let Some(owner) = ancestors(node).find(|ancestor| is_executable_scope(ancestor.kind())) {
186        return executable_body(owner);
187    }
188    if let Some(owner) = ancestors(node).find(|ancestor| {
189        matches!(
190            ancestor.kind(),
191            "class_declaration" | "object_declaration" | "companion_object"
192        )
193    }) {
194        return class_body_scope(owner).unwrap_or_else(|| range(owner));
195    }
196    if let Some(owner) = recovered_types
197        .iter()
198        .filter(|owner| {
199            owner.body_range.start <= node.start_byte() && node.end_byte() <= owner.body_range.end
200        })
201        .min_by_key(|owner| owner.body_range.end - owner.body_range.start)
202    {
203        return owner.body_range.clone();
204    }
205    range(root)
206}

208pub(crate) fn modifier_metadata(
209    modifiers: Option<Node<'_>>,
210    source: &[u8],
211) -> (SemanticVisibility, DeclarationModifiers, Vec<String>) {
212    let nodes = modifiers.map(all_nodes).unwrap_or_default();
213    let words = nodes
214        .iter()
215        .copied()
216        .filter(|child| child.child_count() == 0)
217        .flat_map(|child| [text(child, source), child.kind().to_owned()])
218        .collect::<HashSet<_>>();
219    let contains = |word: &str| words.contains(word);
220    let visibility = if contains("private") {
221        SemanticVisibility::Private
222    } else if contains("protected") {
223        SemanticVisibility::Protected
224    } else if contains("internal") {
225        SemanticVisibility::Internal
226    } else {
227        SemanticVisibility::Public
228    };
229    let flags = DeclarationModifiers {
230        is_expect: contains("expect"),
231        is_actual: contains("actual"),
232        is_abstract: contains("abstract"),
233        is_open: contains("open"),
234        is_final: contains("final"),
235        is_override: contains("override"),
236        is_operator: contains("operator"),
237        is_infix: contains("infix"),
238    };
239    let annotations = nodes
240        .into_iter()
241        .filter(|child| child.kind() == "annotation")
242        .flat_map(all_nodes)
243        .filter(|child| child.kind() == "user_type")
244        .map(|child| normalized_type(text(child, source)))
245        .collect();
246    (visibility, flags, annotations)
247}
