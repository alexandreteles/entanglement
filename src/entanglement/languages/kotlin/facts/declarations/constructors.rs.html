3pub(crate) fn capture_primary_constructor_parameters(
4    node: Node<'_>,
5    context: &DefinitionContext<'_, '_>,
6    sinks: &mut DefinitionSinks<'_>,
7    owner: &[String],
8    class_name: &str,
9) {
10    if !matches!(node.kind(), "class_declaration" | "object_declaration") {
11        return;
12    }
13    let Some(parameters) = primary_constructor_parameters(node) else {
14        return;
15    };
16    let source = context.source;
17    let class_body = class_body_scope(node);
18    for parameter in parameters {
19        let Some(name_node) = named_children(parameter)
20            .into_iter()
21            .find(|child| child.kind() == "simple_identifier")
22        else {
23            continue;
24        };
25        let type_name = direct_type_children(parameter)
26            .next()
27            .map(|ty| normalized_type(text(ty, source)));
28        let is_property = named_children(parameter)
29            .into_iter()
30            .any(|child| child.kind() == "binding_pattern_kind");
31        if is_property {
32            emit_constructor_property(
33                node,
34                parameter,
35                context,
36                sinks,
37                owner
38                    .iter()
39                    .cloned()
40                    .chain(std::iter::once(class_name.to_owned()))
41                    .collect(),
42                name_node,
43                type_name,
44            );
45        } else if let Some(body) = class_body.clone() {
46            sinks.result.locals.push(LocalBinding {
47                evidence: BindingEvidence::Both,
48                context_id: 0,
49                name: text(name_node, source).trim_matches('`').to_owned(),
50                start_byte: name_node.start_byte(),
51                end_byte: name_node.end_byte(),
52                scope_start: body.start,
53                scope_end: body.end,
54                type_name,
55                alias_target: None,
56                callable_reference: false,
57                immutable: true,
58            });
59            record_declared_name(sinks, name_node);
60        }
61    }
62}

64pub(crate) fn emit_constructor_property(
65    class_node: Node<'_>,
66    parameter: Node<'_>,
67    context: &DefinitionContext<'_, '_>,
68    sinks: &mut DefinitionSinks<'_>,
69    owner: Vec<String>,
70    name_node: Node<'_>,
71    declared_type: Option<String>,
72) {
73    let source = context.source;
74    let class_scope = class_body_scope(class_node).unwrap_or_else(|| range(class_node));
75    let modifier_node = named_children(parameter)
76        .into_iter()
77        .find(|child| child.kind() == "modifiers");
78    let (visibility, modifiers, _) = modifier_metadata(modifier_node, source);
79
80    let declaration = DeclarationFacts {
81        start_byte: parameter.start_byte(),
82        owner,
83        visibility,
84        kind: DeclarationKind::Property,
85        declared_type,
86        modifiers,
87        ..DeclarationFacts::default()
88    };
89    sinks.result.definitions.push(Definition {
90        declaration: Some(Box::new(declaration)),
91        name: text(name_node, source).trim_matches('`').to_owned(),
92        kind: DefinitionKind::Other,
93        module: ModulePath(context.package.to_vec()),
94        start_byte: parameter.start_byte(),
95        end_byte: parameter.end_byte(),
96        scope_start: class_scope.start,
97        scope_end: class_scope.end,
98        is_public: visibility == SemanticVisibility::Public,
99        inline_module: false,
100        external_module: false,
101        context_id: 0,
102    });
103    record_declared_name(sinks, name_node);
104}
