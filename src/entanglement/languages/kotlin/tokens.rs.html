9/// Capture Kotlin terminal tokens while treating text segments as normalized
10/// string operands and retaining code tokens inside string interpolation.
11pub(super) fn capture(
12    root: Node<'_>,
13    source: &[u8],
14    contexts: &[ContextParameters],
15) -> Vec<HalsteadToken> {
16    let mut tokens = Vec::new();
17    let context_names = contexts
18        .iter()
19        .flat_map(|context| context.parameters.iter())
20        .filter_map(|parameter| parameter.name.as_deref())
21        .map(str::to_owned)
22        .collect::<HashSet<_>>();
23    let mut pending = vec![root];
24    while let Some(node) = pending.pop() {
25        let kind = node.kind();
26        if node.is_missing() || is_comment(kind) {
27            continue;
28        }
29        if kind == "string_literal"
30            && contexts.iter().any(|context| {
31                context.range.start <= node.start_byte() && node.end_byte() <= context.range.end
32            })
33        {
34            push(
35                &mut tokens,
36                node.byte_range(),
37                "<string>".to_owned(),
38                HalsteadTokenKind::Operand,
39                source,
40            );
41            continue;
42        }
43        if kind == "string_literal" {
44            capture_string(node, source, &mut tokens, &context_names, &mut pending);
45            continue;
46        }
47        if is_atomic_operand(kind) {
48            push(
49                &mut tokens,
50                node.byte_range(),
51                node_text(node, source),
52                HalsteadTokenKind::Operand,
53                source,
54            );
55            continue;
56        }
57        if node.child_count() == 0 {
58            let text = node_text(node, source);
59            let (text, token_kind) = normalize_leaf(node, &text, &context_names, contexts);
60            push(&mut tokens, node.byte_range(), text, token_kind, source);
61            continue;
62        }
63        pending.extend(
64            (0..node.child_count())
65                .rev()
66                .filter_map(|index| node.child(index)),
67        );
68    }
69
70    tokens.sort_by_key(|token| (token.start_byte, token.end_byte));
71    tokens
72}

74fn capture_string<'tree>(
75    node: Node<'tree>,
76    source: &[u8],
77    tokens: &mut Vec<HalsteadToken>,
78    context_names: &HashSet<String>,
79    pending: &mut Vec<Node<'tree>>,
80) {
81    let mut had_string_content = false;
82    let mut had_interpolation = false;
83    for child in (0..node.child_count())
84        .rev()
85        .filter_map(|index| node.child(index))
86    {
87        match child.kind() {
88            "string_content" => {
89                had_string_content = true;
90                push(
91                    tokens,
92                    child.byte_range(),
93                    "<string>".to_owned(),
94                    HalsteadTokenKind::Operand,
95                    source,
96                );
97            }
98            "interpolated_expression" => {
99                had_interpolation = true;
100                pending.push(child);
101            }
102            "interpolated_identifier" => {
103                had_interpolation = true;
104                let (text, kind) =
105                    normalize_leaf(child, &node_text(child, source), context_names, &[]);
106                push(tokens, child.byte_range(), text, kind, source);
107            }
108            "interpolation_expression_start"
109            | "interpolation_expression_end"
110            | "interpolation_identifier_start" => push(
111                tokens,
112                child.byte_range(),
113                node_text(child, source),
114                HalsteadTokenKind::Operator,
115                source,
116            ),
117            _ => {}
118        }
119    }
120    if !had_string_content && !had_interpolation {
121        push(
122            tokens,
123            node.byte_range(),
124            "<string>".to_owned(),
125            HalsteadTokenKind::Operand,
126            source,
127        );
128    }
129}

131fn normalize_leaf(
132    node: Node<'_>,
133    raw: &str,
134    context_names: &HashSet<String>,
135    contexts: &[ContextParameters],
136) -> (String, HalsteadTokenKind) {
137    if let Some(context) = contexts.iter().find(|context| {
138        context.range.start <= node.start_byte() && node.end_byte() <= context.range.end
139    }) {
140        let parameter_name = context
141            .parameters
142            .iter()
143            .any(|parameter| parameter.name_range.as_ref() == Some(&node.byte_range()));
144        return normalize_context_leaf(raw, parameter_name);
145    }
146    let kind = node.kind();
147    if matches!(
148        kind,
149        "interpolation_expression_start" | "interpolation_expression_end"
150    ) {
151        return (raw.to_owned(), HalsteadTokenKind::Operator);
152    }
153    if matches!(
154        kind,
155        "simple_identifier" | "type_identifier" | "identifier" | "interpolated_identifier"
156    ) {
157        return (normalize_identifier(raw), HalsteadTokenKind::Operand);
158    }
159    let identifier = normalize_identifier(raw);
160    if context_names.contains(&identifier) {
161        return (identifier, HalsteadTokenKind::Operand);
162    }
163    if kind == "string_content" {
164        return ("<string>".to_owned(), HalsteadTokenKind::Operand);
165    }
166    (
167        raw.to_owned(),
168        if node.is_named() {
169            HalsteadTokenKind::Operand
170        } else {
171            HalsteadTokenKind::Operator
172        },
173    )
174}

176/// Keep context-prefix token classes compatible with the former source scan.
177fn normalize_context_leaf(raw: &str, parameter_name: bool) -> (String, HalsteadTokenKind) {
178    const KEYWORDS: &str = "context in out suspend dynamic where this super fun val var constructor
179        class interface object enum annotation value data final open abstract sealed public private
180        protected internal expect actual external override lateinit const inline noinline crossinline
181        operator infix tailrec reified companion inner get set";
182    let identifier = raw
183        .as_bytes()
184        .first()
185        .is_some_and(|byte| byte.is_ascii_alphabetic() || *byte == b'_' || *byte >= 0x80);
186    let operand = parameter_name
187        || raw.starts_with('`')
188        || raw.as_bytes().first().is_some_and(u8::is_ascii_digit)
189        || (identifier
190            && !KEYWORDS
191                .split_ascii_whitespace()
192                .any(|keyword| keyword == raw));
193    (
194        normalize_identifier(raw),
195        if operand {
196            HalsteadTokenKind::Operand
197        } else {
198            HalsteadTokenKind::Operator
199        },
200    )
201}
202
203fn normalize_identifier(raw: &str) -> String {
204    raw.strip_prefix('`')
205        .and_then(|value| value.strip_suffix('`'))
206        .unwrap_or(raw)
207        .to_owned()
208}

210fn is_comment(kind: &str) -> bool {
211    matches!(kind, "line_comment" | "multiline_comment")
212}
213
214fn is_atomic_operand(kind: &str) -> bool {
215    matches!(
216        kind,
217        "boolean_literal"
218            | "integer_literal"
219            | "long_literal"
220            | "unsigned_literal"
221            | "hex_literal"
222            | "bin_literal"
223            | "real_literal"
224            | "character_literal"
225            | "null_literal"
226    )
227}

229fn node_text(node: Node<'_>, source: &[u8]) -> String {
230    String::from_utf8_lossy(&source[node.byte_range()]).into_owned()
231}

233fn push(
234    tokens: &mut Vec<HalsteadToken>,
235    range: std::ops::Range<usize>,
236    token: String,
237    kind: HalsteadTokenKind,
238    source: &[u8],
239) {
240    if token.is_empty() || range.start >= range.end || range.end > source.len() {
241        return;
242    }
243    let line = source[..range.start]
244        .iter()
245        .filter(|byte| **byte == b'\n')
246        .count()
247        + 1;
248    tokens.push(HalsteadToken {
249        kind,
250        token,
251        start_byte: range.start,
252        end_byte: range.end,
253        line,
254    });
