3pub(super) struct Mutation<'a> {
4    owners: Vec<usize>,
5    names: Option<Vec<&'a str>>,
6    write: Option<usize>,
7}
8
9pub(super) fn stable(
10    name: &str,
11    scope: Node<'_>,
12    mutations: &[Mutation<'_>],
13    own_write: Option<usize>,
14) -> bool {
15    mutations.iter().all(|mutation| {
16        !mutation.owners.contains(&scope.id())
17            || mutation
18                .names
19                .as_ref()
20                .is_some_and(|names| !names.contains(&name))
21            || mutation.write.is_some() && mutation.write == own_write
22    })
23}

25pub(super) fn mutations<'a>(all_nodes: &[Node<'_>], source: &'a [u8]) -> Vec<Mutation<'a>> {
26    let variables = |node| {
27        nodes(node)
28            .into_iter()
29            .filter(|node| node.kind() == "variable_name")
30            .map(|node| text(node, source))
31            .collect::<Vec<_>>()
32    };
33    all_nodes
34        .iter()
35        .copied()
36        .filter_map(|node| {
37            let (names, write) = match node.kind() {
38                "by_ref"
39                | "reference_assignment_expression"
40                | "global_declaration"
41                | "argument"
42                | "unset_statement" => (Some(variables(node)), None),
43                "dynamic_variable_name"
44                | "eval_intrinsic"
45                | "include_expression"
46                | "include_once_expression"
47                | "require_expression"
48                | "require_once_expression" => (None, None),
49                "function_call_expression"
50                    if node
51                        .child_by_field_name("function")
52                        .is_some_and(|function| {
53                            text(function, source)
54                                .trim_start_matches('\\')
55                                .eq_ignore_ascii_case("extract")
56                                || text(function, source).eq_ignore_ascii_case("eval")
57                        }) =>
58                {
59                    (None, None)
60                }
61                "assignment_expression"
62                | "augmented_assignment_expression"
63                | "update_expression"
64                | "catch_clause"
65                | "static_variable_declaration" => {
66                    let field = match node.kind() {
67                        "assignment_expression" | "augmented_assignment_expression" => "left",
68                        "update_expression" => "argument",
69                        _ => "name",
70                    };
71                    (
72                        Some(
73                            node.child_by_field_name(field)
74                                .into_iter()
75                                .flat_map(write_targets)
76                                .map(|node| text(node, source))
77                                .collect(),
78                        ),
79                        (field == "left").then(|| node.id()),
80                    )
81                }
82                "foreach_statement" => (
83                    Some(
84                        children(node)
85                            .skip(1)
86                            .filter(|child| {
87                                node.child_by_field_name("body")
88                                    .is_none_or(|body| body.id() != child.id())
89                            })
90                            .flat_map(variables)
91                            .collect(),
92                    ),
93                    None,
94                ),
95                _ => return None,
96            };
97            let owners = if matches!(node.kind(), "by_ref" | "reference_assignment_expression") {
98                ancestors(node)
99                    .filter(|node| is_callable(*node))
100                    .map(|node| node.id())
101                    .chain(all_nodes.first().map(|node| node.id()))
102                    .collect()
103            } else {
104                vec![scope(node).id()]
105            };
106            Some(Mutation {
107                owners,
108                names,
109                write,
110            })
111        })
112        .collect()
113}
114
115fn write_targets(node: Node<'_>) -> Vec<Node<'_>> {
116    match node.kind() {
117        "variable_name" => vec![node],
118        "list_literal" | "by_ref" => children(node).flat_map(write_targets).collect(),
119        "subscript_expression" => node
120            .named_child(0)
121            .into_iter()
122            .flat_map(write_targets)
123            .collect(),
124        _ => Vec::new(),
125    }
