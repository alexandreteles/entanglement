17pub(super) fn local_scopes(name: Node<'_>, kind: LocalKind) -> Vec<Range<usize>> {
18    let scope = match kind {
19        LocalKind::Parameter => parameter_scope(name),
20        LocalKind::TypeParameter => parameters::type_parameter_scope(name),
21        LocalKind::Declaration => declaration_scope(name),
22        LocalKind::SwitchAlias => return parameters::switch_alias_scopes(name),
23    };
24    scope.into_iter().collect()
25}

39/// Explicit blocks, and the statements and clauses that form implicit blocks.

42    "if_statement",
43    "for_statement",
44    "expression_switch_statement",
45    "type_switch_statement",
46    "expression_case",

55pub(super) fn enclosing_function(node: Node<'_>) -> Option<Node<'_>> {
56    ancestors(node)
57        .skip(1)
58        .find(|item| FUNCTIONS.contains(&item.kind()))
59}

61/// Parameters of function types and interface methods bind no runtime names.
62pub(super) fn parameter_scope(name: Node<'_>) -> Option<Range<usize>> {
63    ancestors(name)
64        .find(|item| {
65            FUNCTIONS.contains(&item.kind())
66                || matches!(item.kind(), "function_type" | "method_elem")
67        })
68        .filter(|owner| FUNCTIONS.contains(&owner.kind()))
69        .and_then(|owner| owner.child_by_field_name("body"))
70        .map(|body| body.byte_range())
71}

73/// A local name is visible from the end of its declaration, so `x := x + 1`
74/// and `switch x := x.(type)` read the outer `x`. Type names are visible in
75/// their own declaration to allow recursive types.
76pub(super) fn declaration_scope(name: Node<'_>) -> Option<Range<usize>> {
77    let declaration = ancestors(name).find(|item| DECLARATIONS.contains(&item.kind()))?;
78    let start = match declaration.kind() {
79        "type_spec" | "type_alias" => declaration.start_byte(),
80        _ => declaration.end_byte(),
81    };
82    let block = ancestors(declaration).find(|item| BLOCKS.contains(&item.kind()))?;
83    Some(start..block.end_byte())
84}
