10pub(super) fn add_target_bindings(
11    node: Node<'_>,
12    target: Node<'_>,
13    root: Node<'_>,
14    source: &[u8],
15    facts: &mut SemanticFacts,
16    excluded: &mut Vec<Range<usize>>,
17    visibility_start: usize,
18) {
19    let names = target_names(target, source);
20    excluded.extend(names.iter().map(|(_, span)| span.clone()));
21    add_names(node, names, root, facts, visibility_start);
22}

24pub(super) fn add_names(
25    node: Node<'_>,
26    names: Vec<(String, Range<usize>)>,
27    root: Node<'_>,
28    facts: &mut SemanticFacts,
29    visibility_start: usize,
30) {
31    let start = binding_scope_owner(node).map_or(root.start_byte(), |_| visibility_start);
32    let scopes = binding_scopes(node, root, start);
33    for (name, name_range) in names {
34        for scope in &scopes {
35            push_local(facts, name.clone(), name_range.clone(), scope.clone());
36        }
37    }
38}

40fn binding_scopes(node: Node<'_>, root: Node<'_>, start: usize) -> Vec<Range<usize>> {
41    match binding_scope_owner(node) {
42        Some(owner) if owner.kind() == "class_definition" => class_segments(owner, start),
43        Some(owner) => vec![range(body(owner))],
44        None => vec![range(root)],
45    }
46}

48pub(super) fn binding_scope_owner(node: Node<'_>) -> Option<Node<'_>> {
49    ancestors(node).find(|item| {
50        matches!(item.kind(), "function_definition" | "class_definition")
51            && range(body(*item)).start <= node.start_byte()
52            && node.end_byte() <= range(body(*item)).end
53    })
54}

56pub(crate) fn class_segments(class: Node<'_>, start: usize) -> Vec<Range<usize>> {
57    let class_body = body(class);
58    let mut holes = nodes(class_body)
59        .into_iter()
60        .filter_map(|node| match node.kind() {
61            "function_definition" | "class_definition" | "lambda" => Some(
62                field(node, "body")
63                    .map(range)
64                    .unwrap_or_else(|| range(node)),
65            ),
66            kind if super::comprehensions::is_comprehension(kind) => Some(range(node)),
67            _ => None,
68        })
69        .filter(|item| item.start < class_body.end_byte() && start < item.end)
70        .collect::<Vec<_>>();
71    holes.sort_by_key(|item| (item.start, item.end));
72    let mut cursor = start.max(class_body.start_byte());
73    let end = class_body.end_byte();
74    let mut segments = Vec::new();
75    for hole in holes {
76        if cursor < hole.start {
77            segments.push(cursor..hole.start.min(end));
78        }
79        cursor = cursor.max(hole.end);
80    }
81    if cursor < end {
82        segments.push(cursor..end);
83    }
84    segments
85}

87pub(super) fn push_local(
88    facts: &mut SemanticFacts,
89    name: String,
90    name_range: Range<usize>,
91    scope: Range<usize>,
92) {
93    if scope.start < scope.end {
94        facts.locals.push(LocalBinding {
95            name,
96            start_byte: name_range.start,
97            end_byte: name_range.end,
98            scope_start: scope.start,
99            scope_end: scope.end,
100            context_id: 0,
101            ..LocalBinding::default()
102        });
103    }
104}
