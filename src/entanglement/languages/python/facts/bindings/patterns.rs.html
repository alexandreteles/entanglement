9pub(super) fn parameter_names(parameter: Node<'_>, source: &[u8]) -> Vec<(String, Range<usize>)> {
10    match parameter.kind() {
11        "identifier" => vec![(text(parameter, source), range(parameter))],
12        "default_parameter" | "typed_default_parameter" => field(parameter, "name")
13            .map(|name| target_names(name, source))
14            .unwrap_or_default(),
15        "typed_parameter" => children(parameter)
16            .into_iter()
17            .filter(|child| {
18                matches!(
19                    child.kind(),
20                    "identifier" | "list_splat_pattern" | "dictionary_splat_pattern"
21                )
22            })
23            .flat_map(|child| target_names(child, source))
24            .collect(),
25        "list_splat_pattern" | "dictionary_splat_pattern" | "tuple_pattern" => {
26            target_names(parameter, source)
27        }
28        _ => Vec::new(),
29    }
30}

32pub(super) fn collect_exception_target(
33    clause: Node<'_>,
34    root: Node<'_>,
35    source: &[u8],
36    facts: &mut SemanticFacts,
37    excluded: &mut Vec<Range<usize>>,
38) {
39    let aliases = field(clause, "alias").into_iter().collect::<Vec<_>>();
40    let mut pending = children(clause);
41    let mut aliases = aliases;
42    while let Some(node) = pending.pop() {
43        if node.kind() == "as_pattern" {
44            aliases.extend(field(node, "alias"));
45        } else {
46            pending.extend(children(node));
47        }
48    }
49    let names = aliases
50        .into_iter()
51        .flat_map(|alias| target_names(alias, source))
52        .collect::<Vec<_>>();
53    if names.is_empty() {
54        return;
55    }
56    excluded.extend(names.iter().map(|(_, span)| span.clone()));
57    add_names(clause, names, root, facts, clause.end_byte());
58}

60pub(super) fn collect_case_targets(
61    clause: Node<'_>,
62    root: Node<'_>,
63    source: &[u8],
64    facts: &mut SemanticFacts,
65    excluded: &mut Vec<Range<usize>>,
66) {
67    for pattern in children(clause)
68        .into_iter()
69        .filter(|child| child.kind() == "case_pattern")
70    {
71        let names = pattern_captures(pattern, source);
72        excluded.extend(names.iter().map(|(_, span)| span.clone()));
73        add_names(clause, names, root, facts, clause.end_byte());
74    }
75}

77fn pattern_captures(pattern: Node<'_>, source: &[u8]) -> Vec<(String, Range<usize>)> {
78    let mut captures = Vec::new();
79    let mut pending = vec![pattern];
80    while let Some(node) = pending.pop() {
81        match node.kind() {
82            "identifier" if text(node, source) != "_" => {
83                if !node
84                    .parent()
85                    .is_some_and(|parent| parent.kind() == "class_pattern")
86                {
87                    captures.push((text(node, source), range(node)));
88                }
89            }
90            "dotted_name" => {
91                let parts = children(node);
92                if let [part] = parts.as_slice()
93                    && part.kind() == "identifier"
94                    && text(*part, source) != "_"
95                {
96                    captures.push((text(*part, source), range(*part)));
97                }
98            }
99            "keyword_identifier" | "string" | "integer" | "float" => {}
100            "keyword_pattern" => pending.extend(children(node).into_iter().skip(1)),
101            "class_pattern" => pending.extend(children(node).into_iter().skip(1)),
102            "dict_pattern" => pending.extend(dict_values(node)),
103            _ => pending.extend(children(node)),
104        }
105    }
106    captures
107}

109fn dict_values(pattern: Node<'_>) -> Vec<Node<'_>> {
110    let mut cursor = pattern.walk();
111    let mut values: Vec<_> = pattern
112        .children_by_field_name("value", &mut cursor)
113        .collect();
114    values.extend(
115        children(pattern)
116            .into_iter()
117            .filter(|child| child.kind() == "splat_pattern"),
118    );
119    values
120}

140fn bind_parameter_scope(
141    callable: Node<'_>,
142    source: &[u8],
143    facts: &mut SemanticFacts,
144    excluded: &mut Vec<Range<usize>>,
145) {
146    let Some(parameters) = field(callable, "parameters") else {
147        return;
148    };
149    let scope = field(callable, "body")
150        .map(range)
151        .unwrap_or_else(|| range(callable));
152    for parameter in children(parameters) {
153        for (name, name_range) in parameter_names(parameter, source) {
154            excluded.push(name_range.clone());
155            push_local(facts, name, name_range, scope.clone());
156        }
157    }
158}
