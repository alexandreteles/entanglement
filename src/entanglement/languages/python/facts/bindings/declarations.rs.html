9pub(super) fn collect_with_target(
10    item: Node<'_>,
11    root: Node<'_>,
12    source: &[u8],
13    facts: &mut SemanticFacts,
14    excluded: &mut Vec<Range<usize>>,
15) {
16    let mut pending = children(item);
17    while let Some(node) = pending.pop() {
18        if node.kind() == "as_pattern" {
19            if let Some(alias) = field(node, "alias") {
20                add_target_bindings(item, alias, root, source, facts, excluded, item.end_byte());
21            }
22        } else {
23            pending.extend(children(node));
24        }
25    }
26}

28pub(super) fn collect_named_expression(
29    node: Node<'_>,
30    root: Node<'_>,
31    source: &[u8],
32    facts: &mut SemanticFacts,
33    excluded: &mut Vec<Range<usize>>,
34) {
35    let Some(name_node) = field(node, "name") else {
36        return;
37    };
38    let name_range = range(name_node);
39    excluded.push(name_range.clone());
40    add_names(
41        node,
42        vec![(text(name_node, source), name_range)],
43        root,
44        facts,
45        node.end_byte(),
46    );
47}

49pub(super) fn collect_scope_declaration(
50    declaration: Node<'_>,
51    root: Node<'_>,
52    source: &[u8],
53    facts: &mut SemanticFacts,
54    excluded: &mut Vec<Range<usize>>,
55) {
56    let names = children(declaration)
57        .into_iter()
58        .filter(|node| node.kind() == "identifier")
59        .map(|node| (text(node, source), range(node)))
60        .collect::<Vec<_>>();
61    excluded.extend(names.iter().map(|(_, span)| span.clone()));
62    add_names(declaration, names, root, facts, declaration.start_byte());
63}

65pub(super) fn collect_deletion(
66    statement: Node<'_>,
67    root: Node<'_>,
68    source: &[u8],
69    facts: &mut SemanticFacts,
70    excluded: &mut Vec<Range<usize>>,
71) {
72    let names = children(statement)
73        .into_iter()
74        .flat_map(|target| {
75            if matches!(
76                target.kind(),
77                "identifier" | "tuple_pattern" | "list_pattern"
78            ) {
79                target_names(target, source)
80            } else {
81                Vec::new()
82            }
83        })
84        .collect::<Vec<_>>();
85    excluded.extend(names.iter().map(|(_, span)| span.clone()));
86    add_names(statement, names, root, facts, statement.end_byte());
87}

89pub(super) fn collect_definition(
90    node: Node<'_>,
91    root: Node<'_>,
92    source: &[u8],
93    facts: &mut SemanticFacts,
94    excluded: &mut Vec<Range<usize>>,
95) {
96    if !binding_scope_owner(node).is_some_and(|owner| owner.kind() == "class_definition") {
97        return;
98    }
99    let Some(name) = field(node, "name") else {
100        return;
101    };
102    let name_range = range(name);
103    excluded.push(name_range.clone());
104    add_names(
105        node,
106        vec![(text(name, source), name_range)],
107        root,
108        facts,
109        node.end_byte(),
110    );
