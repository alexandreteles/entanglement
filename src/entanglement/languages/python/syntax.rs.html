1use std::collections::HashMap;
2
3use tree_sitter::Node;

5/// Normalize comprehension clauses into their logical control-flow order.
6pub(super) fn normalize_comprehension_parents(
7    root: Node<'_>,
8    parents: &mut HashMap<usize, Option<usize>>,
9) {
10    let mut pending = vec![root];
11    while let Some(node) = pending.pop() {
12        if is_comprehension(node.kind()) {
13            normalize_comprehension(node, parents);
14        }
15        let mut cursor = node.walk();
16        pending.extend(node.named_children(&mut cursor));
17    }
18}
19
20fn is_comprehension(kind: &str) -> bool {
21    matches!(
22        kind,
23        "list_comprehension"
24            | "set_comprehension"
25            | "dictionary_comprehension"
26            | "generator_expression"
27    )
28}
29
30fn normalize_comprehension(comprehension: Node<'_>, parents: &mut HashMap<usize, Option<usize>>) {
31    let original_parent = parents.get(&comprehension.id()).copied().flatten();
32    let body = comprehension.child_by_field_name("body");
33    let mut current_parent = original_parent;
34    let mut final_clause = None;
35    let mut cursor = comprehension.walk();
36    for child in comprehension.named_children(&mut cursor) {
37        if Some(child.id()) == body.map(|node| node.id()) {
38            continue;
39        }
40        match child.kind() {
41            "for_in_clause" => {
42                parents.insert(child.id(), current_parent);
43                if let Some(right) = child.child_by_field_name("right") {
44                    parents.insert(right.id(), current_parent);
45                }
46                current_parent = Some(child.id());
47                final_clause = current_parent;
48            }
49            "if_clause" => {
50                parents.insert(child.id(), current_parent);
51                current_parent = Some(child.id());
52                final_clause = current_parent;
53            }
54            _ => {}
55        }
56    }
57    if let (Some(body), Some(parent)) = (body, final_clause) {
58        parents.insert(body.id(), Some(parent));
59    }
60}
