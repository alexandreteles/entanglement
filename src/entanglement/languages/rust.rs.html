104impl LanguageHandler for Analyzer {
105    fn capture(&self, tree: &Tree, source: &[u8], include_tokens: bool) -> Result<CapturedTree> {
106        let mut facts = CaptureFacts {
107            source,
108            ..Default::default()
109        };
110        let common = self
111            .query
112            .capture_with(tree, source, false, |_, captures, source| {
113                facts.record_semantics(&self.captures, captures, source);
114                facts.record_tags(&self.captures, captures, source);
115            });
116        Ok(facts.finish(tree, common, include_tokens))
117    }
118}

120impl<'a> CaptureFacts<'a> {
121    fn finish(mut self, tree: &Tree, mut common: QueryFacts, include_tokens: bool) -> CapturedTree {
122        deduplicate(&mut self.definitions, |item| {
123            (item.node.start_byte(), item.node.end_byte())
124        });
125        let mut modules: Vec<_> = self
126            .definitions
127            .iter()
128            .filter(|item| item.node.kind() == "mod_item")
129            .filter_map(|item| {
130                Some(ModuleSpan {
131                    name: node_text(item.name, self.source),
132                    body: item.node.child_by_field_name("body")?.byte_range(),
133                })
134            })
135            .collect();
136        modules.sort_by_key(|module| (module.body.end - module.body.start, module.body.start));
137        modules.reverse();
138        let root_range = tree.root_node().start_byte()..tree.root_node().end_byte();
139        let definitions = self.normalize_definitions(&modules, &root_range);
140        let mut imports = self.normalize_imports(&modules, &root_range);
141        let mut references = self.normalize_references(&modules, &root_range);
142        let locals = self.locals.iter().flat_map(local_bindings).collect();
143
144        deduplicate(&mut common.events, |event| (event.role, event.node_id));
145        deduplicate(&mut references, |reference| {
146            (reference.start_byte, reference.path.clone(), reference.kind)
147        });
148        deduplicate(&mut imports, |import| {
149            (
150                import.path.clone(),
151                import.alias.clone(),
152                import.scope_start,
153            )
154        });
155
156        common.tokens = if include_tokens {
157            super::rust_tokens::capture(tree.root_node(), self.source)
158        } else {
159            Vec::new()
160        };
161        CapturedTree::from(common).with_semantics(super::SemanticStore {
162            definitions,
163            imports,
164            references,
165            locals,
166            ..Default::default()
167        })
168    }
169
170    fn normalize_definitions(
171        &self,
172        modules: &[ModuleSpan],
173        root: &Range<usize>,
174    ) -> Vec<Definition> {
175        self.definitions
176            .iter()
177            .map(|item| {
178                let node = item.node;
179                let (scope_start, scope_end) = lexical_scope(node, true, root);
180                Definition {
181                    declaration: None,
182                    name: node_text(item.name, self.source),
183                    kind: item.kind.clone(),
184                    module: relative_module(node.start_byte(), modules),
185                    start_byte: node.start_byte(),
186                    end_byte: node.end_byte(),
187                    scope_start,
188                    scope_end,
189                    is_public: has_visibility(node),
190                    inline_module: node.kind() == "mod_item"
191                        && node.child_by_field_name("body").is_some(),
192                    external_module: node.kind() == "mod_item"
193                        && node.child_by_field_name("body").is_none(),
194                    context_id: 0,
195                }
196            })
197            .collect()
198    }
199}

201fn node_text(node: tree_sitter::Node<'_>, source: &[u8]) -> String {
202    String::from_utf8_lossy(&source[node.byte_range()]).into_owned()
203}

205fn deduplicate<T, K: Eq + std::hash::Hash>(values: &mut Vec<T>, key: impl Fn(&T) -> K) {
206    let mut seen = HashSet::new();
207    values.retain(|value| seen.insert(key(value)));
