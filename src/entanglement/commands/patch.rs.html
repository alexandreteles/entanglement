12/// Apply a diff in memory and resolve the full before and after file sets.
13///
14/// A file target accepts one patch whose old path matches that file. A
15/// directory target accepts patches relative to that directory. Reject empty
16/// diffs and attempts to overwrite another file. Never write source bytes.
17pub(super) fn analyze(
18    path: &Path,
19    diff: &Path,
20    single: bool,
21    selection: Selection,
22) -> crate::Result<model::AnalysisResult> {
23    let scope = if single || path.is_file() {
24        "local"
25    } else {
26        "repo"
27    };
28    let diff = input::read_diff(diff)?;
29    let patches = input::parse_diff(&diff)?;
30    let (root, source_paths) = paths::patch_sources(path, &patches, single)?;
31    let mut parsed = analysis::analyze_paths(source_paths, selection, std::convert::identity)?;
32    if single && parsed.is_empty() {
33        return Err("The selected file has no supported language".into());
34    }
35    let mut before: Vec<_> = parsed.iter().map(|file| file.facts.clone()).collect();
36    let before_snapshot = snapshot::Snapshot::default();
37    resolver::resolve_with_snapshot(&mut before, selection, &before_snapshot);
38    let before_structure = selection
39        .includes(crate::metrics::selection::Metric::Structure)
40        .then(|| crate::structure::analyze(&before, scope, &before_snapshot));
41    let after_snapshot = before_snapshot.clone();
42    let mut worker = analysis::Worker::new(selection)?;
43    let changes: Vec<_> = patches
44        .iter()
45        .map(|patch| apply_file_patch(&mut worker, &mut parsed, &root, patch, &after_snapshot))
46        .collect::<crate::Result<_>>()?;
47    let mut after: Vec<_> = parsed.into_iter().map(|file| file.facts).collect();
48    after.sort_by(|left, right| left.path.cmp(&right.path));
49    resolver::resolve_with_snapshot(&mut after, selection, &after_snapshot);
50    let structure = before_structure.map(|before| {
51        let mut report = crate::structure::analyze(&after, scope, &after_snapshot);
52        report.state.snapshot.removed_paths = changes
53            .iter()
54            .filter(|change| change.paths.new_path.is_none())
55            .filter_map(|change| change.paths.old_path.as_ref())
56            .map(|path| path.display().to_string())
57            .collect();
58        crate::structure::compare(before, report)
59    });
60    let files = changes
61        .into_iter()
62        .map(|change| change.compare(&before, &after, selection))
63        .collect();
64    Ok(model::AnalysisResult {
65        structure,
66        files: after.into_iter().map(|file| file.analysis).collect(),
67        patch: Some(model::PatchAnalysis { files }),
68        maintainability_index_bands: crate::metrics::maintainability::BANDS,
69    })
70}

72fn apply_file_patch(
73    worker: &mut analysis::Worker,
74    parsed: &mut Vec<analysis::ParsedFile>,
75    root: &Path,
76    patch: &input::FilePatch,
77    source_snapshot: &snapshot::Snapshot,
78) -> crate::Result<PatchChange> {
79    let paths = PatchPaths::new(root, patch)?;
80    paths.ensure_destination_available(parsed)?;
81    let old_index = paths.old_index(parsed);
82    let source = paths.source(parsed, old_index)?;
83    let applied = source_snapshot.apply_patch(
84        &source,
85        patch,
86        paths.old_path.as_deref(),
87        paths.new_path.as_deref(),
88    )?;
89    let new_file = analyze_after_patch(
90        worker,
91        parsed,
92        &paths,
93        old_index,
94        applied.source,
95        &applied.edits,
96    )?;
97    replace_parsed_file(parsed, old_index, new_file);
98    Ok(PatchChange {
99        paths,
100        edits: applied.edits,
101    })
102}

104fn analyze_after_patch(
105    worker: &mut analysis::Worker,
106    parsed: &[analysis::ParsedFile],
107    paths: &PatchPaths,
108    old_index: Option<usize>,
109    source: Vec<u8>,
110    edits: &[InputEdit],
111) -> crate::Result<Option<analysis::ParsedFile>> {
112    let Some(path) = &paths.new_path else {
113        if !source.is_empty() {
114            return Err("A deletion patch must remove the complete file".into());
115        }
116        return Ok(None);
117    };
118    let analysis_path = paths.analysis_path(parsed, old_index, path);
119    let Some(choice) = worker.select_file(analysis_path)? else {
120        return Ok(None);
121    };
122    worker
123        .analyze_selected_source(
124            analysis_path,
125            source,
126            old_index.map(|index| &parsed[index]),
127            edits,
128            choice,
129        )
130        .map(Some)
131}

133fn replace_parsed_file(
134    parsed: &mut Vec<analysis::ParsedFile>,
135    old_index: Option<usize>,
136    new_file: Option<analysis::ParsedFile>,
137) {
138    if let Some(index) = old_index {
139        parsed.remove(index);
140    }
141    if let Some(file) = new_file {
142        parsed.push(file);
143    }
