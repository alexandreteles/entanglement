9pub(in crate::analysis) fn summary(
10    choice: &LanguageChoice,
11    captured: CapturedTree,
12    context: SummaryContext<'_>,
13) -> TreeSummary {
14    let SummaryContext {
15        semantic_context,
16        metric_id,
17        excluded,
18        inherited_module,
19        owner_range,
20        inherited_scope,
21    } = context;
22    let prefix = &inherited_module.0;
23    let mut facts = captured.facts;
24    let semantic = &mut facts.semantic;
25    semantic.context_id = semantic_context;
26    semantic
27        .structure_resources
28        .retain(|item| !inside_any(item.start..item.end, &excluded));
29    semantic
30        .structure
31        .retain(|item| !excluded.iter().any(|range| range.contains(&item.start)));
32    semantic.resolution_family = Some(choice.resolution_family);
33    semantic.file_module_rules = choice.file_module_rules;
34    if let Some(php) = &mut semantic.php {
35        php.references.retain(|item| {
36            !excluded
37                .iter()
38                .any(|range| range.contains(&item.start_byte))
39        });
40        php.imports
41            .retain(|item| !inside_any(item.range.start_byte..item.range.end_byte, &excluded));
42        php.traits.retain(|item| {
43            !excluded
44                .iter()
45                .any(|range| range.contains(&item.start_byte))
46        });
47        php.includes.retain(|item| {
48            !excluded
49                .iter()
50                .any(|range| range.contains(&item.start_byte))
51        });
52    }
53    semantic
54        .receivers
55        .retain(|item| !inside_any(item.start_byte..item.end_byte, &excluded));
56    semantic
57        .calls
58        .retain(|item| !inside_any(item.start_byte..item.end_byte, &excluded));
59    let definitions = &mut facts.definitions;
60    definitions.retain(|item| !inside_any(item.start_byte..item.end_byte, &excluded));
61    for definition in definitions.iter_mut() {
62        if excluded
63            .iter()
64            .any(|range| range.contains(&definition.start_byte))
65        {
66            definition.declaration = None;
67        }
68        definition.context_id = semantic_context;
69        inherit_scope(
70            &mut definition.scope_start,
71            &mut definition.scope_end,
72            owner_range,
73            inherited_scope,
74        );
75        prepend_module(&mut definition.module, prefix);
76    }
77    let imports = &mut facts.imports;
78    imports.retain(|item| !inside_any(item.start_byte..item.end_byte, &excluded));
79    for import in imports.iter_mut() {
80        import.context_id = semantic_context;
81        if import.scope_start == 0 && import.scope_end == 0 {
82            import.scope_start = owner_range.start;
83            import.scope_end = owner_range.end;
84        } else {
85            inherit_scope(
86                &mut import.scope_start,
87                &mut import.scope_end,
88                owner_range,
89                inherited_scope,
90            );
91        }
92        prepend_module(&mut import.module, prefix);
93    }
94    let exports = &mut facts.exports;
95    exports.retain(|item| !inside_any(item.start_byte..item.end_byte, &excluded));
96    for export in exports.iter_mut() {
97        export.context_id = semantic_context;
98        inherit_scope(
99            &mut export.scope_start,
100            &mut export.scope_end,
101            owner_range,
102            inherited_scope,
103        );
104    }
105    let references = &mut facts.references;
106    references.retain(|item| !inside_any(item.start_byte..item.end_byte, &excluded));
107    for reference in references.iter_mut() {
108        reference.context_id = semantic_context;
109        inherit_scope(
110            &mut reference.scope_start,
111            &mut reference.scope_end,
112            owner_range,
113            inherited_scope,
114        );
115        if reference.call_owner.is_none()
116            && reference.kind == crate::model::ReferenceKind::Call
117            && let Some(scope) = inherited_scope
118        {
119            reference.call_owner = Some(scope.start);
120        }
121        prepend_module(&mut reference.module, prefix);
122    }
123    let locals = &mut facts.locals;
124    locals.retain(|item| !inside_any(item.start_byte..item.end_byte, &excluded));
125    for local in locals.iter_mut() {
126        local.context_id = semantic_context;
127        inherit_scope(
128            &mut local.scope_start,
129            &mut local.scope_end,
130            owner_range,
131            inherited_scope,
132        );
133    }
134    let functions = captured
135        .functions
136        .into_iter()
137        .filter(|item| !inside_any(item.range.clone(), &excluded))
138        .collect();
139    let mut tokens = captured.tokens;
140    tokens.retain(|token| !inside_any(token.start_byte..token.end_byte, &excluded));
141    TreeSummary {
142        facts,
143        metric_id,
144        events: captured.events,
145        parents: captured.parents,
146        cognitive_parent: None,
147        functions,
148        tokens,
149        excluded,
150    }
151}

153impl TreeSummary {
154    /// Add explicit runtime bindings to this layer's isolated semantic context.
155    pub(in crate::analysis) fn add_serialized_bindings(&mut self, mut bindings: Vec<LocalBinding>) {
156        let mut seen = HashSet::new();
157        bindings.retain(|binding| {
158            seen.insert((
159                binding.name.clone(),
160                binding.start_byte,
161                binding.end_byte,
162                binding.scope_start,
163                binding.scope_end,
164            ))
165        });
166        self.facts
167            .locals
168            .extend(bindings.into_iter().map(|mut binding| {
169                binding.context_id = self.facts.semantic.context_id;
170                binding
171            }));
172    }
173
174    pub(in crate::analysis) fn discard_exports(&mut self) {
175        self.facts.exports.clear();
176    }
177
178    pub(in crate::analysis) fn share_top_level_bindings_into(
179        &self,
180        target: &mut TreeSummary,
181        owner_range: &Range<usize>,
182        target_scope: &Range<usize>,
183    ) {
184        target.facts.definitions.extend(
185            self.facts
186                .definitions
187                .iter()
188                .filter(|item| {
189                    item.scope_start <= owner_range.start && owner_range.end <= item.scope_end
190                })
191                .cloned()
192                .map(|mut item| {
193                    item.declaration = None;
194                    item.context_id = target.facts.semantic.context_id;
195                    item.scope_start = target_scope.start;
196                    item.scope_end = target_scope.end;
197                    item
198                }),
199        );
200        target.facts.imports.extend(
201            self.facts
202                .imports
203                .iter()
204                .filter(|item| {
205                    item.scope_start <= owner_range.start && owner_range.end <= item.scope_end
206                })
207                .cloned()
208                .map(|mut item| {
209                    item.context_id = target.facts.semantic.context_id;
210                    item.scope_start = target_scope.start;
211                    item.scope_end = target_scope.end;
212                    item
213                }),
214        );
215    }
216}

218fn prepend_module(module: &mut ModulePath, prefix: &[String]) {
219    if !prefix.is_empty() {
220        let mut path = prefix.to_vec();
221        path.append(&mut module.0);
222        module.0 = path;
223    }
224}

226fn inherit_scope(
227    scope_start: &mut usize,
228    scope_end: &mut usize,
229    owner_range: &Range<usize>,
230    inherited_scope: Option<&Range<usize>>,
231) {
232    if let Some(scope) = inherited_scope
233        .filter(|_| *scope_start <= owner_range.start && owner_range.end <= *scope_end)
234    {
235        *scope_start = scope.start;
236        *scope_end = scope.end;
237    }
238}
239
240fn inside_any(range: Range<usize>, excluded: &[Range<usize>]) -> bool {
241    excluded
242        .iter()
243        .any(|item| item.start <= range.start && range.end <= item.end)
244}
