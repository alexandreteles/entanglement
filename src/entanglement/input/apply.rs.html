5/// Apply one file patch to source bytes and return Tree-sitter edits.
6///
7/// `source` must contain the file named by `patch.old_path`. Pass an empty
8/// slice for a new file. The function checks each hunk against the current
9/// source, then applies hunks in order. It returns an error if a hunk does not
10/// match, if the hunk positions are invalid, or if an edit range is invalid.
11/// It does not write to disk.
12pub fn apply_patch(source: &[u8], patch: &FilePatch<'_>) -> crate::Result<AppliedPatch> {
13    let mut result = source.to_vec();
14    let mut edits = Vec::with_capacity(patch.hunks.len());
15    let mut line_delta = 0_isize;
16    let mut previous_old_end = 0_usize;
17
18    for hunk in patch.hunks.iter().flat_map(|patch| patch.hunks()) {
19        let old = hunk.old_range();
20        let new = hunk.new_range();
21        let old_line = hunk_line_index(old.start(), old.len())?;
22        let new_line = hunk_line_index(new.start(), new.len())?;
23        if old_line < previous_old_end {
24            return Err(invalid("unified diff hunks overlap or are out of order"));
25        }
26        let expected_new_line = old_line
27            .checked_add_signed(line_delta)
28            .ok_or_else(|| invalid("unified diff hunk position is out of range"))?;
29        if new_line != expected_new_line {
30            return Err(invalid("unified diff hunk positions do not match"));
31        }
32
33        let start_byte = line_start(&result, new_line)?;
34        let old_bytes = hunk_bytes(hunk.lines(), true);
35        let end_byte = start_byte
36            .checked_add(old_bytes.len())
37            .ok_or_else(|| invalid("unified diff byte range is out of range"))?;
38        if result.get(start_byte..end_byte) != Some(old_bytes.as_slice()) {
39            return Err(invalid(
40                "unified diff context does not match the source file",
41            ));
42        }
43        apply_hunk(&mut result, start_byte, hunk.lines(), &mut edits)?;
44
45        let changed_lines = isize::try_from(new.len())
46            .and_then(|new_count| isize::try_from(old.len()).map(|old_count| new_count - old_count))
47            .map_err(|_| invalid("unified diff line count is out of range"))?;
48        line_delta = line_delta
49            .checked_add(changed_lines)
50            .ok_or_else(|| invalid("unified diff line count is out of range"))?;
51        previous_old_end = old_line
52            .checked_add(old.len())
53            .ok_or_else(|| invalid("unified diff hunk position is out of range"))?;
54    }
55
56    Ok(AppliedPatch {
57        source: result,
58        edits,
59    })
60}

62fn hunk_bytes(lines: &[diffy::Line<'_, [u8]>], old: bool) -> Vec<u8> {
63    lines
64        .iter()
65        .filter_map(|line| match (old, line) {
66            (true, diffy::Line::Context(bytes) | diffy::Line::Delete(bytes)) => Some(*bytes),
67            (false, diffy::Line::Context(bytes) | diffy::Line::Insert(bytes)) => Some(*bytes),
68            _ => None,
69        })
70        .flatten()
71        .copied()
72        .collect()
73}

75fn apply_hunk(
76    source: &mut Vec<u8>,
77    start_byte: usize,
78    lines: &[diffy::Line<'_, [u8]>],
79    edits: &mut Vec<InputEdit>,
80) -> crate::Result<()> {
81    // The caller validated the complete old hunk image before applying it.
82    let mut cursor = start_byte;
83    let mut index = 0;
84    while index < lines.len() {
85        if let diffy::Line::Context(bytes) = lines[index] {
86            cursor += bytes.len();
87            index += 1;
88            continue;
89        }
90
91        let run_start = index;
92        while index < lines.len() && !matches!(lines[index], diffy::Line::Context(_)) {
93            index += 1;
94        }
95        let old_bytes = hunk_bytes(&lines[run_start..index], true);
96        let new_bytes = hunk_bytes(&lines[run_start..index], false);
97        let end_byte = cursor
98            .checked_add(old_bytes.len())
99            .ok_or_else(|| invalid("unified diff byte range is out of range"))?;
100        let new_end_byte = cursor
101            .checked_add(new_bytes.len())
102            .ok_or_else(|| invalid("unified diff byte range is out of range"))?;
103        if old_bytes != new_bytes {
104            let start_position = point_at(source, cursor);
105            let old_end_position = point_at(source, end_byte);
106            let new_end_position = point_after(start_position, &new_bytes);
107            edits.push(InputEdit {
108                start_byte: cursor,
109                old_end_byte: end_byte,
110                new_end_byte,
111                start_position,
112                old_end_position,
113                new_end_position,
114            });
115            drop(source.splice(cursor..end_byte, new_bytes.iter().copied()));
116        }
117        cursor = new_end_byte;
118    }
119    Ok(())
120}

122fn hunk_line_index(start: usize, count: usize) -> crate::Result<usize> {
123    if count == 0 {
124        Ok(start)
125    } else {
126        start
127            .checked_sub(1)
128            .ok_or_else(|| invalid("unified diff line numbers must start at one"))
129    }
130}

132fn line_start(source: &[u8], line_index: usize) -> crate::Result<usize> {
133    if line_index == 0 {
134        return Ok(0);
135    }
136    let mut line_count = 0;
137    for index in memchr::memchr_iter(b'\n', source) {
138        line_count += 1;
139        if line_count == line_index {
140            return Ok(index + 1);
141        }
142    }
143    if line_count + usize::from(source.last().is_some_and(|byte| *byte != b'\n')) == line_index {
144        Ok(source.len())
145    } else {
146        Err(invalid(
147            "unified diff hunk starts past the end of the source",
148        ))
149    }
150}

152fn point_at(source: &[u8], byte: usize) -> Point {
153    let prefix = &source[..byte];
154    let row = memchr::memchr_iter(b'\n', prefix).count();
155    let column =
156        memchr::memrchr(b'\n', prefix).map_or(prefix.len(), |newline| prefix.len() - newline - 1);
157    Point { row, column }
158}
159
160fn point_after(start: Point, inserted: &[u8]) -> Point {
161    let end = point_at(inserted, inserted.len());
162    Point {
163        row: start.row + end.row,
164        column: if end.row == 0 {
165            start.column + end.column
166        } else {
167            end.column
168        },
169    }
