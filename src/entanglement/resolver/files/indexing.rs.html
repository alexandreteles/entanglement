11pub(super) struct Index<'a, 'g> {
12    pub(super) facts: &'a [FileFactsView<'a>],
13    pub(super) graph: &'g ProgramGraph<'a>,
14    pub(super) by_path: BTreeMap<PathBuf, BTreeSet<usize>>,
15}

17impl<'a, 'g> Index<'a, 'g> {
18    /// Resolve an exact literal resource path through canonical source aliases.
19    pub(super) fn resolve_resource(&self, file: usize, source: &str) -> Vec<Target> {
20        if source.contains(':') || source.starts_with("//") || source.starts_with('#') {
21            return vec![Target::External];
22        }
23        if source.starts_with('/') {
24            return vec![Target::Unresolved];
25        }
26        let source = source.split(['?', '#']).next().unwrap_or(source);
27        let targets = self.facts[file]
28            .aliases
29            .iter()
30            .filter_map(|alias| alias.parent())
31            .filter_map(|parent| self.graph.aliases().get(&normalize(&parent.join(source))))
32            .flatten()
33            .copied()
34            .collect::<BTreeSet<_>>();
35        if targets.is_empty() {
36            vec![Target::Unresolved]
37        } else {
38            targets.into_iter().map(Target::Namespace).collect()
39        }
40    }
41
42    pub(super) fn new(facts: &'a [FileFactsView<'a>], graph: &'g ProgramGraph<'a>) -> Self {
43        let mut by_path = BTreeMap::<PathBuf, BTreeSet<usize>>::new();
44        for (path, indexes) in graph.aliases() {
45            for index in indexes {
46                if facts[*index].resolution_family == ResolutionFamily::FileModules {
47                    by_path.entry(normalize(path)).or_default().insert(*index);
48                }
49            }
50        }
51        Self {
52            facts,
53            graph,
54            by_path,
55        }
56    }
57
58    pub(super) fn resolve_module(&self, file: usize, source: &str) -> Vec<Target> {
59        let Some(rules) = self.facts[file].file_module_rules else {
60            return vec![Target::Unresolved];
61        };
62        match rules.layout {
63            FileModuleLayout::SlashRelative => layout::resolve_slash(self, file, source, rules),
64            FileModuleLayout::Dotted {
65                roots,
66                project_markers,
67                ..
68            } => layout::dotted::resolve(self, file, source, rules, roots, project_markers),
69        }
70    }
71}

73pub(super) fn module_candidates(requested: &Path, rules: &FileModuleRules) -> Vec<PathBuf> {
74    let extension = requested.extension().and_then(|value| value.to_str());
75    let recognized = extension.is_some_and(|extension| rules.extensions.contains(&extension));
76    let mut candidates = BTreeSet::from([requested.to_path_buf()]);
77    if let Some(extension) = extension.filter(|_| recognized) {
78        for mapped in remapped_extensions(extension, rules) {
79            candidates.insert(requested.with_extension(mapped));
80        }
81    } else {
82        for candidate in extension_candidates(requested, rules) {
83            candidates.insert(candidate);
84        }
85    }
86    candidates.into_iter().collect()
87}
88
89fn extension_candidates(requested: &Path, rules: &FileModuleRules) -> Vec<PathBuf> {
90    let mut candidates = Vec::new();
91    for extension in rules.extensions {
92        candidates.push(append_extension(requested, extension));
93        for stem in rules.index_stems {
94            candidates.push(append_extension(&requested.join(stem), extension));
95        }
96    }
97    candidates
98}

100fn append_extension(path: &Path, extension: &str) -> PathBuf {
101    let mut candidate = path.as_os_str().to_owned();
102    candidate.push(".");
103    candidate.push(extension);
104    candidate.into()
105}

107fn remapped_extensions<'a>(extension: &str, rules: &'a FileModuleRules) -> &'a [&'static str] {
108    rules
109        .remaps
110        .iter()
111        .find_map(|(source, targets)| (*source == extension).then_some(*targets))
112        .unwrap_or(&[])
113}

115pub(in crate::resolver) fn normalize(path: &Path) -> PathBuf {
116    let mut normalized = PathBuf::new();
117    for component in path.components() {
118        match component {
119            Component::CurDir => {}
120            Component::ParentDir if normalized.file_name().is_some() => {
121                normalized.pop();
122            }
123            Component::Prefix(_)
124            | Component::RootDir
125            | Component::Normal(_)
126            | Component::ParentDir => {
127                normalized.push(component.as_os_str());
128            }
129        }
130    }
131    normalized
132}
