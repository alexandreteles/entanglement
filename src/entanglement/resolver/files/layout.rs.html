12pub(super) fn resolve_slash(
13    index: &Index<'_, '_>,
14    file: usize,
15    source: &str,
16    rules: &FileModuleRules,
17) -> Vec<Target> {
18    if source.starts_with('#') {
19        return subpath::resolve(index, file, source, rules);
20    }
21    if is_unresolved_alias(source, rules) {
22        return vec![Target::Unresolved];
23    }
24    if !is_relative_slash(source) {
25        return vec![Target::External];
26    }
27    let requests = index.facts[file]
28        .aliases
29        .iter()
30        .filter_map(|alias| alias.parent().map(|parent| parent.join(source)));
31    resolve_requests(index, requests, rules)
32}

34pub(super) fn resolve_requests(
35    index: &Index<'_, '_>,
36    requests: impl IntoIterator<Item = PathBuf>,
37    rules: &FileModuleRules,
38) -> Vec<Target> {
39    let files = requests
40        .into_iter()
41        .flat_map(|path| module_candidates(&path, rules))
42        .filter_map(|path| index.by_path.get(&normalize(&path)))
43        .flatten()
44        .copied()
45        .filter(|file| {
46            index.facts[*file]
47                .file_module_rules
48                .is_some_and(|target| target.layout == rules.layout)
49        })
50        .collect::<BTreeSet<_>>();
51    if files.is_empty() {
52        vec![Target::Unresolved]
53    } else {
54        files.into_iter().map(Target::Namespace).collect()
55    }
56}

58pub(super) fn resolve_submodule(
59    index: &Index<'_, '_>,
60    package_file: usize,
61    name: &str,
62) -> Vec<Target> {
63    let fact = &index.facts[package_file];
64    let Some(rules) = fact.file_module_rules else {
65        return Vec::new();
66    };
67    if !matches!(
68        rules.layout,
69        crate::languages::FileModuleLayout::Dotted {
70            submodule_imports: true,
71            ..
72        }
73    ) {
74        return Vec::new();
75    }
76    let requests = fact
77        .aliases
78        .iter()
79        .filter(|path| {
80            path.file_stem()
81                .and_then(|stem| stem.to_str())
82                .is_some_and(|stem| rules.index_stems.contains(&stem))
83        })
84        .filter_map(|path| path.parent())
85        .filter_map(|parent| dotted::module_path(name).map(|module| parent.join(module)))
86        .collect::<BTreeSet<_>>();
87    resolve_requests(index, requests, rules)
88}

90fn is_relative_slash(source: &str) -> bool {
91    matches!(source, "." | "..") || source.starts_with("./") || source.starts_with("../")
92}

94fn is_unresolved_alias(source: &str, rules: &FileModuleRules) -> bool {
95    rules
96        .unresolved_prefixes
97        .iter()
98        .any(|prefix| source.starts_with(prefix))
99        || source.starts_with('/')
100}
