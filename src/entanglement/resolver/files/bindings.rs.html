8pub(super) enum BindingTarget<'a> {
9    Local(Vec<Target>),
10    Import(&'a Import),
11    Unresolved,
12}
13
14pub(super) fn binding_target<'a>(
15    fact: &'a FileFactsView<'_>,
16    graph: &ProgramGraph<'_>,
17    file_index: usize,
18    name: &str,
19    position: usize,
20    context_id: usize,
21) -> BindingTarget<'a> {
22    let local = local_symbols(graph, file_index, name, position, context_id)
23        .map(|(rank, targets)| (rank, BindingTarget::Local(targets)));
24    let shadow = local_binding_rank(fact, name, position, context_id)
25        .map(|rank| (rank, BindingTarget::Unresolved));
26    let imports = fact
27        .imports()
28        .filter(|import| {
29            import.source.is_some()
30                && import.alias.as_deref().or(import.imported_name.as_deref()) == Some(name)
31                && import.scope_start <= position
32                && position < import.scope_end
33                && import.context_id == context_id
34        })
35        .map(|import| {
36            (
37                import.scope_end.saturating_sub(import.scope_start),
38                BindingTarget::Import(import),
39            )
40        });
41    match best_ranked(local.into_iter().chain(shadow).chain(imports)) {
42        Some((_, mut targets)) if targets.len() == 1 => targets.pop().unwrap(),
43        _ => BindingTarget::Unresolved,
44    }
45}

47/// Retain all candidates in the nearest scope, including ties.
48pub(super) fn best_ranked<T>(items: impl Iterator<Item = (usize, T)>) -> Option<(usize, Vec<T>)> {
49    let mut best: Option<(usize, Vec<T>)> = None;
50    for (rank, item) in items {
51        match &mut best {
52            Some((nearest, items)) if rank == *nearest => items.push(item),
53            Some((nearest, _)) if rank > *nearest => {}
54            _ => best = Some((rank, vec![item])),
55        }
56    }
57    best
58}

60pub(super) fn local_binding_rank(
61    fact: &FileFactsView<'_>,
62    name: &str,
63    position: usize,
64    context_id: usize,
65) -> Option<usize> {
66    let rules = fact.file_module_rules;
67    let locals = fact
68        .locals()
69        .filter(|local| {
70            local.name == name
71                && local.scope_start <= position
72                && position < local.scope_end
73                && local.context_id == context_id
74                && (local.scope_start > 0
75                    || local.scope_end < fact.source.len()
76                    || rules.is_some_and(|rules| rules.module_bindings_shadow))
77                && !import_local_is_visible(fact, local, name, position, context_id)
78        })
79        .map(|local| local.scope_end.saturating_sub(local.scope_start));
80    let wildcard_imports = fact
81        .imports()
82        .filter(|import| {
83            rules.is_some_and(|rules| rules.wildcard_imports_shadow)
84                && import.imported_name.as_deref() == Some("*")
85                && import.context_id == context_id
86                && import.scope_start <= position
87                && position < import.scope_end
88        })
89        .map(|import| import.scope_end.saturating_sub(import.scope_start));
90    locals.chain(wildcard_imports).min()
91}

93fn import_local_is_visible(
94    fact: &FileFactsView<'_>,
95    local: &crate::model::LocalBinding,
96    name: &str,
97    position: usize,
98    context_id: usize,
99) -> bool {
100    fact.imports().any(|import| {
101        import.source.is_some()
102            && import.alias.as_deref().or(import.imported_name.as_deref()) == Some(name)
103            && import.context_id == context_id
104            && import.start_byte <= local.start_byte
105            && local.end_byte <= import.end_byte
106            && import.scope_start <= position
107            && position < import.scope_end
108    })
109}
