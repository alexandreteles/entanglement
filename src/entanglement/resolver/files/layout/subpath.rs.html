1//! Resolve Node subpath imports, such as SvelteKit's `#lib/*`, through the
2//! `imports` field of the nearest `package.json`.
3
4use std::path::{Path, PathBuf};

15/// The conditions TypeScript applies to ESM imports, in Node's matching order.
16const CONDITIONS: &[&str] = &["types", "import", "default"];
17
18#[derive(Deserialize)]
19struct PackageManifest {
20    #[serde(default)]
21    imports: IndexMap<String, ImportTarget>,
22}
23
24#[derive(Deserialize)]
25#[serde(untagged)]
26enum ImportTarget {
27    Path(String),
28    Conditions(IndexMap<String, ImportTarget>),
29    Other(serde::de::IgnoredAny),
30}
31
32pub(in crate::resolver::files) fn resolve(
33    index: &Index<'_, '_>,
34    file: usize,
35    source: &str,
36    rules: &FileModuleRules,
37) -> Vec<Target> {
38    let mut requests = Vec::new();
39    for alias in index.facts[file].aliases {
40        let Some((scope, target)) = package_target(alias, source) else {
41            return vec![Target::Unresolved];
42        };
43        match target.strip_prefix("./") {
44            Some(relative) => requests.push(scope.join(relative)),
45            None => return vec![Target::External],
46        }
47    }
48    resolve_requests(index, requests, rules)
49}

51/// Node reads only the nearest `package.json`, prefers an exact key, and
52/// otherwise uses the `*` pattern with the longest prefix.
53fn package_target(file: &Path, specifier: &str) -> Option<(PathBuf, String)> {
54    let scope = file
55        .ancestors()
56        .skip(1)
57        .find(|directory| directory.join("package.json").is_file())?;
58    let manifest = std::fs::read_to_string(scope.join("package.json")).ok()?;
59    let imports = serde_json::from_str::<PackageManifest>(&manifest)
60        .ok()?
61        .imports;
62    let target = match imports.get(specifier).filter(|_| !specifier.contains('*')) {
63        Some(target) => target.path()?.to_owned(),
64        None => {
65            let (key, matched) = imports
66                .keys()
67                .filter_map(|key| pattern_match(key, specifier).map(|matched| (key, matched)))
68                .max_by_key(|(key, _)| (key.find('*'), key.len()))?;
69            imports[key].path()?.replace('*', matched)
70        }
71    };
72    Some((scope.to_path_buf(), target))
73}
74
75fn pattern_match<'s>(key: &str, specifier: &'s str) -> Option<&'s str> {
76    let (prefix, suffix) = key.split_once('*')?;
77    if suffix.contains('*') || specifier.len() < key.len() {
78        return None;
79    }
80    specifier.strip_prefix(prefix)?.strip_suffix(suffix)
81}
82
83impl ImportTarget {
84    fn path(&self) -> Option<&str> {
85        match self {
86            Self::Path(path) => Some(path),
87            Self::Conditions(conditions) => conditions.iter().find_map(|(condition, target)| {
88                CONDITIONS
89                    .contains(&condition.as_str())
90                    .then(|| target.path())
91                    .flatten()
92            }),
93            Self::Other(_) => None,
94        }
95    }
96}
