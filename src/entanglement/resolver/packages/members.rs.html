21pub(super) fn index(graph: &ProgramGraph<'_>) -> BTreeMap<PackageKey, Package> {
22    let mut packages = BTreeMap::<PackageKey, Package>::new();
23    for file_index in 0..graph.file_count() {
24        let fact = graph.file(file_index);
25        if !is_go(fact) {
26            continue;
27        }
28        let ids = graph
29            .symbols_for_file(file_index)
30            .map(|item| item.scope.context_id)
31            .collect::<BTreeSet<_>>();
32        for id in ids {
33            for context in contexts::for_reference(fact, graph, file_index, id) {
34                let package = packages.entry(context.key.clone()).or_default();
35                package.production |= !context.test;
36                add_members(package, graph, file_index, id, &context);
37            }
38        }
39    }
40    packages
41}

43fn add_members(
44    package: &mut Package,
45    graph: &ProgramGraph<'_>,
46    file_index: usize,
47    context_id: usize,
48    context: &Context,
49) {
50    for symbol in graph.symbols_for_file(file_index).filter(|item| {
51        item.scope.context_id == context_id
52            && !matches!(item.kind, DefinitionKind::Method | DefinitionKind::Module)
53    }) {
54        package
55            .members
56            .entry(symbol.name.clone())
57            .or_default()
58            .push(Member {
59                symbol: symbol.semantic_id.clone(),
60                public: symbol.is_public,
61                test: context.test,
62                conditional: context.conditional,
63            });
64    }
65}

67impl Index {
68    pub(super) fn own_members(&self, context: &Context, name: &str) -> Option<Resolution> {
69        let members = self.packages.get(&context.key)?.members.get(name)?;
70        select(members.iter().filter(|member| context.test || !member.test))
71    }
72
73    pub(super) fn imported_members(&self, packages: &[PackageKey], name: &str) -> Resolution {
74        // Missing members in one possible package must not disappear from the outcome.
75        agree(packages.iter().map(|key| {
76            let members = self
77                .packages
78                .get(key)
79                .and_then(|package| package.members.get(name));
80            select(
81                members
82                    .into_iter()
83                    .flatten()
84                    .filter(|member| member.public && !member.test),
85            )
86            .unwrap_or(Resolution::Unresolved)
87        }))
88    }
89}

91fn select<'a>(members: impl Iterator<Item = &'a Member>) -> Option<Resolution> {
92    let members = members.collect::<Vec<_>>();
93    if members.is_empty() {
94        return None;
95    }
96    let symbols = members
97        .iter()
98        .map(|item| item.symbol.clone())
99        .collect::<BTreeSet<_>>();
100    if symbols.len() == 1 && members.iter().any(|item| item.conditional) {
101        return Some(Resolution::Unresolved);
102    }
103    Some(symbol_resolution(symbols))
104}

106/// Combine full context outcomes, never just their successful symbol matches.
107pub(super) fn agree(outcomes: impl IntoIterator<Item = Resolution>) -> Resolution {
108    let mut symbols = BTreeSet::new();
109    let mut external = false;
110    for outcome in outcomes {
111        match outcome {
112            Resolution::Exact(symbol) => {
113                symbols.insert(symbol);
114            }
115            Resolution::Ambiguous(items) => symbols.extend(items),
116            Resolution::External => external = true,
117            Resolution::Unresolved => return Resolution::Unresolved,
118        }
119    }
120    if external {
121        return if symbols.is_empty() {
122            Resolution::External
123        } else {
124            Resolution::Unresolved
125        };
126    }
127    symbol_resolution(symbols)
128}
129
130fn symbol_resolution(symbols: BTreeSet<SymbolId>) -> Resolution {
131    match symbols.len() {
132        0 => Resolution::Unresolved,
133        1 => Resolution::Exact(symbols.into_iter().next().expect("one symbol")),
134        _ => Resolution::Ambiguous(symbols.into_iter().collect()),
135    }
136}
