3pub(crate) fn project_dependencies(
4    root: &Path,
5    snapshot: &Snapshot,
6    cache: &mut ProjectCache,
7) -> BTreeMap<String, BTreeSet<build::ProjectDependency>> {
8    let mut dependencies = BTreeMap::<String, BTreeSet<build::ProjectDependency>>::new();
9    let mut module_roots = vec![(":".to_owned(), root.to_path_buf())];
10    module_roots.extend_from_slice(cache.modules(root, snapshot));
11    for (module, path) in module_roots {
12        let mut deps = BTreeSet::new();
13        for name in ["build.gradle.kts", "build.gradle"] {
14            deps.extend(
15                cache
16                    .script(path.join(name), snapshot)
17                    .dependencies
18                    .iter()
19                    .cloned(),
20            );
21        }
22        dependencies.insert(module, deps);
23    }
24    dependencies
25}

27pub(crate) fn source_set_reaches(
28    caller: &str,
29    target: &str,
30    edges: &BTreeMap<String, BTreeSet<String>>,
31) -> bool {
32    let mut pending = vec![caller.to_owned()];
33    let mut seen = HashSet::new();
34    while let Some(current) = pending.pop() {
35        if current == target {
36            return true;
37        }
38        if !seen.insert(current.clone()) {
39            continue;
40        }
41        pending.extend(edges.get(&current).into_iter().flatten().cloned());
42    }
43    false
44}

46pub(crate) fn conventional_source_edge(caller: &str, target: &str) -> bool {
47    if caller == "commonTest" && target == "commonMain" {
48        return true;
49    }
50    if caller.ends_with("Test") {
51        let main = caller.trim_end_matches("Test").to_owned() + "Main";
52        return target == main || target == "commonTest" || target == "commonMain";
53    }
54    caller.ends_with("Main") && target == "commonMain"
55}

57pub(crate) fn transitive_closure(
58    root: &str,
59    graph: &BTreeMap<String, BTreeSet<build::ProjectDependency>>,
60    source_set: Option<&str>,
61) -> BTreeSet<String> {
62    let mut result = BTreeSet::new();
63    let mut pending = graph
64        .get(root)
65        .into_iter()
66        .flatten()
67        .filter(|dependency| {
68            dependency_visible_in_source_set(&dependency.scope, source_set)
69                && dependency_compile_visible(&dependency.exposure)
70        })
71        .map(|dependency| dependency.path.clone())
72        .collect::<Vec<_>>();
73    while let Some(item) = pending.pop() {
74        if result.insert(item.clone()) {
75            pending.extend(
76                graph
77                    .get(&item)
78                    .into_iter()
79                    .flatten()
80                    .filter(|dependency| {
81                        dependency_visible_in_source_set(&dependency.scope, source_set)
82                            && dependency.exposure == build::DependencyExposure::Api
83                    })
84                    .map(|dependency| dependency.path.clone()),
85            );
86        }
87    }
88    result
89}

91fn dependency_compile_visible(exposure: &build::DependencyExposure) -> bool {
92    matches!(
93        exposure,
94        build::DependencyExposure::Api
95            | build::DependencyExposure::Implementation
96            | build::DependencyExposure::CompileOnly
97    )
98}

100fn dependency_visible_in_source_set(
101    scope: &build::DependencyScope,
102    source_set: Option<&str>,
103) -> bool {
104    match scope {
105        build::DependencyScope::Production => true,
106        build::DependencyScope::Test(name) | build::DependencyScope::Variant(name) => {
107            source_set.is_some_and(|set| set == name)
108        }
109        build::DependencyScope::Other => false,
110    }
