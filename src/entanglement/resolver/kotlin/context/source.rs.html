4pub(crate) struct SourceContext {
5    pub(crate) path: PathBuf,
6    pub(crate) project_root: PathBuf,
7    pub(crate) module_root: PathBuf,

10    /// Gradle's module coordinate relative to project_root (for example `:lib`).

12    pub(crate) source_set: Option<String>,
13    pub(crate) isolated_source: bool,

16/// Project evidence belongs to one resolution snapshot, never to a global cache.
17#[derive(Default)]
18pub(in crate::resolver) struct ProjectCache {
19    modules: HashMap<PathBuf, Vec<(String, PathBuf)>>,
20    scripts: HashMap<PathBuf, build::BuildFacts>,
21}
22
23impl ProjectCache {
24    pub(super) fn modules(&mut self, root: &Path, snapshot: &Snapshot) -> &[(String, PathBuf)] {
25        self.modules.entry(root.to_path_buf()).or_insert_with(|| {
26            let (source, kind) = ["settings.gradle.kts", "settings.gradle"]
27                .iter()
28                .find_map(|name| {
29                    read_text(snapshot, &root.join(name))
30                        .map(|source| (source, build::ScriptKind::for_path(Path::new(name))))
31                })
32                .unwrap_or_else(|| (String::new(), build::ScriptKind::Kotlin));
33            let settings = build::parse(&source, kind).0;
34            settings
35                .modules
36                .into_iter()
37                .map(|module| {
38                    let directory =
39                        settings
40                            .directories
41                            .get(&module)
42                            .cloned()
43                            .unwrap_or_else(|| {
44                                PathBuf::from(module.trim_start_matches(':').replace(':', "/"))
45                            });
46                    (module, root.join(directory))
47                })
48                .collect()
49        })
50    }
51
52    pub(super) fn script(&mut self, path: PathBuf, snapshot: &Snapshot) -> &build::BuildFacts {
53        self.scripts.entry(path.clone()).or_insert_with(|| {
54            read_text(snapshot, &path).map_or_else(build::BuildFacts::default, |source| {
55                build::parse(&source, build::ScriptKind::for_path(&path)).1
56            })
57        })
58    }
59}

61pub(crate) fn source_context(
62    path: &Path,
63    snapshot: &Snapshot,
64    cache: &mut ProjectCache,
65) -> SourceContext {
66    let absolute = path.to_path_buf();
67    let settings_root = manifest_root(path, snapshot, &["settings.gradle.kts", "settings.gradle"]);
68    let source_parts = path
69        .components()
70        .filter_map(|component| match component {
71            Component::Normal(part) => part.to_str().map(str::to_owned),
72            _ => None,
73        })
74        .collect::<Vec<_>>();
75    let source_index = source_parts.iter().rposition(|part| part == "src");
76    let source_set = source_index
77        .filter(|index| *index + 2 < source_parts.len())
78        .filter(|index| matches!(source_parts[*index + 2].as_str(), "kotlin" | "java"))
79        .map(|index| source_parts[index + 1].clone());
80    let inferred_module_root = source_index
81        .and_then(|index| {
82            path.ancestors()
83                .nth(path.components().count().saturating_sub(index + 1))
84        })
85        .map(Path::to_path_buf);
86    let build_root = manifest_root(path, snapshot, &["build.gradle.kts", "build.gradle"]);
87    let settings_root = settings_root.or_else(|| build_root.clone());
88    let project_root = settings_root
89        .clone()
90        .or_else(|| inferred_module_root.clone())
91        .unwrap_or_else(|| absolute.parent().unwrap_or(&absolute).to_path_buf());
92    let module_root = build_root
93        .or(inferred_module_root)
94        .unwrap_or_else(|| project_root.clone());
95    let gradle_module = module_name(
96        &project_root,
97        &module_root,
98        cache.modules(&project_root, snapshot),
99    );
100    let module = format!("{}{}", project_root.display(), gradle_module);
101    SourceContext {
102        path: absolute.clone(),
103        project_root,
104        module_root,
105        module,
106        gradle_module,
107        source_set,
108        isolated_source: path.extension().and_then(|extension| extension.to_str()) != Some("kt"),
109    }
110}

112fn manifest_root(path: &Path, snapshot: &Snapshot, names: &[&str]) -> Option<PathBuf> {
113    path.ancestors()
114        .find(|directory| {
115            names.iter().any(|name| {
116                snapshot
117                    .read(&directory.join(name))
118                    .ok()
119                    .flatten()
120                    .is_some()
121            })
122        })
123        .map(Path::to_path_buf)
124}

126fn module_name(project_root: &Path, module_root: &Path, modules: &[(String, PathBuf)]) -> String {
127    let normalized_root = normalize_path(module_root);
128    if let Some((module, _)) = modules
129        .iter()
130        .find(|(_, path)| normalize_path(path) == normalized_root)
131    {
132        return module.clone();
133    }
134    if normalize_path(project_root) == normalized_root {
135        return ":".to_owned();
136    }
137    module_root
138        .strip_prefix(project_root)
139        .ok()
140        .map(|relative| {
141            relative
142                .components()
143                .filter_map(|part| part.as_os_str().to_str())
144                .collect::<Vec<_>>()
145        })
146        .filter(|parts| !parts.is_empty())
147        .map(|parts| format!(":{}", parts.join(":")))
148        .unwrap_or_else(|| ":".to_owned())
149}

151fn normalize_path(path: &Path) -> PathBuf {
152    let mut normalized = PathBuf::new();
153    for component in path.components() {
154        match component {
155            Component::CurDir => {}
156            Component::ParentDir => {
157                normalized.pop();
158            }
159            other => normalized.push(other.as_os_str()),
160        }
161    }
162    normalized
163}

165fn read_text(snapshot: &Snapshot, path: &Path) -> Option<String> {
166    snapshot
167        .read(path)
168        .ok()
169        .flatten()
170        .and_then(|bytes| String::from_utf8(bytes.to_vec()).ok())
