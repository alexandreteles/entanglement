3use super::ast::{Call, ScriptKind, children, identifier, literal, project_path, text, valid_name};
4use super::{BuildFacts, DependencyScope, ProjectDependency, Settings, classify_configuration};
5use std::path::PathBuf;
6use tree_sitter::Node;

8#[derive(Default)]
9struct ScriptFacts {
10    settings: Settings,
11    build: BuildFacts,
12}
13
14#[derive(Clone)]
15enum Scope {
16    Root,
17    Kotlin,
18    Android,
19    SourceSets(bool),
20    Variants,
21    SourceSet {
22        name: String,
23        dependencies: bool,
24        edges: bool,
25    },
26    Dependencies(DependencyScope),
27}

29pub(in crate::resolver::kotlin) fn parse(source: &str, kind: ScriptKind) -> (Settings, BuildFacts) {
30    let mut facts = ScriptFacts::default();
31    if let Some(tree) = kind.parse(source) {
32        facts.walk(tree.root_node(), source, Scope::Root);
33    }
34    (facts.settings, facts.build)
35}
36
37impl ScriptFacts {
38    fn walk(&mut self, body: Node<'_>, source: &str, scope: Scope) {
39        for node in children(body) {
40            match (&scope, node.kind()) {
41                (Scope::Root, "assignment") => self.settings.directories.extend(
42                    project_directory(node, source).map(|(module, path)| (module.into(), path)),
43                ),
44                (Scope::SourceSets(_), "property_declaration") => {
45                    if let Some((name, call)) = declared_source_set(node, source)
46                        && let Some(body) = call.body
47                    {
48                        self.walk(
49                            body,
50                            source,
51                            Scope::SourceSet {
52                                name: name.into(),
53                                dependencies: ["getting", "creating"].contains(&call.name)
54                                    && call.args.is_empty(),
55                                edges: true,
56                            },
57                        );
58                    }
59                }
60                _ => {
61                    if let Some(call) = Call::read(node, source) {
62                        self.call(call, source, &scope);
63                    }
64                }
65            }
66        }
67    }
68
69    fn call(&mut self, call: Call<'_, '_>, source: &str, scope: &Scope) {
70        match scope {
71            Scope::Root if call.name == "include" && call.receiver.is_none() => {
72                self.settings.modules.extend(
73                    call.args
74                        .iter()
75                        .filter_map(|node| literal(*node, source))
76                        .filter(|value| project_path(value))
77                        .map(str::to_owned),
78                );
79            }
80            Scope::Dependencies(scope) if call.receiver.is_none() => {
81                self.dependencies(&call, source, scope);
82            }
83            Scope::SourceSet {
84                name, edges: true, ..
85            } if call.name == "dependsOn" && call.receiver.is_none() => {
86                self.source_edge(name, &call, source);
87            }
88            Scope::SourceSets(_) if call.name == "dependsOn" => {
89                if let Some(child) = call.receiver {
90                    self.source_edge(child, &call, source);
91                }
92            }
93            _ => {}
94        }
95        if let Some(body) = call.body
96            && let Some(next) = child_scope(scope, &call, source)
97        {
98            self.walk(body, source, next);
99        }
100    }
101
102    fn dependencies(&mut self, call: &Call<'_, '_>, source: &str, scope: &DependencyScope) {
103        let (declared_scope, exposure) = classify_configuration(call.name);
104        let scope = match scope {
105            DependencyScope::Production => declared_scope,
106            scope => scope.clone(),
107        };
108        self.build
109            .dependencies
110            .extend(call.args.iter().filter_map(|node| {
111                let project = Call::read(*node, source)
112                    .filter(|call| call.name == "project" && call.receiver.is_none())?;
113                let path = project.string(source).filter(|path| project_path(path))?;
114                Some(ProjectDependency {
115                    path: path.into(),
116                    configuration: call.name.into(),
117                    scope: scope.clone(),
118                    exposure: exposure.clone(),
119                })
120            }));
121    }
122
123    fn source_edge(&mut self, child: &str, call: &Call<'_, '_>, source: &str) {
124        let Some(first) = call.args.first() else {
125            return;
126        };
127        // Preserve the existing static extractor's first-token rule.
128        let mut head = *first;
129        while let Some(child) = head
130            .named_child(0)
131            .filter(|child| child.start_byte() == first.start_byte())
132        {
133            head = child;
134        }
135        let parent = identifier(head, source).or_else(|| call.string(source));
136        if let Some(parent) = parent.filter(|name| valid_name(name)) {
137            self.build
138                .source_edges
139                .entry(child.into())
140                .or_default()
141                .insert(parent.into());
142        }
143    }
144}

146fn child_scope(scope: &Scope, call: &Call<'_, '_>, source: &str) -> Option<Scope> {
147    match (scope, call.receiver, call.name) {
148        (Scope::Root, None, "kotlin") => Some(Scope::Kotlin),
149        (Scope::Root, None, "android") => Some(Scope::Android),
150        (Scope::Root, None, "dependencies") => {
151            Some(Scope::Dependencies(DependencyScope::Production))
152        }
153        (Scope::Kotlin, None, "sourceSets") => Some(Scope::SourceSets(true)),
154        (Scope::Android, None, "sourceSets") => Some(Scope::SourceSets(false)),
155        (Scope::Android, None, "buildTypes" | "productFlavors") => Some(Scope::Variants),
156        (
157            Scope::SourceSet {
158                name,
159                dependencies: true,
160                ..
161            },
162            None,
163            "dependencies",
164        ) => Some(Scope::Dependencies(DependencyScope::Variant(name.clone()))),
165        (Scope::SourceSets(true), Some(name), "dependencies") => {
166            Some(Scope::Dependencies(DependencyScope::Variant(name.into())))
167        }
168        (Scope::SourceSets(_) | Scope::Variants, None, _) => {
169            named_source_set(call, source).map(|name| Scope::SourceSet {
170                name,
171                dependencies: true,
172                edges: matches!(scope, Scope::SourceSets(_))
173                    && matches!(call.name, "getByName" | "named"),
174            })
175        }
176        _ => None,
177    }
178}
179
180fn named_source_set(call: &Call<'_, '_>, source: &str) -> Option<String> {
181    let name = match call.name {
182        "getByName" | "named" | "create" | "maybeCreate" => call.string(source)?,
183        "getting" | "creating" => return None,
184        name if call.args.is_empty() => name,
185        _ => return None,
186    };
187    valid_name(name).then(|| name.into())
188}
189
190fn declared_source_set<'tree, 'source>(
191    node: Node<'tree>,
192    source: &'source str,
193) -> Option<(&'source str, Call<'tree, 'source>)> {
194    let variable = children(node).find(|child| child.kind() == "variable_declaration")?;
195    let name = identifier(variable.named_child(0)?, source)?;
196    valid_name(name).then_some(())?;
197    let delegate = children(node).find(|child| child.kind() == "property_delegate")?;
198    let call = Call::read(delegate.named_child(0)?, source)?;
199    Some((name, call))
200}

202fn project_directory<'a>(node: Node<'_>, source: &'a str) -> Option<(&'a str, PathBuf)> {
203    let left = node.named_child(0)?;
204    let property = children(left).last()?;
205    (text(property, source).trim_start_matches('.') == "projectDir").then_some(())?;
206    let project = Call::read(left.named_child(0)?, source)?;
207    (project.name == "project" && project.receiver.is_none()).then_some(())?;
208    let module = project.string(source).filter(|path| project_path(path))?;
209    let file = Call::read(node.named_child(1)?, source)?;
210    (file.name == "file" && file.receiver.is_none()).then_some(())?;
211    let path = PathBuf::from(file.string(source)?);
212    (!path.is_absolute()).then_some((module, path))
213}
