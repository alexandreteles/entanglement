1//! Borrowed syntax views for static Gradle calls. Build scripts are never executed.
2
3use tree_sitter::{Node, Parser, Tree};
4
5#[derive(Clone, Copy)]
6pub(in crate::resolver::kotlin) enum ScriptKind {
7    Kotlin,
8    Groovy,
9}
10
11impl ScriptKind {
12    pub(in crate::resolver::kotlin) fn for_path(path: &std::path::Path) -> Self {
13        match path.extension().and_then(|extension| extension.to_str()) {
14            Some("kts") => Self::Kotlin,
15            _ => Self::Groovy,
16        }
17    }
18
19    pub(super) fn parse(self, source: &str) -> Option<Tree> {
20        let language = match self {
21            Self::Kotlin => tree_sitter_kotlin::LANGUAGE,
22            Self::Groovy => tree_sitter_groovy::LANGUAGE,
23        };
24        let mut parser = Parser::new();
25        parser.set_language(&language.into()).ok()?;
26        parser.parse(source, None)
27    }
28}

30pub(super) fn children(node: Node<'_>) -> impl Iterator<Item = Node<'_>> {
31    (0..node.named_child_count()).filter_map(move |index| node.named_child(index as u32))
32}
33
34pub(super) fn text<'a>(node: Node<'_>, source: &'a str) -> &'a str {
35    &source[node.byte_range()]
36}
37
38pub(super) fn literal<'a>(node: Node<'_>, source: &'a str) -> Option<&'a str> {
39    matches!(node.kind(), "string" | "string_literal").then_some(())?;
40    let raw = text(node, source);
41    raw.starts_with(['\'', '"']).then_some(())?;
42    let width = if raw.starts_with("\"\"\"") || raw.starts_with("'''") {
43        3
44    } else {
45        1
46    };
47    let value = raw.get(width..raw.len().checked_sub(width)?)?;
48    (!node.has_error() && !value.contains(['\\', '$'])).then_some(value)
49}
50
51pub(super) fn identifier<'a>(node: Node<'_>, source: &'a str) -> Option<&'a str> {
52    matches!(node.kind(), "identifier" | "simple_identifier").then(|| text(node, source))
53}

55/// One direct invocation, with the grammar's argument and body ownership.
56pub(super) struct Call<'tree, 'source> {
57    pub(super) name: &'source str,
58    pub(super) receiver: Option<&'source str>,
59    pub(super) args: Vec<Node<'tree>>,
60    pub(super) body: Option<Node<'tree>>,
61}
62
63impl<'tree, 'source> Call<'tree, 'source> {
64    pub(super) fn read(node: Node<'tree>, source: &'source str) -> Option<Self> {
65        let (callee, arguments, body) = match node.kind() {
66            "call_expression" => {
67                let callee = node.named_child(0)?;
68                let suffix = children(node).find(|child| child.kind() == "call_suffix")?;
69                let arguments = children(suffix).find(|child| child.kind() == "value_arguments");
70                let body = children(suffix)
71                    .find(|child| child.kind() == "annotated_lambda")
72                    .and_then(|lambda| {
73                        children(lambda).find(|child| child.kind() == "lambda_literal")
74                    })
75                    .and_then(|lambda| children(lambda).find(|child| child.kind() == "statements"));
76                (callee, arguments, body)
77            }
78            "function_call" | "juxt_function_call" => {
79                let callee = node.child_by_field_name("function")?;
80                let arguments = node.child_by_field_name("args");
81                let body = arguments
82                    .and_then(|args| children(args).find(|child| child.kind() == "closure"));
83                (callee, arguments, body)
84            }
85            _ => return None,
86        };
87        let (name, receiver) = match callee.kind() {
88            "identifier" | "simple_identifier" => (text(callee, source), None),
89            "navigation_expression" | "dotted_identifier" => {
90                let raw = text(callee, source);
91                let (receiver, name) = raw.split_once('.')?;
92                (!name.contains('.') && valid_name(receiver) && valid_name(name)).then_some(())?;
93                (name, Some(receiver))
94            }
95            _ => return None,
96        };
97        let args = arguments
98            .into_iter()
99            .flat_map(children)
100            .filter(|child| !matches!(child.kind(), "closure" | "comment"))
101            .filter_map(|child| match child.kind() {
102                "value_argument" if child.named_child_count() == 1 => child.named_child(0),
103                "value_argument" => Some(child),
104                _ => Some(child),
105            })
106            .collect();
107        Some(Self {
108            name,
109            receiver,
110            args,
111            body,
112        })
113    }
114
115    pub(super) fn string(&self, source: &'source str) -> Option<&'source str> {
116        match self.args.as_slice() {
117            [value] => literal(*value, source),
118            _ => None,
119        }
120    }
121}

123pub(super) fn valid_name(name: &str) -> bool {
124    !name.is_empty()
125        && name
126            .bytes()
127            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'_')
128}
129
130pub(super) fn project_path(path: &str) -> bool {
131    path.strip_prefix(':').is_some_and(|path| {
132        path.split(':').all(|part| {
133            !part.is_empty()
134                && part
135                    .bytes()
136                    .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'_' | b'-' | b'.'))
137        })
138    })
139}
