15#[derive(Clone, Debug, Eq, Ord, PartialEq, PartialOrd)]
16pub(in crate::resolver::kotlin) enum DependencyScope {
17    /// Standard production configurations such as `implementation` and `api`.
18    Production,
19    /// Test configurations retain the exact source-set prefix.
20    Test(String),
21    /// A named variant or source set such as `debug` or `commonMain`.
22    Variant(String),
23    /// A known but unsupported configuration that cannot provide visibility.
24    Other,
25}

61pub(super) fn classify_configuration(configuration: &str) -> (DependencyScope, DependencyExposure) {
62    const SUFFIXES: [(&str, DependencyExposure); 6] = [
63        ("implementation", DependencyExposure::Implementation),
64        ("api", DependencyExposure::Api),
65        ("compileonly", DependencyExposure::CompileOnly),
66        ("runtimeonly", DependencyExposure::Runtime),
67        ("compile", DependencyExposure::Implementation),
68        ("runtime", DependencyExposure::Runtime),
69    ];
70    let normalized = configuration.to_ascii_lowercase();
71    let Some((suffix, exposure)) = SUFFIXES
72        .iter()
73        .find(|(suffix, _)| normalized.ends_with(*suffix))
74    else {
75        return (DependencyScope::Other, DependencyExposure::Other);
76    };
77    let prefix = &configuration[..configuration.len() - suffix.len()];
78    let scope = if prefix.is_empty() || prefix.eq_ignore_ascii_case("main") {
79        DependencyScope::Production
80    } else if prefix == "test" || prefix.ends_with("Test") {
81        DependencyScope::Test(prefix.to_owned())
82    } else {
83        DependencyScope::Variant(prefix.to_owned())
84    };
85    (scope, exposure.clone())
86}
