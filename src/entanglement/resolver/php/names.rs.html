5pub(super) fn key(kind: PhpSymbolKind, name: &str) -> String {
6    match kind {
7        PhpSymbolKind::Class | PhpSymbolKind::Function | PhpSymbolKind::Method => {
8            name.to_ascii_lowercase()
9        }
10        _ => name.to_owned(),
11    }
12}

14impl Index<'_> {
15    pub(super) fn expanded_names(
16        &self,
17        site: Site,
18        name: &str,
19        kind: PhpSymbolKind,
20    ) -> Vec<String> {
21        let Some(facts) = self.facts(site.file, site.context) else {
22            return Vec::new();
23        };
24        if let Some(name) = name.strip_prefix('\\') {
25            return vec![name.to_owned()];
26        }
27        let region = facts.namespace_at(site.position);
28        let namespace = region.map_or(String::new(), |namespace| namespace.name.join("\\"));
29        let qualify = |name: &str| match namespace.is_empty() {
30            true => name.to_owned(),
31            false => format!("{namespace}\\{name}"),
32        };
33        let (first, suffix) = name
34            .split_once('\\')
35            .map_or((name, None), |(first, rest)| (first, Some(rest)));
36        if first.eq_ignore_ascii_case("namespace")
37            && let Some(suffix) = suffix
38        {
39            return vec![qualify(suffix)];
40        }
41        let import_kind = match suffix {
42            Some(_) => PhpSymbolKind::Class,
43            None => kind,
44        };
45        let namespace_start = region.map_or(0, |namespace| namespace.range.start_byte);
46        let imports = facts
47            .imports
48            .iter()
49            .filter(|import| {
50                import.namespace_start == namespace_start
51                    && import.kind == import_kind
52                    && key(import_kind, &import.alias) == key(import_kind, first)
53            })
54            .map(|import| match suffix {
55                Some(suffix) => format!("{}\\{suffix}", import.path.join("\\")),
56                None => import.path.join("\\"),
57            })
58            .collect::<Vec<_>>();
59        match imports.is_empty() {
60            false => imports,
61            true if suffix.is_none()
62                && !namespace.is_empty()
63                && matches!(kind, PhpSymbolKind::Function | PhpSymbolKind::Constant) =>
64            {
65                vec![qualify(name), name.to_owned()]
66            }
67            true => vec![qualify(name)],
68        }
69    }

71    pub(super) fn lookup(
72        &self,
73        site: Site,
74        name: &str,
75        kind: PhpSymbolKind,
76        lexical: Option<&str>,
77    ) -> Vec<usize> {
78        match kind == PhpSymbolKind::Class
79            && ["self", "parent", "static"]
80                .iter()
81                .any(|keyword| name.eq_ignore_ascii_case(keyword))
82        {
83            true => {
84                let Some(class) = lexical else {
85                    return Vec::new();
86                };
87                let classes = self
88                    .lookup(site, &format!("\\{class}"), kind, None)
89                    .into_iter()
90                    .flat_map(|class| self.consumers(class, &mut HashSet::new()))
91                    .collect::<Vec<_>>();
92                match name.eq_ignore_ascii_case("parent") {
93                    false => classes,
94                    true => classes
95                        .into_iter()
96                        .flat_map(|class| self.parents(class))
97                        .filter(|parent| !self.candidates[*parent].php.is_interface)
98                        .collect(),
99                }
100            }
101            false => self
102                .expanded_names(site, name, kind)
103                .into_iter()
104                .find_map(|name| {
105                    let candidates = self
106                        .symbols
107                        .get(&(None, kind, key(kind, &name)))
108                        .into_iter()
109                        .flatten()
110                        .filter(|index| {
111                            self.visible(site.file, site.context, &self.candidates[**index])
112                        })
113                        .copied()
114                        .collect::<Vec<_>>();
115                    (!candidates.is_empty()).then_some(candidates)
116                })
117                .unwrap_or_default(),
118        }
119    }
120}

122pub(super) fn reference_kind(kind: PhpReferenceKind) -> PhpSymbolKind {
123    match kind {
124        PhpReferenceKind::Function => PhpSymbolKind::Function,
125        PhpReferenceKind::Constant | PhpReferenceKind::ClassConstant => PhpSymbolKind::Constant,
126        PhpReferenceKind::Method | PhpReferenceKind::StaticMethod => PhpSymbolKind::Method,
127        PhpReferenceKind::Property | PhpReferenceKind::StaticProperty => PhpSymbolKind::Property,
128        _ => PhpSymbolKind::Class,
129    }
