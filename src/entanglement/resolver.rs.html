81/// Resolve references in all facts and store their results in each analysis.
82///
83/// Call this after all files have been parsed. The function sorts symbols and
84/// ambiguity results to keep output stable. Only statically proven execution
85/// targets become call-graph edges.
86pub(crate) fn resolve_with_snapshot(
87    facts: &mut [FileFacts],
88    selection: crate::metrics::selection::Selection,
89    snapshot: &crate::snapshot::Snapshot,
90) {
91    let graph = ProgramGraph::new(facts);
92    let resolutions = strategies::resolve(facts, snapshot, &graph);
93    let calls = selection
94        .includes(crate::metrics::selection::Metric::Cogc)
95        .then(|| graph.confirmed_relations(&resolutions));
96    drop(graph);
97    facts
98        .par_iter_mut()
99        .zip(resolutions)
100        .for_each(|(fact, resolved)| fact.analysis.resolution = resolved);
101    if let Some(calls) = calls {
102        crate::metrics::recursion::annotate(facts, &calls);
103    }
104}

106fn resolve_references(
107    facts: &[FileFactsView<'_>],
108    snapshot: &crate::snapshot::Snapshot,
109    graph: &ProgramGraph<'_>,
110    kotlin_cache: &mut kotlin::ProjectCache,
111) -> Vec<Vec<ReferenceAnalysis>> {
112    let mut index = Index::new(graph);
113    index.resolve_imports();
114    let file_resolutions = files::resolve(facts, graph);
115    let kotlin_resolutions = kotlin::resolve(facts, snapshot, graph, kotlin_cache);
116    let package_resolutions = packages::resolve(facts, snapshot, graph);
117    let php_resolutions = php::resolve(facts, snapshot, graph);
118    facts
119        .iter()
120        .enumerate()
121        .map(|(file_index, fact)| {
122            let resolved = file_resolutions[file_index]
123                .as_ref()
124                .or(kotlin_resolutions[file_index].as_ref())
125                .or(php_resolutions[file_index].as_ref())
126                .or(package_resolutions[file_index].as_ref());
127            resolve_fact(&index, file_index, fact, resolved)
128        })
129        .collect::<Vec<_>>()
130}

132fn resolve_fact(
133    index: &Index,
134    file_index: usize,
135    fact: &FileFactsView<'_>,
136    resolved: Option<&Vec<ReferenceAnalysis>>,
137) -> Vec<ReferenceAnalysis> {
138    if let Some(resolved) = resolved {
139        return resolved.clone();
140    }
141    if fact.resolution_family != crate::languages::ResolutionFamily::RustCrates {
142        return Vec::new();
143    }
144    let context = &index.contexts[file_index];
145    fact.references()
146        .map(|reference| {
147            analysis(
148                reference,
149                index.to_resolution(
150                    context
151                        .iter()
152                        .map(|context| {
153                            match index.resolve_reference(file_index, context, reference, fact) {
154                                Resolution::Exact(symbol) => Target::Symbol(symbol),
155                                Resolution::Ambiguous(symbols) => Target::Ambiguous(symbols),
156                                Resolution::External => Target::External,
157                                Resolution::Unresolved => Target::Unresolved,
158                            }
159                        })
160                        .collect(),
161                ),
162                None,
163            )
164        })
165        .collect()
166}

168/// Build a reference result with no execution targets.
169fn analysis(
170    reference: &Reference,
171    resolution: Resolution,
172    dispatch: Option<CallDispatch>,
173) -> ReferenceAnalysis {
174    ReferenceAnalysis {
175        path: reference.path.clone(),
176        start_byte: reference.start_byte,
177        end_byte: reference.end_byte,
178        resolution,
179        dispatch,
180        execution_targets: Vec::new(),
181    }
182}

188/// Return a value only when the input contains exactly one item.
189fn only<T>(items: impl IntoIterator<Item = T>) -> Option<T> {
190    let mut items = items.into_iter();
191    let first = items.next()?;
192    items.next().is_none().then_some(first)
193}

195/// Resolve captured literal resources with the shared file and alias index.
196pub(crate) fn structural_resources(
197    facts: &[FileFacts],
198) -> Vec<(
199    usize,
200    usize,
201    crate::structure::resources::ResourceFact,
202    Resolution,
203)> {
204    let graph = ProgramGraph::new(facts);
205    let views = facts.iter().map(FileFactsView::new).collect::<Vec<_>>();
206    let resources = facts
207        .iter()
208        .enumerate()
209        .flat_map(|(file, fact)| {
210            fact.contexts.iter().flat_map(move |store| {
211                store
212                    .semantic
213                    .structure_resources
214                    .iter()
215                    .cloned()
216                    .map(move |resource| (file, store.semantic.context_id, resource))
217            })
218        })
219        .collect();
220    files::structural_resources(&views, &graph, resources)
221}

232/// Read selected member origins through the existing language strategies.
233pub(crate) fn structural_members(
234    facts: &[FileFacts],
235    snapshot: &crate::snapshot::Snapshot,
236) -> Vec<StructuralMemberSelection> {
237    let graph = ProgramGraph::new(facts);
238    let views = facts.iter().map(FileFactsView::new).collect::<Vec<_>>();
239    let mut selections = php::structural_members(&graph, snapshot);
240    selections.extend(kotlin::structural_members(&views, &graph, snapshot));
241    selections
242}
