12#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
13pub enum SyntaxRole {
14    /// A syntax node used to find concrete source rows.
15    Node,
16    /// A function boundary.
17    Function,
18    /// A control-flow decision.
19    Condition,
20    /// A short-circuit Boolean decision.
21    LogicalCondition,
22    /// A structural `if` decision for cognitive complexity.
23    CognitiveIf,
24    /// An expression scanned before its owning conditional increases nesting.
25    CognitiveConditionBoundary,
26    /// An `else` branch for cognitive complexity.
27    CognitiveElse,
28    /// An `else if` chain boundary, which suppresses the enclosing `if` depth.
29    CognitiveElseIf,
30    /// A loop for cognitive complexity.
31    CognitiveLoop,
32    /// A Rust `let ... else` conditional.
33    CognitiveLetElse,
34    /// A `match` expression for cognitive complexity.
35    CognitiveMultiway,
36    /// A catch clause for cognitive complexity.
37    CognitiveCatch,
38    /// A binary expression joining logical operators.
39    CognitiveLogicalExpression,
40    /// A logical AND operator.
41    CognitiveLogicalAnd,
42    /// A logical OR operator.
43    CognitiveLogicalOr,
44    /// Parentheses, which are transparent when flattening logical groups.
45    CognitiveParentheses,
46    /// A closure boundary that increases nesting for its contents.
47    CognitiveClosure,
48    /// A labeled `break` or `continue` jump.
49    CognitiveLabeledJump,
50    /// A decision with multiple cases.
51    Multiway,
52    /// One case in a multiway decision.
53    Case,
54    /// A comment range.
55    Comment,
56    /// A name definition.
57    Definition,
58    /// A name reference.
59    Reference,
60    /// An import declaration.
61    Import,
62    /// The content range for an injected language.
63    InjectionContent,
64    /// A language label for an injected range.
65    InjectionLanguage,
66}

68/// One syntax fact with a role and source position.
69#[derive(Debug, Clone, Copy)]
70pub struct SyntaxEvent {
71    /// The role that the language query gives this node.
72    pub role: SyntaxRole,
73    /// The node's start byte in the source.
74    pub start_byte: usize,
75    /// The node's exclusive end byte in the source.
76    pub end_byte: usize,
77    /// The zero-based row for the node start.
78    pub start_row: usize,
79    /// The zero-based row for the node end.
80    pub end_row: usize,
81    /// The byte column for the node end.
82    pub end_column: usize,
83    /// The node identity used by this syntax tree.
84    pub node_id: usize,
85    /// The parent node identity, when the node has a parent.
86    pub parent_id: Option<usize>,
87    /// True when the node has no children.
88    pub terminal: bool,
89}

91impl SyntaxEvent {
92    /// Return the half-open source byte range for this fact.
93    pub fn range(self) -> Range<usize> {
94        self.start_byte..self.end_byte
95    }
96}

98/// Assign each item to its smallest containing function; first wins equal ranges.
99/// Items outside every function are omitted. Input order is retained per function.
100pub(crate) fn assign_to_functions<'a, T: Clone + 'a>(
101    functions: &[cyclomatic::FunctionScope],
102    items: impl Iterator<Item = &'a T>,
103    range: impl Fn(&T) -> Range<usize>,
104) -> std::collections::BTreeMap<usize, Vec<T>> {
105    let mut assigned = std::collections::BTreeMap::<usize, Vec<T>>::new();
106    for item in items {
107        let range = range(item);
108        if let Some((index, _)) = functions
109            .iter()
110            .enumerate()
111            .filter(|(_, function)| {
112                function.range.start <= range.start && range.end <= function.range.end
113            })
114            .min_by_key(|(_, function)| function.range.len())
115        {
116            assigned.entry(index).or_default().push(item.clone());
117        }
118    }
119    assigned
120}
