1//! Add cognitive-complexity points for functions in recursive call cycles.
2//!
3//! The shared program graph supplies only statically proven execution edges.
4//! Symbol references retain their source identities while invocation evidence
5//! can reach callable values, accessors, and constructors. Uncertain dispatch
6//! and non-invoked values do not create cycles.
7
8use std::collections::HashSet;
9
10use petgraph::algo::kosaraju_scc;
11
12use crate::graph::ExecutionGraph;
13use crate::model::{ComplexityContribution, FileFacts};
14
15/// Add one cognitive-complexity point to each function in a resolved cycle.
16pub(crate) fn annotate(facts: &mut [FileFacts], graph: &ExecutionGraph) {
17    let recursive_nodes: HashSet<_> = kosaraju_scc(graph)
18        .into_iter()
19        .filter(|component| {
20            component.len() > 1
21                || component
22                    .first()
23                    .is_some_and(|node| graph.find_edge(*node, *node).is_some())
24        })
25        .flatten()
26        .collect();
27
28    for node in recursive_nodes {
29        let function_id = graph[node];
30        let fact = &mut facts[function_id.file_index];
31        let function = &mut fact.analysis.functions[function_id.function_index];
32        let contribution = ComplexityContribution {
33            kind: "recursion".into(),
34            value: 1,
35            start_byte: function.start_byte,
36            end_byte: function.start_byte,
37            line: memchr::memchr_iter(b'\n', &fact.source[..function.start_byte]).count() + 1,
38        };
39        function.cognitive_complexity += 1;
40        function.cognitive_contributions.push(contribution);
41        function
42            .cognitive_contributions
43            .sort_by_key(|item| (item.start_byte, item.end_byte, item.kind.clone()));
44    }
45}
