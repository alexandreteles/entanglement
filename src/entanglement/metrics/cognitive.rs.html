10/// Return true when the metric needs a syntax role.
11///
12/// Function and closure events affect nesting, even when they add no points.
13pub fn is_event(role: SyntaxRole) -> bool {
14    matches!(
15        role,
16        SyntaxRole::Function
17            | SyntaxRole::CognitiveIf
18            | SyntaxRole::CognitiveConditionBoundary
19            | SyntaxRole::CognitiveElse
20            | SyntaxRole::CognitiveElseIf
21            | SyntaxRole::CognitiveLoop
22            | SyntaxRole::CognitiveLetElse
23            | SyntaxRole::CognitiveMultiway
24            | SyntaxRole::CognitiveCatch
25            | SyntaxRole::CognitiveLogicalExpression
26            | SyntaxRole::CognitiveLogicalAnd
27            | SyntaxRole::CognitiveLogicalOr
28            | SyntaxRole::CognitiveParentheses
29            | SyntaxRole::CognitiveClosure
30            | SyntaxRole::CognitiveLabeledJump
31    )
32}

34/// Keep indexes for one syntax tree and analyze its functions.
35///
36/// Build one context per tree. Then call [`Context::analyze_functions`] to
37/// reuse its indexes across all function metrics.
38pub struct Context<'a> {
39    events: &'a [SyntaxEvent],
40    parents: &'a HashMap<usize, Option<usize>>,
41    roles: HashMap<usize, SyntaxRole>,
42    function_ranges_by_node: HashMap<usize, std::ops::Range<usize>>,
43    else_if_nodes: HashSet<usize>,
44    else_if_parent: HashMap<usize, usize>,
45    condition_parent: HashMap<usize, usize>,
46    logical_expressions: HashSet<usize>,
47    parentheses: HashSet<usize>,
48}

50impl Context<'_> {
51    /// Calculate metrics for all functions, in input order.
52    pub fn analyze_functions(
53        &self,
54        functions: &[FunctionScope],
55    ) -> Vec<(usize, Vec<ComplexityContribution>)> {
56        let mut owned = vec![Vec::<&SyntaxEvent>::new(); functions.len()];
57        let mut function_order = (0..functions.len()).collect::<Vec<_>>();
58        function_order.sort_by_key(|index| {
59            (
60                functions[*index].range.start,
61                std::cmp::Reverse(functions[*index].range.end),
62            )
63        });
64        let mut direct_events = self
65            .events
66            .iter()
67            .filter(|event| is_scored_event(event.role))
68            .collect::<Vec<_>>();
69        direct_events.sort_by_key(|event| (event.start_byte, event.end_byte));
70
71        let mut active = Vec::<usize>::new();
72        let mut next_function = 0;
73        for event in direct_events {
74            while next_function < function_order.len()
75                && functions[function_order[next_function]].range.start <= event.start_byte
76            {
77                let index = function_order[next_function];
78                while active.last().is_some_and(|active_index| {
79                    functions[*active_index].range.end <= functions[index].range.start
80                }) {
81                    active.pop();
82                }
83                active.push(index);
84                next_function += 1;
85            }
86            while active
87                .last()
88                .is_some_and(|index| !contains_range(&functions[*index].range, &event.range()))
89            {
90                active.pop();
91            }
92            if let Some(index) = active.last().copied() {
93                owned[index].push(event);
94            }
95        }
96
97        functions
98            .iter()
99            .zip(owned)
100            .map(|(function, events)| self.analyze_owned(function, &events))
101            .collect()
102    }
103
104    fn analyze_owned(
105        &self,
106        function: &FunctionScope,
107        owned_events: &[&SyntaxEvent],
108    ) -> (usize, Vec<ComplexityContribution>) {
109        let mut contributions = Vec::new();
110        for event in owned_events {
111            let (kind, value) = match event.role {
112                SyntaxRole::CognitiveIf if self.else_if_nodes.contains(&event.node_id) => ("if", 1),
113                SyntaxRole::CognitiveIf => ("if", 1 + self.nesting(event, function)),
114                SyntaxRole::CognitiveElse => ("else", 1),
115                SyntaxRole::CognitiveLoop => ("loop", 1 + self.nesting(event, function)),
116                SyntaxRole::CognitiveLetElse => ("let_else", 1 + self.nesting(event, function)),
117                SyntaxRole::CognitiveMultiway => ("match", 1 + self.nesting(event, function)),
118                SyntaxRole::CognitiveCatch => ("catch", 1 + self.nesting(event, function)),
119                SyntaxRole::CognitiveLabeledJump => ("labeled_jump", 1),
120                _ => continue,
121            };
122            contributions.push(contribution(event, kind, value));
123        }
124        contributions.extend(self.logical_contributions(owned_events));
125        contributions.sort_by_key(|item| (item.start_byte, item.end_byte, item.kind.clone()));
126        let complexity = contributions
127            .iter()
128            .map(|item| item.value.max(0) as usize)
129            .sum();
130        (complexity, contributions)
131    }
132}

134fn is_scored_event(role: SyntaxRole) -> bool {
135    matches!(
136        role,
137        SyntaxRole::CognitiveIf
138            | SyntaxRole::CognitiveElse
139            | SyntaxRole::CognitiveLoop
140            | SyntaxRole::CognitiveLetElse
141            | SyntaxRole::CognitiveMultiway
142            | SyntaxRole::CognitiveCatch
143            | SyntaxRole::CognitiveLogicalAnd
144            | SyntaxRole::CognitiveLogicalOr
145            | SyntaxRole::CognitiveLabeledJump
146    )
147}

149fn contribution(event: &SyntaxEvent, kind: &str, value: usize) -> ComplexityContribution {
150    ComplexityContribution {
151        kind: kind.into(),
152        value: value.min(i32::MAX as usize) as i32,
153        start_byte: event.start_byte,
154        end_byte: event.end_byte,
155        line: event.start_row + 1,
156    }
157}

159fn contains_range(outer: &std::ops::Range<usize>, inner: &std::ops::Range<usize>) -> bool {
160    outer.start <= inner.start && inner.end <= outer.end
161}

163/// Iterate from the immediate parent to the root of one syntax node.
164fn ancestors(
165    node: usize,
166    parents: &HashMap<usize, Option<usize>>,
167) -> impl Iterator<Item = usize> + '_ {
168    std::iter::successors(parents.get(&node).copied().flatten(), |node| {
169        parents.get(node).copied().flatten()
170    })
